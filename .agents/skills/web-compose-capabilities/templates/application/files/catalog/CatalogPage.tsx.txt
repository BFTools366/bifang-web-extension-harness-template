import { formatPrice } from "../../data/api/commerce/price"
import { Button, Card, Group, SimpleGrid, Stack, Text, Title } from "@mantine/core"
import { useQuery } from "@tanstack/react-query"
import { useState } from "react"
import { useTranslation } from "react-i18next"
import type { CatalogApi, Product } from "../../data/api/catalog/contracts"
import { ModuleState } from "../application/ModuleState"
import classes from "../application/application.module.css"

/** 商品卡片的购买槽只有完整购买依赖已接线时出现。 */
export function CatalogPage({
  api,
  scope,
  canPurchase = false,
  onPurchase
}: {
  api: CatalogApi
  scope: number
  canPurchase?: boolean
  onPurchase?: (product: Product) => void
}) {
  const { t, i18n } = useTranslation()
  const [category, setCategory] = useState<string | null>(null)
  const query = useQuery({
    queryKey: ["catalog", scope, i18n.language],
    queryFn: ({ signal }) => api.list(signal)
  })
  const categories = [
    ...new Set(
      query.data?.map((product) => product.category).filter((value): value is string => !!value)
    )
  ]
  const currentCategory = category !== null && categories.includes(category) ? category : null
  const products = query.data?.filter(
    (product) => currentCategory === null || product.category === currentCategory
  )
  return (
    <Stack>
      <Title order={1}>{t("catalogTitle")}</Title>
      <Text c="dimmed">{t("catalogDescription")}</Text>
      <ModuleState
        loading={query.isPending}
        error={query.isError}
        empty={query.data?.length === 0}
        retry={() => {
          void query.refetch()
        }}
      />
      <div className={categories.length ? classes.catalogLayout : undefined}>
        {categories.length > 0 && (
          <nav aria-label={t("catalogCategories")}>
            <Stack gap="xs">
              <Text fw={600}>{t("catalogCategories")}</Text>
              <Button
                variant={currentCategory === null ? "light" : "subtle"}
                aria-pressed={currentCategory === null}
                onClick={() => setCategory(null)}
              >
                {t("catalogAll")}
              </Button>
              {categories.map((name) => (
                <Button
                  key={name}
                  variant={currentCategory === name ? "light" : "subtle"}
                  aria-pressed={currentCategory === name}
                  onClick={() => setCategory(name)}
                >
                  {name}
                </Button>
              ))}
            </Stack>
          </nav>
        )}
        <SimpleGrid cols={{ base: 1, sm: 2, lg: 3 }}>
          {products?.map((product) => (
            <Card key={product.id} withBorder>
              <Stack>
                <Title order={2} size="h4">
                  {product.name}
                </Title>
                <Text>{product.description}</Text>
                <Group justify="space-between">
                  <Text>{formatPrice(product.price, i18n.language)}</Text>
                  {canPurchase && onPurchase && (
                    <Button onClick={() => onPurchase(product)}>{t("commercePurchase")}</Button>
                  )}
                </Group>
              </Stack>
            </Card>
          ))}
        </SimpleGrid>
      </div>
    </Stack>
  )
}
