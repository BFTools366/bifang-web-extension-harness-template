import { ActionIcon, Anchor, AppShell, Burger, Group, Image, Stack, Text } from "@mantine/core"
import { useElementSize, useMediaQuery } from "@mantine/hooks"
import { Link, Outlet, useLocation } from "@tanstack/react-router"
import { useEffect, useState, type ReactNode } from "react"
import { useTranslation } from "react-i18next"
import { BRAND } from "../../brand"
import { BUILD_VERSION } from "../../buildVersion"
import { navigation } from "./navigation"
import { ApplicationIcon } from "./ApplicationIcon"
import classes from "./application.module.css"
import "../../i18n/application"

/** 固定主界面只观察已接线的用户与导航权限，不自己拥有会话。 */
export interface ApplicationLayoutProps {
  account?: ReactNode
  allow?: (path: string) => boolean
  content?: ReactNode
}

/** 从客户端主壳提取侧栏、顶部工具区和正文插槽；单页及扩展使用紧凑导航。 */
export function ApplicationLayout({ account, allow, content }: ApplicationLayoutProps) {
  const { t } = useTranslation()
  const { pathname, href } = useLocation()
  const { ref: headerRef, height: headerHeight } = useElementSize()
  const desktop = useMediaQuery("(min-width: 48em)")
  const [collapsed, setCollapsed] = useState(false)
  const [mobileOpened, setMobileOpened] = useState(false)
  const useSidebar = true
  const compact = desktop && collapsed
  const current = navigation.find((page) => page.path === pathname)
  useEffect(() => setMobileOpened(false), [pathname])
  const visible = navigation.filter((page) => !allow || allow(page.path))
  const organizationLinks = visible
    .filter((page) => page.group === "organizationNavigation")
    .sort(
      (left, right) =>
        Number(String(left.path) !== "/organization/manage") -
        Number(String(right.path) !== "/organization/manage")
    )
  /** 每个子项仍是独立路由链接，父分组只负责展开，当前页语义留在子链接。 */
  const renderLink = (page: (typeof navigation)[number]) => (
    <Anchor
      key={page.path}
      component={Link}
      to={page.path}
      activeOptions={{ exact: true, includeSearch: false }}
      preload={false}
      className={classes.navLink}
      aria-label={t(page.title)}
      title={compact ? t(page.title) : undefined}
      underline="never"
    >
      <ApplicationIcon name="workspace" />
      {(!useSidebar || !compact) && <span>{t(page.title)}</span>}
    </Anchor>
  )
  const links = (
    <>
      {visible.filter((page) => page.group === null).map(renderLink)}
      {organizationLinks.length > 0 && (
        <details className={classes.navGroup} open>
          <summary className={classes.navLink} title={t("organizationNavigation")}>
            <ApplicationIcon name="organization" />
            {(!useSidebar || !compact) && <span>{t("organizationNavigation")}</span>}
          </summary>
          <div className={compact ? undefined : classes.navChildren}>
            {organizationLinks.map(renderLink)}
          </div>
        </details>
      )}
    </>
  )
  return (
    <AppShell
      className={classes.shell}
      layout="alt"
      header={{ height: useSidebar ? 72 : headerHeight || 112 }}
      navbar={
        useSidebar
          ? { width: compact ? 76 : 248, breakpoint: "sm", collapsed: { mobile: !mobileOpened } }
          : undefined
      }
      padding={{ base: "md", sm: "lg" }}
    >
      <a className={classes.skip} href="#application-main">
        {t("applicationSkip")}
      </a>
      <AppShell.Header>
        <div ref={headerRef} className={classes.header}>
          <Stack px="lg" gap="xs">
            <Group justify="space-between" wrap="nowrap" mih={72}>
              <Group>
                {useSidebar && (
                  <Burger
                    hiddenFrom="sm"
                    opened={mobileOpened}
                    aria-label={t("applicationNavigation")}
                    aria-expanded={mobileOpened}
                    onClick={() => setMobileOpened((value) => !value)}
                  />
                )}
                <Text fw={600} visibleFrom="sm">
                  {t(current?.title ?? "projectName")}
                </Text>
              </Group>
              <Group>
                {!useSidebar && (
                  <ActionIcon
                    variant="default"
                    size="lg"
                    aria-label={t("applicationSettings")}
                    renderRoot={(props) => (
                      <Link {...props} to="/settings" state={{ settingsOrigin: href }} />
                    )}
                  >
                    <ApplicationIcon name="settings" />
                  </ActionIcon>
                )}
                {account}
              </Group>
            </Group>
            {!useSidebar && (
              <nav aria-label={t("applicationNavigation")}>
                <Group gap="xs">{links}</Group>
              </nav>
            )}
          </Stack>
        </div>
      </AppShell.Header>
      {useSidebar && (
        <AppShell.Navbar p="md" className={classes.sidebar} aria-label={t("applicationSidebar")}>
          <Anchor
            component={Link}
            to="/"
            preload={false}
            className={classes.brand}
            underline="never"
            aria-label={t("projectName")}
          >
            <Image src={BRAND.logoPath} w={36} h={36} alt="" />
            {!compact && (
              <div>
                <Text fw={700}>{t("projectName")}</Text>
                <Text size="xs" c="dimmed">
                  {t("applicationSubtitle")}
                </Text>
              </div>
            )}
          </Anchor>
          <nav className={classes.nav} aria-label={t("applicationNavigation")}>
            {links}
          </nav>

          <Group className={classes.sidebarBottom} justify={compact ? "center" : "space-between"}>
            <Anchor
              renderRoot={(props) => (
                <Link {...props} to="/settings" state={{ settingsOrigin: href }} />
              )}
              aria-label={t("applicationSettings")}
              title={compact ? t("applicationSettings") : undefined}
              className={classes.navLink}
              underline="never"
            >
              <ApplicationIcon name="settings" />
              {!compact && <span>{t("applicationSettings")}</span>}
            </Anchor>
            <ActionIcon
              variant="subtle"
              visibleFrom="sm"
              aria-label={t(collapsed ? "applicationExpand" : "applicationCollapse")}
              aria-expanded={!collapsed}
              onClick={() => setCollapsed((value) => !value)}
            >
              <ApplicationIcon name={collapsed ? "expand" : "collapse"} />
            </ActionIcon>
          </Group>
        </AppShell.Navbar>
      )}
      <AppShell.Main id="application-main" tabIndex={-1} className={classes.main}>
        <Stack gap="lg">
          {content ?? <Outlet />}
          <footer>
            <Text size="xs" c="dimmed">
              {t("buildVersion", { version: BUILD_VERSION })}
            </Text>
          </footer>
        </Stack>
      </AppShell.Main>
    </AppShell>
  )
}
