import type { TeamApi } from "./teamContracts"

/** 团队命令复用当前账号及代次，授权不根据角色名称判断。 */
export function createTeamService(
  api: TeamApi,
  owner: {
    account(): string | null
    scope(): number
    blocked(): boolean
    allow(codes: readonly string[]): boolean
    refresh(): Promise<void>
  }
): TeamApi {
  /** 先检查权限再写，成功仅在原作用域内刷新；未知命令不重放。 */
  async function write(action: () => Promise<void>, code?: string) {
    const account = owner.account(),
      scope = owner.scope()
    const group = code?.split(".")[1]
    const required = code
      ? [code, ...(group && group !== "leave" ? [`team.${group}.read`] : [])]
      : []
    if (!account || owner.blocked() || (required.length > 0 && !owner.allow(required)))
      throw new Error("team_denied")
    await action()
    if (account !== owner.account() || scope !== owner.scope()) throw new Error("team_superseded")
    await owner.refresh()
    if (account !== owner.account() || scope !== owner.scope()) throw new Error("team_superseded")
  }
  return {
    ...api,
    invite: (org, email, signal) =>
      write(() => api.invite(org, email, signal), "team.members.invite"),
    resolve: (invitation, action, signal) =>
      write(
        () => api.resolve(invitation, action, signal),
        action === "revoke" ? "team.members.invite" : undefined
      ),
    remove: (member, signal) =>
      write(
        () => api.remove(member, signal),
        member.accountId === owner.account() ? "team.leave" : "team.members.remove"
      ),
    nickname: (member, name, signal) =>
      write(() => api.nickname(member, name, signal), "team.members.nickname.update"),
    assignRoles: (member, roles, signal) =>
      write(() => api.assignRoles(member, roles, signal), "team.members.roles.assign"),
    createRole: (org, code, name, codes, signal) =>
      write(() => api.createRole(org, code, name, codes, signal), "team.roles.create"),
    updateRole: (role, name, codes, signal) =>
      write(() => api.updateRole(role, name, codes, signal), "team.roles.permissions.update"),
    deleteRole: (role, signal) => write(() => api.deleteRole(role, signal), "team.roles.delete")
  }
}
