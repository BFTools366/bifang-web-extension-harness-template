import { Alert, NativeSelect, Stack, Title } from "@mantine/core"
import { useQuery } from "@tanstack/react-query"
import { useState } from "react"
import { useTranslation } from "react-i18next"
import type { OrganizationApi } from "../../data/api/organization/contracts"
import { ModuleState } from "../application/ModuleState"
import { organizationQueryOptions } from "./OrganizationContext"

/** 固定组织页只接线当前组织和允许切换项；写入未知先重读。 */
export function OrganizationPage({
  api,
  scope,
  changed,
  canSwitch
}: {
  api: OrganizationApi
  scope: number
  changed(): Promise<void>
  canSwitch: boolean
}) {
  const { t, i18n } = useTranslation()
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState(false)
  const query = useQuery(organizationQueryOptions(api, scope, i18n.language))
  /** 先复核动作授权，切换成功后使旧作用域缓存与权限失效。 */
  async function select(id: string) {
    if (busy || !canSwitch || id === query.data?.current) return
    setBusy(true)
    setError(false)
    try {
      await api.switch(id)
      await changed()
    } catch {
      setError(true)
    } finally {
      setBusy(false)
    }
  }
  return (
    <Stack>
      <Title order={1}>{t("organizationTitle")}</Title>
      <ModuleState
        loading={query.isPending}
        error={query.error}
        empty={query.data?.available.length === 0}
        retry={() => {
          void query.refetch()
        }}
      />
      {error && <Alert role="alert">{t("applicationError")}</Alert>}
      {query.data && (
        <NativeSelect
          label={t("organizationCurrent")}
          disabled={
            busy || !canSwitch || !query.data.available.some((row) => row.id !== query.data.current)
          }
          value={query.data.current ?? ""}
          data={[
            ...(query.data.current === null
              ? [{ value: "", label: t("organizationNone"), disabled: true }]
              : []),
            ...query.data.available.map((row) => ({
              value: row.id,
              label: row.name,
              disabled: row.id === query.data?.current
            }))
          ]}
          onChange={(event) => {
            void select(event.currentTarget.value).catch(() => undefined)
          }}
        />
      )}
    </Stack>
  )
}
