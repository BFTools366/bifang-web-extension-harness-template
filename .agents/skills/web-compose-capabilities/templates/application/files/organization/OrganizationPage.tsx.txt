import { Button, Stack } from "@mantine/core"
import { useState, type ReactNode } from "react"
import { useTranslation } from "react-i18next"
import type { OrganizationApi } from "../../data/api/organization/contracts"
import { identitySession } from "../application/runtime"
import { OrganizationSwitchDialog } from "./OrganizationSwitchDialog"

/** 拒绝访问后的恢复 URL 仍使用同一切换弹窗，不提供个人空间或第二套选择页。 */
export function OrganizationPage({
  api,
  scope,
  blocked = false,
  canSwitch = true,
  changed,
  invitations
}: {
  api: OrganizationApi
  scope: number
  blocked?: boolean
  canSwitch?: boolean
  invitations?: ReactNode
  changed(): Promise<void>
}) {
  const { t } = useTranslation()
  const [opened, setOpened] = useState(canSwitch)
  return (
    <Stack>
      <Button disabled={!canSwitch} onClick={() => setOpened(true)}>
        {t("organizationSwitch")}
      </Button>
      {opened && (
        <OrganizationSwitchDialog
          api={api}
          scope={scope}
          blocked={blocked}
          accountId={identitySession.getSnapshot().identity?.id ?? null}
          changed={changed}
          invitations={invitations}
          close={() => setOpened(false)}
        />
      )}
    </Stack>
  )
}
