import type {
  OrganizationApi,
  OrganizationManagementApi,
  OrganizationMembership
} from "./contracts"

/** 团队成员 ID 是授权主体，不能以全局账号 ID 替代。 */
export interface TeamMember {
  id: string
  accountId: string
  organizationId: string
  displayName: string
  role: "owner" | "member"
  status: "active"
  version: number
  permissions: readonly string[]
  roleIds: readonly string[]
  roleNames: readonly string[]
  canRemove: boolean
  canAssignRoles: boolean
}
/** 角色只归一个团队，系统角色保护属于对象状态，不是前端授权依据。 */
export interface TeamRole {
  id: string
  organizationId: string
  code: string
  name: string
  protected: boolean
  canManage: boolean
  version: number
  permissions: readonly string[]
}
/** 邀请绑定账号、团队和签发版本；Unix 秒时间由服务校验。 */
export interface TeamInvitation {
  id: string
  organizationId: string
  organizationName: string
  accountId: string
  status: "pending" | "accepted" | "rejected" | "revoked" | "expired"
  version: number
  expiresAt: number
}
/** 所有集合采用有界分页，不把第一批结果当作全部成员或邀请。 */
export interface TeamPage<T> {
  items: readonly T[]
  total: number
}
/** 完整团队上下文保留成员版本，供退出和并发管理使用。 */
export interface TeamContext {
  current: string
  available: readonly (Omit<OrganizationMembership, "user"> & { user: TeamMember })[]
}
/** 基线合同复用当前组织 owner 和 HTTP 传输，不保存第二套身份。 */
export interface TeamApi extends OrganizationApi, OrganizationManagementApi {
  read(signal?: AbortSignal): Promise<TeamContext>
  members(
    organizationId: string,
    page: number,
    pageSize: number,
    signal?: AbortSignal
  ): Promise<TeamPage<TeamMember>>
  invitations(
    organizationId: string | null,
    page: number,
    pageSize: number,
    signal?: AbortSignal
  ): Promise<TeamPage<TeamInvitation>>
  permissionCatalog(organizationId: string, signal?: AbortSignal): Promise<readonly string[]>
  nickname(member: TeamMember, nickname: string, signal?: AbortSignal): Promise<void>
  assignableRoles(organizationId: string, signal?: AbortSignal): Promise<TeamPage<TeamRole>>
  invite(organizationId: string, email: string, signal?: AbortSignal): Promise<void>
  resolve(
    invitation: TeamInvitation,
    action: "accept" | "reject" | "revoke",
    signal?: AbortSignal
  ): Promise<void>
  remove(member: TeamMember, signal?: AbortSignal): Promise<void>
  roles(
    organizationId: string,
    page: number,
    pageSize: number,
    signal?: AbortSignal
  ): Promise<TeamPage<TeamRole>>
  assignRoles(member: TeamMember, roleIds: readonly string[], signal?: AbortSignal): Promise<void>
  createRole(
    organizationId: string,
    code: string,
    name: string,
    permissions: readonly string[],
    signal?: AbortSignal
  ): Promise<void>
  updateRole(
    role: TeamRole,
    name: string,
    permissions: readonly string[],
    signal?: AbortSignal
  ): Promise<void>
  deleteRole(role: TeamRole, signal?: AbortSignal): Promise<void>
}
