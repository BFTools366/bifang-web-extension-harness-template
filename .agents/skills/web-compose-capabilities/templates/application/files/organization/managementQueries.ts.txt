import { queryOptions } from "@tanstack/react-query"
import { api } from "../../data/api/organization/adapter"
import type { PageRequest } from "./managementPagination"

/** 所有组织管理查询带身份、组织、成员、代次与语言，撤权统一清理。 */
export function managementQueryKey(owner: string, language: string) {
  return ["organization-management", owner, language] as const
}
/** 成员列表查询保留有界分页和成员投影。 */
export function membersQuery(owner: string, org: string, language: string, page: PageRequest) {
  return queryOptions({
    queryKey: [...managementQueryKey(owner, language), "members", page.page, page.pageSize],
    meta: { private: true },
    staleTime: 0,
    retry: false,
    queryFn: ({ signal }) => api.members(org, page.page, page.pageSize, signal)
  })
}
