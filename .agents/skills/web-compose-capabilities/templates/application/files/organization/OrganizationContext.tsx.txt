import { Anchor, Stack, Text } from "@mantine/core"
import { Link } from "@tanstack/react-router"
import { queryOptions, useQuery } from "@tanstack/react-query"
import { useTranslation } from "react-i18next"
import type { OrganizationReadApi } from "../../data/api/organization/contracts"
import { ApplicationIcon } from "../application/ApplicationIcon"

/** 当前组织在账号菜单、侧栏和选择页共用同一份私有上下文缓存。 */
export function organizationQueryOptions(
  api: OrganizationReadApi,
  scope: number,
  language: string
) {
  return queryOptions({
    queryKey: ["organization", scope, language],
    meta: { private: true },
    queryFn: ({ signal }) => api.read(signal),
    staleTime: 10000,
    retry: false
  })
}

/** 选择、加载与失败分开表达，不把待加载的当前组织误报为未选择。 */
function useOrganizationName(api: OrganizationReadApi, scope: number, blocked: boolean) {
  const { t, i18n } = useTranslation()
  const query = useQuery({
    ...organizationQueryOptions(api, scope, i18n.language),
    enabled: !blocked
  })
  if (blocked) return t("organizationSwitching")
  if (query.isError) return t("organizationUnavailable")
  if (!query.data) return t("applicationLoading")
  if (query.data.current === null) return t("organizationNone")
  return (
    query.data.available.find((row) => row.id === query.data.current)?.name ??
    t("organizationUnavailable")
  )
}

/** 侧栏在收起后保留有名称的当前组织入口，无页面权限时只展示上下文。 */
export function OrganizationContext({
  api,
  scope,
  blocked,
  compact,
  canOpen
}: {
  api: OrganizationReadApi
  scope: number
  blocked: boolean
  compact: boolean
  canOpen: boolean
}) {
  const { t } = useTranslation()
  const name = useOrganizationName(api, scope, blocked)
  const content = compact ? <ApplicationIcon name="organization" /> : name
  return (
    <Stack gap="xs" miw={0}>
      {!compact && (
        <Text size="xs" c="dimmed">
          {t("organizationCurrent")}
        </Text>
      )}
      {canOpen && !blocked ? (
        <Anchor
          renderRoot={(props) => <Link {...props} to="/organization" preload={false} />}
          aria-label={t("organizationContext", { name })}
          title={name}
          size="sm"
          fw={600}
        >
          {content}
        </Anchor>
      ) : (
        <Text size="sm" title={name} aria-label={name}>
          {content}
        </Text>
      )}
    </Stack>
  )
}
