import {
  Alert,
  Button,
  Group,
  MultiSelect,
  Pagination,
  Stack,
  Text,
  TextInput,
  Title
} from "@mantine/core"
import { useQuery } from "@tanstack/react-query"
import { useState } from "react"
import { useTranslation } from "react-i18next"
import type { TeamApi, TeamMember, TeamRole } from "../../data/api/organization/teamContracts"
import { ModuleState } from "../application/ModuleState"
import { useTeamMutation } from "./useTeamMutation"
import { TeamRoles } from "./TeamRoles"

/** 单个成员授权草稿只属于当前实体版本，服务复核版本和团队后提交。 */
function MemberRoles({
  member,
  available,
  disabled,
  save,
  remove
}: {
  member: TeamMember
  available: readonly TeamRole[]
  disabled: boolean
  save(roleIds: readonly string[]): void
  remove(): void
}) {
  const { t } = useTranslation()
  const custom = available.filter((role) => !role.protected)
  const [roleIds, setRoleIds] = useState(
    member.roleIds.filter((id) => custom.some((role) => role.id === id))
  )
  return (
    <Stack gap="xs">
      <Text>{member.displayName}</Text>
      {member.canAssignRoles && (
        <>
          <MultiSelect
            label={t("teamAssignedRoles")}
            data={custom.map((role) => ({ value: role.id, label: role.name }))}
            value={roleIds}
            onChange={setRoleIds}
            disabled={disabled}
          />
          <Group>
            <Button disabled={disabled} onClick={() => save(roleIds)}>
              {t("teamSaveRoles")}
            </Button>
            <Button
              color="red"
              variant="subtle"
              disabled={disabled || !member.canRemove}
              onClick={remove}
            >
              {t("teamRemove")}
            </Button>
          </Group>
        </>
      )}
    </Stack>
  )
}
/** 成员管理只挂载在权限 code 门禁之内，成员、目录和发出邀请均由私有 Query 拥有。 */
export function TeamMembers({
  api,
  organizationId,
  scope,
  isCurrent
}: {
  api: TeamApi
  organizationId: string
  scope: number
  isCurrent(): boolean
}) {
  const { t, i18n } = useTranslation()
  const [page, setPage] = useState(1),
    [invitePage, setInvitePage] = useState(1),
    [email, setEmail] = useState("")
  const members = useQuery({
    queryKey: ["team", scope, organizationId, "members", page, i18n.language],
    meta: { private: true },
    retry: false,
    queryFn: ({ signal }) => api.members(organizationId, page, 20, signal)
  })
  const permissions = useQuery({
    queryKey: ["team", scope, organizationId, "catalog", i18n.language],
    meta: { private: true },
    retry: false,
    queryFn: ({ signal }) => api.permissionCatalog(organizationId, signal)
  })
  const invitations = useQuery({
    queryKey: ["team", scope, organizationId, "invitations", invitePage, i18n.language],
    meta: { private: true },
    retry: false,
    queryFn: ({ signal }) => api.invitations(organizationId, invitePage, 20, signal)
  })
  const roles = useQuery({
    queryKey: ["team", scope, organizationId, "roles", i18n.language],
    meta: { private: true },
    retry: false,
    queryFn: ({ signal }) => api.roles(organizationId, 1, 100, signal)
  })
  const mutation = useTeamMutation(isCurrent, async () => {
    await Promise.all([
      members.refetch({ throwOnError: true }),
      permissions.refetch({ throwOnError: true }),
      invitations.refetch({ throwOnError: true }),
      roles.refetch({ throwOnError: true })
    ])
  })
  return (
    <Stack>
      <Title order={2}>{t("teamMembers")}</Title>
      <ModuleState
        loading={members.isPending || permissions.isPending || roles.isPending}
        error={members.error ?? permissions.error ?? roles.error}
        empty={members.data?.total === 0}
        retry={() => {
          void mutation.verify()
        }}
      />
      {mutation.status === "unknown" && (
        <Alert role="alert">
          {t("teamUnknown")}
          <Button
            onClick={() => {
              void mutation.verify()
            }}
          >
            {t("teamVerify")}
          </Button>
        </Alert>
      )}
      {mutation.status === "saved" && <Text role="status">{t("teamUpdated")}</Text>}
      {members.data &&
        roles.data &&
        permissions.data &&
        members.data.items.map((member) => (
          <MemberRoles
            key={
              member.id +
              ":" +
              member.version +
              ":" +
              roles.data.items.map((role) => role.id + ":" + role.version).join(",") +
              ":" +
              permissions.data.join(",")
            }
            member={member}
            available={roles.data.items.filter((role) =>
              role.permissions.every((code) => permissions.data?.includes(code))
            )}
            disabled={mutation.disabled || !isCurrent()}
            save={(roleIds) => {
              void mutation.run(() => api.assignRoles(member, roleIds))
            }}
            remove={() => {
              void mutation.run(() => api.remove(member))
            }}
          />
        ))}
      {roles.data && permissions.data && (
        <TeamRoles
          api={api}
          organizationId={organizationId}
          roles={roles.data.items}
          permissions={permissions.data}
          disabled={mutation.disabled || !isCurrent()}
          run={mutation.run}
        />
      )}
      {(members.data?.total ?? 0) > 20 && (
        <Pagination
          aria-label={t("teamMemberPages")}
          value={page}
          onChange={setPage}
          total={Math.ceil((members.data?.total ?? 0) / 20)}
          disabled={mutation.disabled}
        />
      )}
      <form
        onSubmit={(event) => {
          event.preventDefault()
          if (email.trim()) void mutation.run(() => api.invite(organizationId, email.trim()))
        }}
      >
        <Stack>
          <TextInput
            type="email"
            label={t("teamInviteEmail")}
            placeholder={t("teamInvitePlaceholder")}
            required
            value={email}
            disabled={mutation.disabled}
            onChange={(event) => setEmail(event.currentTarget.value)}
          />
          <Button type="submit" disabled={mutation.disabled || !email.trim()}>
            {t("teamInvite")}
          </Button>
        </Stack>
      </form>
      <ModuleState
        loading={invitations.isPending}
        error={invitations.error}
        empty={false}
        retry={() => {
          void mutation.verify()
        }}
      />
      {invitations.data?.items.map((invitation) => (
        <Group key={invitation.id}>
          <Text>
            {invitation.accountId} · {t("teamStatus_" + invitation.status)}
          </Text>
          {invitation.status === "pending" && (
            <Button
              variant="subtle"
              disabled={mutation.disabled}
              onClick={() => {
                void mutation.run(() => api.resolve(invitation, "revoke"))
              }}
            >
              {t("teamRevoke")}
            </Button>
          )}
        </Group>
      ))}
      {(invitations.data?.total ?? 0) > 20 && (
        <Pagination
          aria-label={t("teamInvitationPages")}
          value={invitePage}
          onChange={setInvitePage}
          total={Math.ceil((invitations.data?.total ?? 0) / 20)}
          disabled={mutation.disabled}
        />
      )}
    </Stack>
  )
}
