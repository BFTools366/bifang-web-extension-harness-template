import { Menu } from "@mantine/core"
import { useTranslation } from "react-i18next"

/** 组织切换只作为用户下拉中的动作入口，组织列表由弹窗读取。 */
export function OrganizationMenu({ blocked, open }: { blocked: boolean; open(): void }) {
  const { t } = useTranslation()
  return (
    <>
      <Menu.Item disabled={blocked} onClick={open} title={t("organizationSwitch")}>
        {t("organizationSwitch")}
      </Menu.Item>
    </>
  )
}
