import { Menu } from "@mantine/core"
import { useQuery } from "@tanstack/react-query"
import { useTranslation } from "react-i18next"
import type { OrganizationReadApi } from "../../data/api/organization/contracts"
import { ApplicationIcon } from "../application/ApplicationIcon"
import { organizationQueryOptions } from "./OrganizationContext"

/** 当前组织名称本身就是切换入口；管理与加入由应用导航拥有。 */
export function OrganizationMenu({
  api,
  scope,
  blocked,
  canSwitch,
  open
}: {
  api: OrganizationReadApi
  scope: number
  blocked: boolean
  canSwitch: boolean
  open(): void
}) {
  const { t, i18n } = useTranslation()
  const query = useQuery({
    ...organizationQueryOptions(api, scope, i18n.language),
    enabled: !blocked
  })
  const current = query.data?.available.find((row) => row.id === query.data.current)
  return (
    <>
      <Menu.Item
        leftSection={<ApplicationIcon name="organization" />}
        disabled={blocked || !canSwitch || !current}
        onClick={open}
        title={t("organizationSwitch")}
      >
        {blocked ? t("organizationSwitching") : (current?.name ?? t("organizationUnavailable"))}
      </Menu.Item>
    </>
  )
}
