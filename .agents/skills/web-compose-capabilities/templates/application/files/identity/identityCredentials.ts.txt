import type { CredentialStorage } from "../../features/identity/session"

/** 默认凭据只属于当前执行上下文；持久化与跨入口共享另按真实宿主合同适配。 */
export function createIdentityCredentials(): CredentialStorage {
  let value: string | null = null
  return {
    read: () => value,
    /** 仅在原凭据精确匹配时替换，阻止旧会话回写。 */
    compareAndSet(expected, next) {
      if (value !== expected) return false
      value = next
      return true
    }
  }
}
