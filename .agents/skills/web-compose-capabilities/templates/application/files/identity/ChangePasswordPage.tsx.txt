import { Alert, Button, PasswordInput, Stack, Title } from "@mantine/core"
import { useEffect, useRef, useState, useSyncExternalStore, type FormEvent } from "react"
import { useTranslation } from "react-i18next"
import { ApiError } from "../../data/api/client/httpClient"
import type { PersonalSettingsApi } from "../../data/api/identity/personalSettings"
import type { IdentityRuntime } from "./runtime"

/** 改密独立于资料和组织读取，账号变化时销毁原有密码输入。 */
export function ChangePasswordPage({
  api,
  session,
  onPasswordChanged,
  embedded = false,
  onCloseBlockedChange
}: {
  api: PersonalSettingsApi
  session: IdentityRuntime
  onPasswordChanged(): Promise<void>
  embedded?: boolean
  onCloseBlockedChange?(blocked: boolean): void
}) {
  const snapshot = useSyncExternalStore(session.subscribe, session.getSnapshot, session.getSnapshot)
  return (
    <ChangePasswordForm
      key={snapshot.identity?.id}
      api={api}
      session={session}
      onPasswordChanged={onPasswordChanged}
      embedded={embedded}
      onCloseBlockedChange={onCloseBlockedChange}
    />
  )
}

/** 防止重复提交和迟到结果；未知改密结果只能通过重新登录核对。 */
function ChangePasswordForm({
  api,
  session,
  onPasswordChanged,
  embedded,
  onCloseBlockedChange
}: {
  api: PersonalSettingsApi
  session: IdentityRuntime
  onPasswordChanged(): Promise<void>
  embedded?: boolean
  onCloseBlockedChange?(blocked: boolean): void
}) {
  const { t } = useTranslation()
  const visibilityToggleButtonProps = { "aria-label": t("personalPasswordVisibility") }
  const accountId = session.getSnapshot().identity?.id
  const [oldPassword, setOldPassword] = useState("")
  const [newPassword, setNewPassword] = useState("")
  const [confirmation, setConfirmation] = useState("")
  const [busy, setBusy] = useState(false)
  const [status, setStatus] = useState("")
  const [unknown, setUnknown] = useState(false)
  useEffect(() => {
    onCloseBlockedChange?.(busy || unknown)
    return () => onCloseBlockedChange?.(false)
  }, [busy, unknown, onCloseBlockedChange])
  const confirmationInput = useRef<HTMLInputElement>(null)
  const lock = useRef(false)
  const controller = useRef(new AbortController())
  useEffect(() => {
    const owned = new AbortController()
    controller.current = owned
    return () => owned.abort()
  }, [])
  /** 服务确认成功后清空输入并撤下旧会话，后续跳转不再重发改密。 */
  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()
    if (lock.current || unknown || !accountId || !api.changePassword) return
    if (newPassword !== confirmation) {
      confirmationInput.current?.focus()
      return
    }
    lock.current = true
    setBusy(true)
    setStatus("")
    const signal = controller.current.signal
    const current = () => !signal.aborted && session.getSnapshot().identity?.id === accountId
    try {
      await api.changePassword(oldPassword, newPassword, signal)
      if (!current()) return
      setOldPassword("")
      setNewPassword("")
      setConfirmation("")
      session.invalidate()
      await onPasswordChanged()
    } catch (error) {
      if (!current()) return
      const uncertain =
        !(error instanceof ApiError) ||
        ["network", "timeout", "invalid-response"].includes(error.kind) ||
        (error.status !== undefined && error.status >= 500)
      setUnknown(uncertain)
      setStatus(uncertain ? "personalPasswordUnknown" : "personalPasswordFailed")
    } finally {
      lock.current = false
      if (current()) setBusy(false)
    }
  }
  /** 不自动重放结果未知的改密，只清除旧会话并返回登录。 */
  async function recover() {
    if (lock.current || !accountId || session.getSnapshot().identity?.id !== accountId) return
    lock.current = true
    setBusy(true)
    try {
      session.invalidate()
      await onPasswordChanged()
    } finally {
      lock.current = false
    }
  }
  return (
    <Stack maw={embedded ? undefined : 480}>
      {!embedded && <Title order={1}>{t("personalPassword")}</Title>}
      {status && (
        <Alert role="status">
          {t(status)}
          {unknown && (
            <Button
              variant="subtle"
              loading={busy}
              onClick={() => {
                void recover().catch(() => undefined)
              }}
            >
              {t("identityLogin")}
            </Button>
          )}
        </Alert>
      )}
      <form
        onSubmit={(event) => {
          void submit(event)
        }}
      >
        <Stack>
          <PasswordInput
            visibilityToggleButtonProps={visibilityToggleButtonProps}
            label={t("personalOldPassword")}
            placeholder={t("personalOldPasswordPlaceholder")}
            name="current-password"
            autoComplete="current-password"
            value={oldPassword}
            required
            disabled={busy || unknown}
            onChange={(event) => setOldPassword(event.currentTarget.value)}
          />
          <PasswordInput
            visibilityToggleButtonProps={visibilityToggleButtonProps}
            label={t("personalNewPassword")}
            placeholder={t("personalNewPasswordPlaceholder")}
            name="new-password"
            autoComplete="new-password"
            value={newPassword}
            required
            disabled={busy || unknown}
            onChange={(event) => setNewPassword(event.currentTarget.value)}
          />
          <PasswordInput
            visibilityToggleButtonProps={visibilityToggleButtonProps}
            ref={confirmationInput}
            label={t("registrationConfirm")}
            placeholder={t("registrationConfirmPlaceholder")}
            name="confirm-password"
            autoComplete="new-password"
            value={confirmation}
            required
            disabled={busy || unknown}
            error={
              confirmation && newPassword !== confirmation
                ? t("registrationPasswordMismatch")
                : undefined
            }
            onChange={(event) => setConfirmation(event.currentTarget.value)}
          />
          <Button
            type="submit"
            loading={busy}
            disabled={unknown || !accountId || !api.changePassword}
          >
            {t("personalPassword")}
          </Button>
        </Stack>
      </form>
    </Stack>
  )
}
