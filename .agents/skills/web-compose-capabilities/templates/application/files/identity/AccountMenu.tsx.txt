import type { ReactNode, Ref } from "react"
import { IdentityMenu } from "../identity/IdentityMenu"
import type { IdentityRuntime } from "../identity/runtime"

/** 账号菜单只组合真实身份与组织入口；外观和语言统一由独立设置页拥有。 */
export function AccountMenu({
  session,
  scope,
  avatar,
  triggerRef,
  organization,
  settings
}: {
  session: IdentityRuntime
  scope: number
  avatar?: string
  triggerRef?: Ref<HTMLButtonElement>
  organization?: ReactNode
  settings?: ReactNode
}) {
  return (
    <IdentityMenu session={session} avatar={avatar} scope={scope} triggerRef={triggerRef}>
      {organization}
      {settings}
    </IdentityMenu>
  )
}
