import { Alert, Button, Stack, Text } from "@mantine/core"
import { useEffect, useSyncExternalStore, type ReactNode } from "react"
import { useTranslation } from "react-i18next"
import type { IdentityRuntime } from "./runtime"

/** 私有正文在会话失效的同一次渲染中撤下，导航由宿主适配。 */
export function IdentityBoundary({
  session,
  onAnonymous,
  children
}: {
  session: IdentityRuntime
  onAnonymous(): void
  children: ReactNode
}) {
  const { t } = useTranslation()
  const state = useSyncExternalStore(session.subscribe, session.getSnapshot, session.getSnapshot)
  useEffect(() => {
    if (state.phase === "anonymous") onAnonymous()
  }, [state.phase, onAnonymous])
  if (state.identity && ["authenticated", "checking"].includes(state.phase)) return children
  if (state.phase === "error")
    return (
      <Alert role="alert">
        <Stack>
          <Text>{t("applicationError")}</Text>
          <Button
            onClick={() => {
              void session.ensure().catch(() => undefined)
            }}
          >
            {t("applicationRetry")}
          </Button>
        </Stack>
      </Alert>
    )
  return <Text role="status">{t("applicationLoading")}</Text>
}
