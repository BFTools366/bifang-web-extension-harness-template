import { Alert, Button, PasswordInput, Stack, Text, TextInput } from "@mantine/core"
import { useEffect, useRef, useState, type FormEvent, type ReactNode } from "react"
import { useTranslation } from "react-i18next"
import { AuthLayout } from "./AuthLayout"
import classes from "../application/application.module.css"
import type { IdentityRuntime } from "./runtime"

/** 固定登录页只消费基础会话，成功目的地由宿主路由接线提供。 */
export interface LoginPageProps {
  session: IdentityRuntime
  mock: boolean
  registrationLink?: ReactNode
  onSuccess(): Promise<void>
}
/** 登录页固定设计为品牌标题、凭据表单与稳定错误反馈。 */
export function LoginPage({ session, mock, onSuccess, registrationLink }: LoginPageProps) {
  const { t } = useTranslation()
  const [account, setAccount] = useState("")
  const [password, setPassword] = useState("")
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState(false)
  const submitting = useRef(false)
  const alive = useRef(true)
  const controller = useRef(new AbortController())
  useEffect(() => {
    const owned = new AbortController()
    controller.current = owned
    alive.current = true
    return () => {
      alive.current = false
      owned.abort()
    }
  }, [])
  /** 同步阻止重复提交，只在接受本次登录后执行安全导航。 */
  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()
    if (submitting.current) return
    submitting.current = true
    setBusy(true)
    setError(false)
    try {
      await session.login({ account, password }, controller.current.signal)
      if (!alive.current) return
      setPassword("")
      await onSuccess()
    } catch {
      if (alive.current) setError(true)
    } finally {
      submitting.current = false
      if (alive.current) setBusy(false)
    }
  }
  return (
    <AuthLayout title={t("identityLogin")}>
      {mock && (
        <details>
          <summary>{t("identityMockTitle")}</summary>
          <Text size="sm">{t("identityMock")}</Text>
        </details>
      )}
      {error && (
        <Alert color="red" role="alert">
          {t("identityLoginError")}
        </Alert>
      )}
      <form onSubmit={submit}>
        <Stack>
          <TextInput
            label={t("identityAccount")}
            placeholder={t("identityAccountPlaceholder")}
            classNames={{ input: classes.authPlaceholder }}
            name="account"
            autoComplete="username"
            value={account}
            onChange={(event) => setAccount(event.currentTarget.value)}
            required
          />
          <PasswordInput
            label={t("identityPassword")}
            placeholder={t("identityPasswordPlaceholder")}
            classNames={{ innerInput: classes.authPlaceholder }}
            name="password"
            autoComplete="current-password"
            value={password}
            onChange={(event) => setPassword(event.currentTarget.value)}
            required
          />
          <Button type="submit" loading={busy} disabled={busy}>
            {t("identityLogin")}
          </Button>
        </Stack>
      </form>
      {registrationLink}
    </AuthLayout>
  )
}
