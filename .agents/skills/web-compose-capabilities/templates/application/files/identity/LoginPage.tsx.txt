import {
  Alert,
  Button,
  Container,
  Group,
  Image,
  Paper,
  PasswordInput,
  Stack,
  Text,
  TextInput,
  Title
} from "@mantine/core"
import { useEffect, useRef, useState, type FormEvent } from "react"
import { useTranslation } from "react-i18next"
import { BRAND } from "../../brand"
import { Preferences } from "../application/Preferences"
import classes from "../application/application.module.css"
import type { IdentityRuntime } from "./runtime"

/** 固定登录页只消费基础会话，成功目的地由宿主路由接线提供。 */
export interface LoginPageProps {
  session: IdentityRuntime
  mock: boolean
  onSuccess(): Promise<void>
}
/** 登录页固定设计为品牌标题、凭据表单与稳定错误反馈。 */
export function LoginPage({ session, mock, onSuccess }: LoginPageProps) {
  const { t } = useTranslation()
  const [account, setAccount] = useState("")
  const [password, setPassword] = useState("")
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState(false)
  const submitting = useRef(false)
  const alive = useRef(true)
  useEffect(() => {
    alive.current = true
    return () => {
      alive.current = false
    }
  }, [])
  /** 同步阻止重复提交，只在接受本次登录后执行安全导航。 */
  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()
    if (submitting.current) return
    submitting.current = true
    setBusy(true)
    setError(false)
    try {
      await session.login({ account, password })
      if (!alive.current) return
      setPassword("")
      await onSuccess()
    } catch {
      if (alive.current) setError(true)
    } finally {
      submitting.current = false
      if (alive.current) setBusy(false)
    }
  }
  return (
    <main className={classes.authPage}>
      <Container size="lg" w="100%">
        <Paper withBorder className={classes.authLayout}>
          <Stack className={classes.authAside} gap="lg">
            <Group>
              <Image src={BRAND.logoPath} alt="" w={48} h={48} />
              <Text fw={700} size="xl">
                {t("projectName")}
              </Text>
            </Group>
            <Text size="xl" fw={700}>
              {t("identityIntroTitle")}
            </Text>
            <Text c="dimmed">{t("identityIntroDescription")}</Text>
          </Stack>
          <Stack className={classes.authCard}>
            <Title order={1} className={classes.authHeading}>
              {t("identityLogin")}
            </Title>
            {mock && (
              <details>
                <summary>{t("identityMockTitle")}</summary>
                <Text size="sm">{t("identityMock")}</Text>
              </details>
            )}
            {error && (
              <Alert color="red" role="alert">
                {t("identityLoginError")}
              </Alert>
            )}
            <form onSubmit={submit}>
              <Stack>
                <TextInput
                  label={t("identityAccount")}
                  autoComplete="username"
                  value={account}
                  onChange={(event) => setAccount(event.currentTarget.value)}
                  required
                />
                <PasswordInput
                  label={t("identityPassword")}
                  autoComplete="current-password"
                  value={password}
                  onChange={(event) => setPassword(event.currentTarget.value)}
                  required
                />
                <Button type="submit" loading={busy} disabled={busy}>
                  {t("identityLogin")}
                </Button>
              </Stack>
            </form>
            <Preferences />
          </Stack>
        </Paper>
      </Container>
    </main>
  )
}
