import { Alert, Button, Stack, Text, TextInput, Title } from "@mantine/core"
import { useQuery } from "@tanstack/react-query"
import { useEffect, useRef, useState, useSyncExternalStore, type FormEvent } from "react"
import { useTranslation } from "react-i18next"
import { ApiError } from "../../data/api/client/httpClient"
import type { AccountProfile, PersonalSettingsApi } from "../../data/api/identity/personalSettings"
import type { IdentityRuntime } from "./runtime"

/** 个人设置只按账号读取邮箱和昵称；改密由独立页面拥有。 */
export function PersonalSettingsPage({
  api,
  session,
  embedded = false,
  onCloseBlockedChange
}: {
  api: PersonalSettingsApi
  session: IdentityRuntime
  embedded?: boolean
  onCloseBlockedChange?(blocked: boolean): void
}) {
  const { t } = useTranslation()
  const identity = useSyncExternalStore(session.subscribe, session.getSnapshot, session.getSnapshot)
  const accountId = identity.identity?.id
  const query = useQuery({
    queryKey: ["accountProfile", accountId],
    meta: { private: true },
    retry: false,
    queryFn: ({ signal }) => api.read(signal),
    enabled: !!accountId
  })
  const [verifiedAccount, setVerifiedAccount] = useState<string | undefined>()
  useEffect(() => {
    if (query.isSuccess && query.isFetchedAfterMount && query.data.id === accountId)
      setVerifiedAccount(accountId)
  }, [accountId, query.isSuccess, query.isFetchedAfterMount, query.data])
  if (!accountId || verifiedAccount !== accountId || !query.data || query.data.id !== accountId)
    return (
      <Stack>
        {!embedded && <Title order={1}>{t("personalSettings")}</Title>}
        <Text role="status">{t(query.isError ? "applicationError" : "applicationLoading")}</Text>
        {query.isError && (
          <Button
            onClick={() => {
              void query.refetch()
            }}
          >
            {t("applicationRetry")}
          </Button>
        )}
      </Stack>
    )
  return (
    <PersonalSettingsForm
      key={accountId}
      profile={query.data}
      embedded={embedded}
      onCloseBlockedChange={onCloseBlockedChange}
      api={api}
      session={session}
      refresh={async (signal) => {
        const result = await query.refetch({ throwOnError: true })
        signal.throwIfAborted()
        if (session.getSnapshot().identity?.id !== accountId) throw new Error("profile_superseded")
        if (!result.data || result.data.id !== accountId) throw new Error("profile_unavailable")
        await session.check(signal)
        signal.throwIfAborted()
        if (session.getSnapshot().identity?.id !== accountId) throw new Error("profile_superseded")
        return result.data
      }}
    />
  )
}

/** 资料保存后刷新可信身份；未知写结果只读取核对，不重放修改。 */
function PersonalSettingsForm({
  profile,
  api,
  session,
  refresh,
  embedded,
  onCloseBlockedChange
}: {
  profile: AccountProfile
  api: PersonalSettingsApi
  session: IdentityRuntime
  embedded?: boolean
  onCloseBlockedChange?(blocked: boolean): void
  refresh(signal: AbortSignal): Promise<AccountProfile>
}) {
  const { t } = useTranslation()
  const [nickname, setNickname] = useState(profile.nickname)
  const [busy, setBusy] = useState(false)
  const [status, setStatus] = useState("")
  const [unknown, setUnknown] = useState(false)
  useEffect(() => {
    onCloseBlockedChange?.(busy || unknown)
    return () => onCloseBlockedChange?.(false)
  }, [busy, unknown, onCloseBlockedChange])
  const lock = useRef(false)
  const controller = useRef(new AbortController())
  useEffect(() => {
    const owned = new AbortController()
    controller.current = owned
    return () => owned.abort()
  }, [])
  /** 写入与后续刷新分别确认，刷新失败不能导致重新提交已经成功的修改。 */
  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()
    if (
      session.getSnapshot().identity?.id !== profile.id ||
      lock.current ||
      unknown ||
      !nickname.trim()
    )
      return
    lock.current = true
    setBusy(true)
    setStatus("")
    const signal = controller.current.signal
    const current = () => !signal.aborted && session.getSnapshot().identity?.id === profile.id
    let written = false
    try {
      await api.rename(nickname.trim(), signal)
      written = true
      if (!current()) return
      const saved = await refresh(signal)
      if (current()) {
        setNickname(saved.nickname)
        setStatus("personalSaved")
      }
    } catch (error) {
      if (!current()) return
      const uncertain =
        written ||
        !(error instanceof ApiError) ||
        ["network", "timeout", "invalid-response"].includes(error.kind) ||
        (error.status !== undefined && error.status >= 500)
      if (uncertain) {
        setUnknown(true)
        setStatus("personalSaveUnknown")
      } else setStatus("personalSaveFailed")
    } finally {
      lock.current = false
      if (current()) setBusy(false)
    }
  }
  /** 未知昵称写入只刷新核对，不自动重放。 */
  async function recover() {
    if (lock.current) return
    lock.current = true
    setBusy(true)
    try {
      const actual = await refresh(controller.current.signal)
      if (!controller.current.signal.aborted && session.getSnapshot().identity?.id === profile.id) {
        setNickname(actual.nickname)
        setUnknown(false)
        setStatus("")
      }
    } catch {
      /* 保留未知提示，等待用户再次核对。 */
    } finally {
      lock.current = false
      if (!controller.current.signal.aborted) setBusy(false)
    }
  }
  return (
    <Stack maw={embedded ? undefined : 480}>
      {!embedded && <Title order={1}>{t("personalSettings")}</Title>}
      {status && (
        <Alert role="status">
          {t(status)}
          {unknown && (
            <Button
              variant="subtle"
              loading={busy}
              onClick={() => {
                void recover()
              }}
            >
              {t("personalRefresh")}
            </Button>
          )}
        </Alert>
      )}
      <TextInput label={t("registrationEmail")} value={profile.email} readOnly />
      <form
        onSubmit={(event) => {
          void submit(event)
        }}
      >
        <Stack>
          <TextInput
            label={t("personalNickname")}
            placeholder={t("personalNicknamePlaceholder")}
            value={nickname}
            required
            maxLength={64}
            disabled={busy || !!unknown}
            onChange={(event) => setNickname(event.currentTarget.value)}
          />
          <Button type="submit" loading={busy} disabled={!!unknown || !nickname.trim()}>
            {t("personalSaveNickname")}
          </Button>
        </Stack>
      </form>
    </Stack>
  )
}
