import type { QueryClient } from "@tanstack/react-query"
import { AuthFailure, type AuthApi, type AuthContext } from "../../data/api/auth/contracts"
import type { CredentialStorage } from "../../data/storage/auth"
import { withAuthLock } from "../../data/storage/auth"

/** 只投影客户端会话阶段；用户和权限始终留在 Query。 */
export interface SessionSnapshot {
  phase: "anonymous" | "restoring" | "ready" | "checking" | "error"
  contextKey: readonly string[]
  errorKey: string | null
  revision: number
}
/** 建立唯一会话协调者，区分导航取消与登录/退出/组织变化的代次。 */
export function createAccountSession(
  api: AuthApi,
  storage: CredentialStorage,
  queryClient: QueryClient
) {
  let generation = 0
  let validation = 0
  let transition: number | undefined
  let token: string | null = null
  let blocked: { token: string | null; generation: number } | undefined
  let snapshot: SessionSnapshot = {
    phase: "anonymous",
    contextKey: ["account-auth", "anonymous"],
    errorKey: null,
    revision: 0
  }
  const listeners = new Set<() => void>()
  /** 发布阶段变化，不复制远端上下文。 */
  function publish(
    phase: SessionSnapshot["phase"],
    errorKey: string | null = null,
    contextKey = snapshot.contextKey
  ): void {
    snapshot = { phase, errorKey, contextKey, revision: snapshot.revision + 1 }
    for (const listener of listeners) listener()
  }
  /** 读取 Query 当前权威上下文。 */
  function context(): AuthContext | undefined {
    return queryClient.getQueryData<AuthContext>(snapshot.contextKey)
  }
  /** 同步撤下私有上下文，再取消旧请求并清理私有缓存。 */
  function clearPrivate(): void {
    publish("anonymous", null, ["account-auth", "anonymous"])
    void queryClient.cancelQueries({
      predicate: (query) => query.meta?.private === true || query.queryKey[0] === "account-auth"
    })
    queryClient.removeQueries({
      predicate: (query) => query.meta?.private === true || query.queryKey[0] === "account-auth"
    })
  }
  /** 保留同 scope 外壳，仅在真实身份或组织变化时清除旧上下文。 */
  function accept(value: AuthContext): void {
    const key = ["account-auth", value.user.id, value.organization?.id ?? "no-organization"]
    if (key.join(":") !== snapshot.contextKey.join(":")) clearPrivate()
    queryClient.setQueryData(key, value)
    publish("ready", null, key)
  }
  /** 清理仅当前代次拥有的本地记录；持久删除失败保持恢复阻断。 */
  async function removeOwned(expected: string | null, owned: number): Promise<void> {
    blocked = { token: expected, generation: owned }
    try {
      await withAuthLock(async () => {
        if (owned === generation && storage.read() === expected) storage.remove()
      })
      if (blocked?.generation === owned) blocked = undefined
    } catch {
      if (owned === generation) publish("error", "authStorageError")
      throw new AuthFailure("authStorageError")
    }
  }
  /** 校验每次实际导航，取消旧导航不清理后来会话。 */
  async function check(signal?: AbortSignal): Promise<AuthContext> {
    if (signal?.aborted || transition === generation) throw new AuthFailure("authSuperseded")
    const owned = generation
    const request = ++validation
    const previous = context()
    const confirmedToken = token
    let current: string | null = null
    try {
      current = storage.read()
      if (blocked && blocked.token === current) throw new AuthFailure("authStorageError")
      blocked = undefined
      if (current === null) {
        token = null
        clearPrivate()
        throw new AuthFailure("authAnonymous")
      }
      if (confirmedToken !== current) clearPrivate()
      publish(context() === undefined ? "restoring" : "checking")
      const value = await api.validate(current)
      if (owned !== generation || request !== validation || storage.read() !== current)
        throw new AuthFailure("authSuperseded")
      if (signal?.aborted) throw new AuthFailure("authSuperseded")
      token = current
      accept(value)
      return value
    } catch (error) {
      if (owned !== generation || request !== validation) throw new AuthFailure("authSuperseded")
      const key = error instanceof AuthFailure ? error.key : "authUnexpected"
      if (current !== null) {
        let persisted: string | null
        try {
          persisted = storage.read()
        } catch {
          token = null
          clearPrivate()
          publish("error", "authStorageError")
          throw new AuthFailure("authStorageError")
        }
        if (persisted !== current) {
          token = null
          clearPrivate()
          throw new AuthFailure("authSuperseded")
        }
      }
      if (key === "authSessionExpired") {
        token = null
        generation += 1
        clearPrivate()
        await removeOwned(current, generation)
        throw error
      }
      if (key === "authSuperseded" || (signal?.aborted && key === "authNetwork")) {
        if (previous !== undefined && current === confirmedToken && token === current) publish("ready")
        else { token = null; clearPrivate() }
        throw new AuthFailure("authSuperseded")
      }
      if (key === "authAnonymous") throw error
      if (
        key === "authNetwork" &&
        previous !== undefined &&
        current === confirmedToken &&
        token === current
      ) {
        publish("ready", "authNetwork")
        return previous
      }
      clearPrivate()
      publish("error", key)
      throw error
    }
  }
  /** 登录只公布已验证且持久写入的凭据，失败只回收本次签发。 */
  async function login(
    username: string,
    password: string,
    signal?: AbortSignal
  ): Promise<AuthContext> {
    const owned = ++generation
    transition = owned
    token = null
    clearPrivate()
    publish("restoring")
    let issued: string | undefined
    try {
      const baseline = storage.read()
      issued = await api.login(username, password)
      const value = await api.validate(issued)
      const acceptedToken = issued
      await withAuthLock(async () => {
        if (owned !== generation || signal?.aborted || storage.read() !== baseline)
          throw new AuthFailure("authSuperseded")
        storage.write(acceptedToken)
        token = acceptedToken
        blocked = undefined
        transition = undefined
        accept(value)
      })
      return value
    } catch (error) {
      if (issued !== undefined) {
        if (owned === generation) await removeOwned(issued, owned).catch(() => undefined)
        await api.logout(issued).catch(() => undefined)
      }
      if (owned === generation) {
        token = null
        publish(
          blocked ? "error" : "anonymous",
          error instanceof AuthFailure ? error.key : "authUnexpected"
        )
      }
      throw error
    } finally {
      if (transition === owned) transition = undefined
    }
  }
  /** 退出立即隐藏私有界面，本地清理不依赖远端吊销成功。 */
  async function logout(): Promise<boolean> {
    const owned = ++generation
    let previous = token
    token = null
    clearPrivate()
    if (previous === null) {
      try {
        previous = storage.read()
      } catch {
        publish("error", "authStorageError")
        throw new AuthFailure("authStorageError")
      }
    }
    await removeOwned(previous, owned)
    if (previous !== null) {
      try {
        await api.logout(previous)
      } catch {
        if (owned === generation) publish("anonymous", "authRemoteLogoutUnknown")
      }
    }
    return owned === generation
  }
  /** 服务接受切换后只恢复新上下文，失败不回滚到旧组织继续业务。 */
  async function switchOrganization(organizationId: string): Promise<AuthContext> {
    const previous = token
    if (previous === null) throw new AuthFailure("authAnonymous")
    const owned = ++generation
    transition = owned
    token = null
    clearPrivate()
    publish("restoring")
    let issued: string | undefined
    try {
      issued = await api.switchOrganization(previous, organizationId)
      const next = issued
      const value = await api.validate(next)
      await withAuthLock(async () => {
        if (owned !== generation || storage.read() !== previous)
          throw new AuthFailure("authSuperseded")
        storage.write(next)
        token = next
        blocked = undefined
        transition = undefined
        accept(value)
      })
      return value
    } catch (error) {
      if (issued !== undefined) {
        if (owned === generation) await removeOwned(issued, owned).catch(() => undefined)
        await api.logout(issued).catch(() => undefined)
      }
      if (owned === generation) {
        token = null
        blocked ??= { token: previous, generation: owned }
        clearPrivate()
        publish("error", error instanceof AuthFailure ? error.key : "authUnexpected")
      }
      throw error
    } finally {
      if (transition === owned) transition = undefined
    }
  }
  /** 找回仅清理提交代次中可核实属于目标账号的会话。 */
  async function resetPassword(email: string, password: string, code: string): Promise<boolean> {
    const owned = generation
    const target = context()?.user.email
    await api.resetPassword(email, password, code)
    if (owned !== generation) return false
    if (target === email) {
      const previous = token
      token = null
      generation += 1
      clearPrivate()
      await removeOwned(previous, generation)
    }
    return true
  }
  /** 外部标签页只提供变化信号，重新读取记录并撤下旧会话。 */
  function listenStorage(onChange?: () => void | Promise<unknown>): () => void {
    const listener = (event: StorageEvent) => {
      if (event.key !== storage.key && event.key !== null) return
      generation += 1
      transition = undefined
      token = null
      clearPrivate()
      void check()
        .catch(() => undefined)
        .then(() => onChange?.())
        .catch(() => undefined)
    }
    window.addEventListener("storage", listener)
    return () => window.removeEventListener("storage", listener)
  }
  return {
    api,
    check,
    login,
    logout,
    switchOrganization,
    resetPassword,
    context,
    listenStorage,
    getSnapshot: () => snapshot,
    subscribe: (listener: () => void) => {
      listeners.add(listener)
      return () => {
        listeners.delete(listener)
      }
    }
  }
}
/** 描述上层装配后供 Router 和 UI 共用的会话接口。 */
export type AccountSession = ReturnType<typeof createAccountSession>
