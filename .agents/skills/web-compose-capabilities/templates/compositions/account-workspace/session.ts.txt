import { QueryClient, queryOptions } from "@tanstack/react-query"
import {
  AuthError,
  isAuthContext,
  type AuthApi,
  type AuthContext,
  type MockCredential,
  type ResetInput
} from "../../data/api/auth/contracts"
import type { AuthSessionStorage, StoredAuth } from "../../data/storage/authSessionStorage"
import { isMockCredential } from "../../data/storage/authSessionStorage"
import { ApiError } from "../../data/api/client/httpContracts"
import { APP_CONFIG } from "../../config/environment"

/** 协调状态不包含用户或权限快照，这些数据只在当前 Query scope 中。 */
export interface AuthSnapshot {
  phase:
    | "anonymous"
    | "restoring"
    | "ready"
    | "checking"
    | "offline"
    | "switching"
    | "recovery"
    | "logging-out"
    | "blocked"
  generation: number
  issue?: string
}

/** 认证拥有者的依赖由应用装配，数据层不导入 Router 或 feature。 */
interface SessionDependencies {
  api: AuthApi
  storage: AuthSessionStorage
  queryClient: QueryClient
  clearHttpSession: () => void
}

/** 取消或旧操作只安静结束，绝不能清理后来建立的会话。 */
export function isCancelled(error: unknown): boolean {
  return error instanceof DOMException && error.name === "AbortError"
}

/** 只投影本地化错误码，不向界面传递服务端正文或凭据。 */
export function authIssue(error: unknown): string {
  return error instanceof AuthError ? error.code : "operationFailed"
}

/** 单一协调者统一导航校验、存储、组织原子切换及旧异步隔离。 */
export function createAuthSession(dependencies: SessionDependencies) {
  const { api, storage, queryClient } = dependencies
  let snapshot: AuthSnapshot = { phase: "anonymous", generation: 0 }
  let stored: StoredAuth = { raw: null, credential: null }
  let pendingAdoptionRaw: string | undefined
  let initialized = false
  let blocked = false
  let operation = new AbortController()
  let navigation = new AbortController()
  let initialization: Promise<void> | undefined
  const listeners = new Set<() => void>()

  /** 状态更新必须经过此入口，以便 UI 在作用域切换开始时立即隐藏私有内容。 */
  function publish(phase: AuthSnapshot["phase"], issue?: string): void {
    snapshot = { phase, generation: snapshot.generation, issue }
    listeners.forEach((listener) => {
      try {
        listener()
      } catch {
        /* 观察者不能中断认证事务。 */
      }
    })
  }
  /** Query key 只使用非敏感作用域标识和本页代次，不包含演示句柄或 Token。 */
  function contextKey() {
    return [
      "auth-private",
      "account-workspace",
      APP_CONFIG.environment,
      stored.credential?.accountId ?? "anonymous",
      stored.credential?.organizationId ?? "none",
      snapshot.generation
    ] as const
  }
  /** 已有业务模块以后沿用 auth-private 前缀，不清空公共 Query 或偏好。 */
  function clearPrivate(): void {
    void queryClient.cancelQueries({ queryKey: ["auth-private"] })
    queryClient.removeQueries({ queryKey: ["auth-private"] })
    dependencies.clearHttpSession()
  }
  /** 会话边界推进代次，旧 fetch、锁等待及 mutation 回调同时失效。 */
  function begin(phase: AuthSnapshot["phase"]) {
    navigation.abort()
    operation.abort()
    operation = new AbortController()
    snapshot = { phase, generation: snapshot.generation + 1 }
    clearPrivate()
    listeners.forEach((listener) => {
      try {
        listener()
      } catch {
        /* 观察者不能中断认证事务。 */
      }
    })
    const generation = snapshot.generation
    const signal = operation.signal
    const current = () => snapshot.generation === generation && !signal.aborted
    return { signal, current }
  }
  /** 受保护请求前后都核对真实 Storage，跨页变化先阻断再恢复。 */
  async function ensureCurrent(current: () => boolean): Promise<void> {
    if (!current()) throw new DOMException("Superseded", "AbortError")
    await storage.check(stored.raw, current)
    if (!current()) throw new DOMException("Superseded", "AbortError")
  }
  /** 用户与权限必须通过运行时校验，Query 只在当前操作仍有效时接受结果。 */
  async function loadContext(
    signal: AbortSignal,
    current: () => boolean,
    adoptStoredReference = pendingAdoptionRaw === stored.raw,
    navigationCheck = false
  ) {
    const credential = stored.credential
    if (!credential) throw new AuthError("sessionExpired", "invalid-session")
    await ensureCurrent(current)
    let validated: MockCredential
    try {
      validated = await api.validateSession(credential, signal, adoptStoredReference)
    } catch (error) {
      if (isCancelled(error)) throw error
      if (error instanceof ApiError && error.kind === "aborted")
        throw new DOMException("Cancelled", "AbortError")
      if (
        error instanceof ApiError &&
        (error.kind === "network" || error.kind === "timeout") &&
        error.status === undefined
      )
        throw new AuthError("connectionUnavailable", "network")
      if (
        error instanceof AuthError &&
        (error.kind === "network" || error.kind === "configuration")
      )
        throw error
      if (
        (error instanceof ApiError &&
          (error.kind === "http" ||
            error.kind === "invalid-response" ||
            error.status !== undefined)) ||
        (error instanceof AuthError && (error.kind === "input" || error.kind === "invalid-session"))
      )
        throw new AuthError("sessionExpired", "invalid-session")
      throw new AuthError("operationFailed", "unknown")
    }
    if (
      !isMockCredential(validated) ||
      validated.sessionId !== credential.sessionId ||
      validated.accountId !== credential.accountId
    )
      throw new AuthError("sessionInvalid", "invalid-session")
    await ensureCurrent(current)
    if (adoptStoredReference && pendingAdoptionRaw === stored.raw) pendingAdoptionRaw = undefined
    if (JSON.stringify(validated) !== JSON.stringify(credential)) {
      publish("restoring")
      clearPrivate()
      const next = await storage.commit(stored.raw, validated, current)
      if (!current()) throw new DOMException("Superseded", "AbortError")
      stored = next
      clearPrivate()
      if (navigationCheck) {
        // 新组织是会话边界；恢复独立于原导航取消，不能把旧作用域再公布为ready。
        const scope = begin("restoring")
        try {
          await loadContext(scope.signal, scope.current, false)
        } catch (error) {
          await failure(error, scope.current)
        }
        return
      }
    }
    const activeCredential = stored.credential!
    await queryClient.query(
      queryOptions({
        queryKey: contextKey(),
        staleTime: 0,
        retry: false,
        meta: { errorOwner: "auth" },
        queryFn: async ({ signal: querySignal }) => {
          const data = await api.getContext(
            activeCredential,
            AbortSignal.any([signal, querySignal])
          )
          await ensureCurrent(current)
          if (
            !isAuthContext(data) ||
            data.user.id !== activeCredential.accountId ||
            (data.currentOrganization?.id ?? null) !== activeCredential.organizationId
          )
            throw new AuthError("sessionInvalid", "invalid-session")
          return data
        }
      })
    )
    await ensureCurrent(current)
    publish("ready")
  }
  /** 删除失败保持匿名 UI 并禁止自动恢复；冲突不删除其它标签页的新会话。 */
  async function clearLocal(current: () => boolean): Promise<void> {
    await storage.commit(stored.raw, null, current)
    if (!current()) throw new DOMException("Superseded", "AbortError")
    stored = { raw: null, credential: null }
    blocked = false
  }
  /** 区分传输不确定、明确失效和本地错误，未知权限从不自动放行。 */
  async function failure(error: unknown, current: () => boolean): Promise<void> {
    if (!current() || isCancelled(error)) return
    if (error instanceof AuthError && error.kind === "invalid-session") {
      // 已确认失效是独立会话边界，原导航取消不能中断清理或复活旧凭据。
      const scope = begin("logging-out")
      blocked = true
      try {
        await clearLocal(scope.current)
        if (scope.current()) publish("anonymous", authIssue(error))
      } catch (cause) {
        if (scope.current()) {
          blocked = true
          publish("blocked", authIssue(cause))
        }
      }
      return
    }
    publish(
      error instanceof AuthError && error.kind === "network" ? "offline" : "blocked",
      authIssue(error)
    )
  }

  return {
    api,
    storageKey: storage.key,
    /** 订阅只暴露协调状态，不复制 Query 身份。 */
    getSnapshot: () => snapshot,
    /** React 外部存储订阅随卸载解除。 */
    subscribe(listener: () => void) {
      listeners.add(listener)
      return () => {
        listeners.delete(listener)
      }
    },
    contextKey,
    /** 私有上下文只从当前 Query scope 获取。 */
    getContext: () =>
      snapshot.phase === "ready" || snapshot.phase === "checking"
        ? queryClient.getQueryData<AuthContext>(contextKey())
        : undefined,
    /** 同一启动共享恢复，公开页面也能正确显示已经恢复的会话。 */
    async initialize() {
      if (!initialized) initialization ??= this.restore()
      await initialization
    },
    /** 首次/外部变更恢复；损坏项精确删除，不创建默认身份或权限。 */
    async restore() {
      if (blocked) return
      const { signal, current } = begin("restoring")
      initialized = true
      try {
        const incoming = storage.read()
        if (incoming.raw !== stored.raw) pendingAdoptionRaw = incoming.raw ?? undefined
        stored = incoming
        const adoptStoredReference = pendingAdoptionRaw === stored.raw
        if (!stored.credential) {
          if (stored.raw !== null) await clearLocal(current)
          if (current()) publish("anonymous")
          return
        }
        await loadContext(signal, current, adoptStoredReference)
      } catch (error) {
        await failure(error, current)
      }
    },
    /** 每次实际受保护导航重新校验，不把长 staleTime 当作本次认证证据。 */
    async validateNavigation(signal: AbortSignal) {
      if (!initialized) await this.restore()
      if (
        signal.aborted ||
        blocked ||
        !stored.credential ||
        ["switching", "recovery", "logging-out"].includes(snapshot.phase)
      )
        return
      const sameScope =
        (snapshot.phase === "ready" || snapshot.phase === "checking") && !!this.getContext()
      if (!sameScope) {
        const scope = begin("restoring")
        const combined = AbortSignal.any([signal, scope.signal])
        const current = () => scope.current() && !combined.aborted
        try {
          await loadContext(combined, current)
        } catch (error) {
          await failure(error, current)
        } finally {
          if (scope.current() && combined.aborted && snapshot.phase === "restoring")
            publish("recovery")
        }
        return
      }
      // 导航检查不代表会话变化；只取消上一导航，保留同作用域 Query 和布局节点。
      navigation.abort()
      const activeNavigation = new AbortController()
      navigation = activeNavigation
      const generation = snapshot.generation
      const combined = AbortSignal.any([signal, operation.signal, activeNavigation.signal])
      const current = () =>
        snapshot.generation === generation && navigation === activeNavigation && !combined.aborted
      publish("checking")
      try {
        await queryClient.cancelQueries({ queryKey: contextKey(), exact: true })
        await ensureCurrent(current)
        await loadContext(combined, current, undefined, true)
      } catch (error) {
        await failure(error, current)
      } finally {
        if (
          snapshot.generation === generation &&
          navigation === activeNavigation &&
          combined.aborted
        ) {
          if (snapshot.phase === "checking") publish("ready")
          else if (snapshot.phase === "restoring") publish("recovery")
        }
      }
    },
    /** 登录只有在核实持久写入且上下文完整后才公布成功。 */
    async login(username: string, password: string) {
      const { signal, current } = begin("restoring")
      initialized = true
      let temporary: MockCredential | undefined
      try {
        const expected = storage.read()
        await storage.check(expected.raw, current)
        const credential = await api.login(username, password, signal)
        temporary = credential
        if (!current()) throw new DOMException("Superseded", "AbortError")
        const next = await storage.commit(expected.raw, credential, current)
        if (!current()) throw new DOMException("Superseded", "AbortError")
        stored = next
        pendingAdoptionRaw = undefined
        temporary = undefined
        blocked = false
        await loadContext(signal, current)
      } catch (error) {
        if (temporary) api.discardUnpublishedSession?.(temporary)
        if (current() && !isCancelled(error)) {
          if (!stored.credential) publish("anonymous")
          else await failure(error, current)
        }
        throw error
      }
    },
    /** 先隐藏私有内容并精确删除，再做有界单次远端吊销，不自动重放。 */
    async logout() {
      const credential = stored.credential
      const { signal, current } = begin("logging-out")
      blocked = true
      let issue: string | undefined
      try {
        await clearLocal(current)
      } catch (error) {
        issue = authIssue(error)
      }
      if (!current()) return
      if (issue) publish("blocked", issue)
      else publish("anonymous")
      if (issue === "sessionConflict") return
      if (credential) {
        try {
          await api.logout(credential, AbortSignal.any([signal, AbortSignal.timeout(3000)]))
        } catch {
          if (current() && !issue) publish("anonymous", "revocationUnconfirmed")
        }
      }
    },
    /** 显式刷新由同一代次和原子存储拥有，不改变导航失败即退出策略。 */
    async refresh() {
      const credential = stored.credential
      if (!credential || blocked) throw new AuthError("sessionExpired", "invalid-session")
      const { signal, current } = begin("restoring")
      let temporary: MockCredential | undefined
      try {
        await ensureCurrent(current)
        const nextCredential = await api.refresh(credential, signal)
        temporary = nextCredential
        if (
          !isMockCredential(nextCredential) ||
          nextCredential.accountId !== credential.accountId ||
          nextCredential.organizationId !== credential.organizationId
        )
          throw new AuthError("sessionInvalid", "invalid-session")
        await ensureCurrent(current)
        const next = await storage.commit(stored.raw, nextCredential, current)
        if (!current()) throw new DOMException("Superseded", "AbortError")
        stored = next
        temporary = undefined
        pendingAdoptionRaw = undefined
        await loadContext(signal, current)
      } catch (error) {
        if (temporary) api.discardUnpublishedSession?.(temporary)
        await failure(error, current)
        throw error
      }
    },
    /** 改密成功后只清理本地，不向已经吊销的旧凭据再发 logout。 */
    async changePassword(oldPassword: string, newPassword: string) {
      const credential = stored.credential
      if (snapshot.phase !== "ready" || !credential)
        throw new AuthError("sessionExpired", "invalid-session")
      const { signal, current } = begin("restoring")
      let changed = false
      try {
        await ensureCurrent(current)
        await api.changePassword(credential, oldPassword, newPassword, signal)
        changed = true
        if (!current()) return
        blocked = true
        await clearLocal(current)
        if (current()) publish("anonymous")
      } catch (error) {
        if (changed && current()) {
          blocked = true
          publish("blocked", authIssue(error))
        } else if (error instanceof AuthError && error.kind === "input" && current()) {
          try {
            await loadContext(signal, current)
          } catch (cause) {
            await failure(cause, current)
          }
        } else await failure(error, current)
        throw error
      }
    },
    /** 找回响应绑定提交时的代次与账号，迟到成功不能清除后来的登录。 */
    async resetPassword(input: ResetInput, signal: AbortSignal): Promise<boolean> {
      const generation = snapshot.generation
      const email = this.getContext()?.user.email.trim().toLowerCase()
      await api.resetPassword(input, signal)
      if (snapshot.generation !== generation || signal.aborted) return false
      if (stored.credential && email === input.email.trim().toLowerCase()) {
        const { current } = begin("logging-out")
        blocked = true
        try {
          await clearLocal(current)
          if (current()) publish("anonymous")
        } catch (error) {
          if (current()) publish("blocked", authIssue(error))
          throw error
        }
      }
      return true
    },
    /** 组织提交后不回滚旧权限；响应丢失或上下文失败时留在恢复状态。 */
    async switchOrganization(organizationId: string) {
      const context = this.getContext()
      const credential = stored.credential
      if (
        snapshot.phase !== "ready" ||
        !credential ||
        !context?.capabilities.includes("switch-organization") ||
        !context.organizations.some(
          (entry) => entry.id === organizationId && entry.id !== credential.organizationId
        )
      )
        throw new AuthError("organizationUnavailable")
      const { signal, current } = begin("switching")
      try {
        await ensureCurrent(current)
        const nextCredential = await api.switchOrganization(credential, organizationId, signal)
        await ensureCurrent(current)
        const next = await storage.commit(stored.raw, nextCredential, current)
        if (!current()) return
        stored = next
        await loadContext(signal, current)
      } catch (error) {
        if (!current() || isCancelled(error)) return
        if (error instanceof AuthError && error.kind === "invalid-session")
          await failure(error, current)
        else publish("recovery", authIssue(error))
      }
    },
    /** 其它标签页事件只作通知，重新读取当前记录而不信任 event.newValue。 */
    async synchronize() {
      if (blocked) return
      let changed = true
      try {
        changed = storage.read().raw !== stored.raw
      } catch {
        /* 恢复流程提供可见的存储错误。 */
      }
      if (changed) await this.restore()
    },
    /** 删除失败时由用户主动重试，禁止自动读取残留旧凭据恢复。 */
    async retryCleanup() {
      const { current } = begin("logging-out")
      try {
        await clearLocal(current)
        if (current()) publish("anonymous")
      } catch (error) {
        if (current()) {
          blocked = true
          publish("blocked", authIssue(error))
        }
      }
    }
  }
}

/** 上层消费协调器的稳定合同，不另建认证状态 owner。 */
export type AuthSession = ReturnType<typeof createAuthSession>
