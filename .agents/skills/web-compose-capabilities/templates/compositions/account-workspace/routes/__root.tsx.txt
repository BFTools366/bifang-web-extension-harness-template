import { createRootRouteWithContext, redirect } from "@tanstack/react-router"
import { NotFoundPage } from "../features/not-found/NotFoundPage"
import { ApplicationRoot } from "../features/client/ApplicationRoot"
import type { AuthSession } from "../features/auth/session"

/** 说明 Route 的稳定结构与职责边界。 */
export const Route = createRootRouteWithContext<{ auth: AuthSession }>()({
  beforeLoad: async ({ context, location, preload }) => {
    if (preload || location.pathname.replace(/\/$/u, "").toLowerCase() === "/settings") return
    await context.auth.initialize()
    if (location.pathname === "/")
      throw redirect({
        to: context.auth.getContext() ? "/app/dashboard" : "/login",
        replace: true
      })
  },
  component: ApplicationRoot,
  notFoundComponent: NotFoundPage
})
