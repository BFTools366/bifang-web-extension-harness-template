import { createFileRoute } from "@tanstack/react-router"
import { protectRoute } from "../../features/auth/guards"
import { AuthStatePanel } from "../../features/auth/AuthStatePanel"

/** 无权限恢复页只要求登录，不依赖被拒绝的业务能力，避免重定向循环。 */
export const Route = createFileRoute("/app/forbidden")({
  beforeLoad: ({ context, abortController, preload }) =>
    protectRoute(context.auth, "/app/forbidden", abortController.signal, preload),
  component: () => <AuthStatePanel denied />
})
