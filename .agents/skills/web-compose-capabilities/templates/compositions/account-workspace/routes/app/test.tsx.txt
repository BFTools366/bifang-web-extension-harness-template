import { createFileRoute } from "@tanstack/react-router"
import { protectPage } from "../../features/auth/guards"
import { TestMenuPage } from "../../features/client/TestMenuPage"

/** 测试菜单的独立深链接先校验当前组织权限，再挂载正文。 */
export const Route = createFileRoute("/app/test")({
  beforeLoad: ({ context, abortController, preload }) =>
    protectPage(context.auth, "/app/test", abortController.signal, preload),
  component: TestMenuPage
})
