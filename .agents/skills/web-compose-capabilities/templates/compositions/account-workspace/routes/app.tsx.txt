import { createFileRoute, redirect } from "@tanstack/react-router"
import { ClientLayout } from "../features/client/ClientLayout"
import { safeReturnTo } from "../features/auth/access"

/** 私有布局只拥有固定 shell，异步校验落在叶路由，加载状态限于正文插槽。 */
export const Route = createFileRoute("/app")({
  beforeLoad: ({ context, location, preload }) => {
    if (preload) return
    if (context.auth.getSnapshot().phase === "anonymous")
      throw redirect({
        to: "/login",
        search: { returnTo: safeReturnTo(location.pathname) },
        replace: true
      })
    if (location.pathname.replace(/\/$/, "") === "/app")
      throw redirect({ to: "/app/dashboard", replace: true })
  },
  component: ClientLayout
})
