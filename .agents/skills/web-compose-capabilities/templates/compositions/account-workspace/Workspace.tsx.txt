import {
  Alert,
  Anchor,
  AppShell,
  Avatar,
  Burger,
  Button,
  Container,
  Group,
  Image,
  Menu,
  Modal,
  Paper,
  Select,
  Stack,
  Text,
  Title
} from "@mantine/core"
import { Link, Outlet, useNavigate } from "@tanstack/react-router"
import { useEffect, useRef, useState } from "react"
import { useTranslation } from "react-i18next"
import { BRAND } from "../../brand"
import { BUILD_VERSION } from "../../buildVersion"
import { AuthFailure } from "../../data/api/auth/contracts"
import { useAuthContext } from "../auth/SessionProvider"

/** 持久应用外壳；普通菜单导航只替换 Outlet，不重建账号区域。 */
export function WorkspaceLayout() {
  const { t } = useTranslation()
  const { session, state, context } = useAuthContext()
  const navigate = useNavigate()
  const [opened, setOpened] = useState(false)
  const [switching, setSwitching] = useState(false)
  const [target, setTarget] = useState<string | null>(null)
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const trigger = useRef<HTMLButtonElement>(null)
  const mounted = useRef(true)
  useEffect(() => {
    mounted.current = true
    return () => {
      mounted.current = false
    }
  }, [])
  const available = [
    ...new Map(
      (context?.organizations ?? []).map((organization) => [organization.id, organization])
    ).values()
  ]
  useEffect(() => {
    if (state.phase === "anonymous" || state.phase === "error") setSwitching(false)
  }, [state.phase])
  /** 账号退出保持偏好，Storage 失败留在明确阻断状态。 */
  async function logout(): Promise<void> {
    try {
      const current = await session.logout()
      if (current && mounted.current) await navigate({ to: "/login" })
    } catch {
      if (mounted.current) setError("authStorageError")
    }
  }
  /** 切换完成后重新检查当前目标；无权限进入独立恢复路由。 */
  async function switchOrganization(): Promise<void> {
    if (!target || busy) return
    setBusy(true)
    setError(null)
    try {
      const next = await session.switchOrganization(target)
      if (!mounted.current) return
      setSwitching(false)
      await navigate({
        to: next.permissions.includes("workspace:view") ? "/app/dashboard" : "/no-access"
      })
    } catch (failure) {
      if (mounted.current) setError(failure instanceof AuthFailure ? failure.key : "authUnexpected")
    } finally {
      if (mounted.current) setBusy(false)
    }
  }
  if (!context || (state.phase !== "ready" && state.phase !== "checking"))
    return <SessionRecovery />
  return (
    <AppShell
      header={{ height: 64 }}
      navbar={{ width: 260, breakpoint: "sm", collapsed: { mobile: !opened } }}
      padding="lg"
    >
      <AppShell.Header>
        <Group h="100%" px="md" justify="space-between">
          <Group>
            <Burger
              opened={opened}
              onClick={() => setOpened((value) => !value)}
              hiddenFrom="sm"
              aria-label={t("authNavigation")}
            />
            <Image src={BRAND.logoPath} alt="" w={32} h={32} />
            <Text fw={700}>{t("projectName")}</Text>
          </Group>
          <Menu>
            <Menu.Target>
              <Button
                ref={trigger}
                variant="subtle"
                leftSection={
                  <Avatar size="sm" aria-hidden="true">
                    {context.user.username.slice(0, 1).toUpperCase()}
                  </Avatar>
                }
              >
                {context.user.username}
              </Button>
            </Menu.Target>
            <Menu.Dropdown>
              <Menu.Label>{context.user.email}</Menu.Label>
              {available.length > 1 && (
                <Menu.Item
                  onClick={() => {
                    setTarget(null)
                    setSwitching(true)
                  }}
                >
                  {t("authSwitchOrganization")}
                </Menu.Item>
              )}
              <Menu.Item onClick={() => void logout()}>{t("authLogout")}</Menu.Item>
            </Menu.Dropdown>
          </Menu>
        </Group>
      </AppShell.Header>
      <AppShell.Navbar p="md">
        <Stack h="100%" justify="space-between">
          <Stack>
            <Anchor href="#workspace-content">{t("authSkipContent")}</Anchor>
            {context.permissions.includes("workspace:view") && (
              <Anchor
                component={Link}
                to="/app/dashboard"
                activeOptions={{ exact: true }}
                onClick={() => setOpened(false)}
              >
                {t("authDashboard")}
              </Anchor>
            )}
            {context.permissions.includes("account:view") && (
              <Anchor
                component={Link}
                to="/app/account"
                activeOptions={{ exact: true }}
                onClick={() => setOpened(false)}
              >
                {t("authAccount")}
              </Anchor>
            )}
          </Stack>
          <Stack gap="xs">
            <Text size="sm">{context.user.username}</Text>
            <Text size="sm" c="dimmed">
              {context.organization?.name ?? t("authNoOrganization")}
            </Text>
            <Text size="xs" role="status">
              {state.errorKey ? t(state.errorKey) : t("authSignedIn")}
            </Text>
            <Text size="xs">{t("buildVersion", { version: BUILD_VERSION })}</Text>
          </Stack>
        </Stack>
      </AppShell.Navbar>
      <AppShell.Main id="workspace-content">
        {error && (
          <Alert role="alert" color="red">
            {t(error)}
          </Alert>
        )}
        <Outlet />
      </AppShell.Main>
      <Modal
        opened={switching}
        onClose={() => {
          if (!busy) setSwitching(false)
        }}
        title={t("authSwitchOrganization")}
        onExitTransitionEnd={() => trigger.current?.focus()}
      >
        <Stack>
          <Select
            label={t("authOrganization")}
            data={available
              .filter((organization) => organization.id !== context.organization?.id)
              .map((organization) => ({ value: organization.id, label: organization.name }))}
            value={target}
            onChange={setTarget}
            disabled={busy}
          />
          <Button onClick={() => void switchOrganization()} loading={busy} disabled={!target}>
            {t("authSwitchOrganization")}
          </Button>
        </Stack>
      </Modal>
    </AppShell>
  )
}
/** 工作台仅提供真实空态，产品功能在下游继续定义。 */
export function DashboardPage() {
  const { t } = useTranslation()
  const { context } = useAuthContext()
  const heading = useRef<HTMLHeadingElement>(null)
  useEffect(() => heading.current?.focus({ preventScroll: true }), [])
  return (
    <section>
      <Stack>
        <Title ref={heading} tabIndex={-1} order={1}>
          {t("authDashboard")}
        </Title>
        <Paper p="xl" withBorder>
          <Stack>
            <Title order={2}>{t("authWorkspaceReady")}</Title>
            <Text c="dimmed">{t("authWorkspaceEmpty")}</Text>
            {context?.permissions.includes("account:view") && (
              <Anchor component={Link} to="/app/account" activeOptions={{ exact: true }}>
                {t("authAccount")}
              </Anchor>
            )}
          </Stack>
        </Paper>
      </Stack>
    </section>
  )
}
/** 当前账户只呈现 API 已校验的信息，不提供假保存操作。 */
export function AccountPage() {
  const { t } = useTranslation()
  const { context } = useAuthContext()
  return (
    <section>
      <Stack>
        <Title order={1}>{t("authAccount")}</Title>
        <Text>{context?.user.username}</Text>
        <Text>{context?.user.email}</Text>
        <Text>{context?.organization?.name ?? t("authNoOrganization")}</Text>
      </Stack>
    </section>
  )
}
/** 会话未知或无权时隐藏私有正文，提供有边界的恢复和退出。 */
export function SessionRecovery({ denied = false }: { denied?: boolean }) {
  const { t } = useTranslation()
  const { session, state, context } = useAuthContext()
  const navigate = useNavigate()
  const [busy, setBusy] = useState(false)
  const mounted = useRef(true)
  useEffect(() => {
    mounted.current = true
    return () => {
      mounted.current = false
    }
  }, [])
  /** 显式读取恢复当前上下文，不自动重放认证写操作。 */
  async function retry(): Promise<void> {
    setBusy(true)
    try {
      await session.check()
      if (mounted.current) await navigate({ to: "/app/dashboard" })
    } catch {
      /* 状态与错误由唯一会话协调者发布。 */
    } finally {
      if (mounted.current) setBusy(false)
    }
  }
  return (
    <main className="page-shell">
      <Container size="sm">
        <Stack>
          <Title>{t(denied ? "authNoAccess" : "authRestoring")}</Title>
          <Text role={state.errorKey ? "alert" : "status"}>
            {t(
              state.errorKey ??
                (context?.organization === null
                  ? "authNoOrganization"
                  : denied
                    ? "authAccessDenied"
                    : "routeLoading")
            )}
          </Text>
          {!denied && (
            <Button loading={busy} onClick={() => void retry()}>
              {t("routeRetry")}
            </Button>
          )}
          {context?.permissions.includes("workspace:view") && (
            <Anchor component={Link} to="/app/dashboard">
              {t("authDashboard")}
            </Anchor>
          )}
          {context?.permissions.includes("account:view") && (
            <Anchor component={Link} to="/app/account">
              {t("authAccount")}
            </Anchor>
          )}
          <Button
            variant="default"
            onClick={() => {
              void session
                .logout()
                .then((current) => {
                  if (current && mounted.current) return navigate({ to: "/login" })
                })
                .catch(() => undefined)
            }}
          >
            {t("authLogout")}
          </Button>
        </Stack>
      </Container>
    </main>
  )
}
