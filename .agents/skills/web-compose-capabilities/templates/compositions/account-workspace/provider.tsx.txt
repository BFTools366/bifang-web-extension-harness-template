import { createContext, useContext, useEffect, useSyncExternalStore, type ReactNode } from "react"
import { useQuery } from "@tanstack/react-query"
import type { AuthContext } from "../../data/api/auth/contracts"
import type { AccountSession } from "./session"

const SessionContext = createContext<AccountSession | null>(null)
/** 由 app 装配唯一会话并负责浏览器订阅的对称生命周期。 */
export function AccountSessionProvider({
  session,
  onStorageChange,
  children
}: {
  session: AccountSession
  onStorageChange?: () => void | Promise<unknown>
  children: ReactNode
}) {
  useEffect(() => session.listenStorage(onStorageChange), [session, onStorageChange])
  return <SessionContext.Provider value={session}>{children}</SessionContext.Provider>
}
/** 获取已注入的会话协调者，不由 feature 反向导入 app 实例。 */
export function useAccountSession(): AccountSession {
  const session = useContext(SessionContext)
  if (session === null) throw new Error("Missing account session provider")
  return session
}
/** 订阅会话阶段与 Query 上下文，不建立用户或权限的竞争副本。 */
export function useAuthContext() {
  const session = useAccountSession()
  const state = useSyncExternalStore(session.subscribe, session.getSnapshot, session.getSnapshot)
  const query = useQuery<AuthContext>({
    queryKey: state.contextKey,
    queryFn: () => session.check(),
    enabled: false,
    staleTime: Infinity,
    meta: { private: true }
  })
  return { session, state, context: query.data }
}
