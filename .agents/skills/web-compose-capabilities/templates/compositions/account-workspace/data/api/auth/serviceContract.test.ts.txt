import { fixtureValue } from "./mockTestAssertions"
import { expect, test, vi } from "vitest"
import { createMockAuthApi } from "./mockAuth"
import {
  authServiceHeaders,
  isServiceAuthResult,
  isServiceProfile,
  isServiceVerification
} from "./serviceContract"
import type { MockServiceExchange } from "./mockAuthService"
/** 实际应用使用的 Mock 适配层处理原始登录 Token JSON，且不发起网络。 */
test("maps login, public isolation, bearer profile and empty logout without fetch", async () => {
  const fetch = vi.spyOn(globalThis, "fetch")
  const exchanges: MockServiceExchange[] = []
  const api = createMockAuthApi({
    delayMs: 0,
    configured: () => false,
    getLanguage: () => "zh-CN",
    observeExchange: (exchange) => exchanges.push(exchange)
  })
  const signal = new AbortController().signal
  const credential = await api.login("test@example.invalid", "test-password", signal)
  const login = fixtureValue(exchanges[0])
  expect(login.request).toMatchObject({
    method: "POST",
    path: "/api/v1/auth/login",
    body: { account: "test@example.invalid", password: "test-password" },
    headers: { "X-Lang": "zh", "Content-Type": "application/json" }
  })
  expect(login.request.headers.Authorization).toBeUndefined()
  expect(login.response).toMatchObject({
    model: "auth.token",
    request_id: login.request.headers["X-Request-Id"],
    data: {
      token: {
        token_type: "Bearer",
        expires_in: 7200,
        access_token: expect.stringMatching(/^mock-access-/),
        refresh_token: expect.stringMatching(/^mock-refresh-/)
      },
      account: { id: credential.accountId }
    }
  })
  await api.validateSession(credential, signal, true)
  const profile = fixtureValue(exchanges.at(-1))
  expect(profile.request).toMatchObject({
    method: "GET",
    path: "/api/v1/auth/profile",
    headers: { Authorization: expect.stringMatching(/^Bearer mock-access-/) }
  })
  expect(profile.request.body).toBeUndefined()
  expect(profile.request.headers["X-AUTH-TOKEN"]).toBeUndefined()
  expect(profile.request.headers["X-Http-Request"]).toBeUndefined()
  expect(profile.response).toMatchObject({
    model: "account",
    data: { org: { is_default: true } }
  })
  expect(
    (
      profile.response as {
        data: unknown
      }
    ).data
  ).not.toHaveProperty("capabilities")
  expect(profile.request.headers["X-Request-Id"]).not.toBe(login.request.headers["X-Request-Id"])
  await api.logout(credential, signal)
  expect(exchanges.at(-1)).toMatchObject({ status: 200, response: { model: "empty" } })
  expect(fixtureValue(exchanges.at(-1)).request.body).toBeUndefined()
  expect(fetch).not.toHaveBeenCalled()
  fetch.mockRestore()
})
/** 业务错误从统一 errors 封装解析稳定码及追踪标识，不匹配翻译消息。 */
test("preserves service error codes, request ids and server cooldown mapping", async () => {
  const exchanges: MockServiceExchange[] = []
  const api = createMockAuthApi({
    delayMs: 0,
    configured: () => false,
    observeExchange: (exchange) => exchanges.push(exchange)
  })
  const signal = new AbortController().signal
  await expect(api.login("test", "wrong", signal)).rejects.toMatchObject({
    code: "credentialsInvalid",
    serviceCode: "auth.password.wrong",
    status: 401,
    requestId: expect.any(String)
  })
  expect(fixtureValue(exchanges[0]).response).toMatchObject({
    model: "errors",
    errors: {
      model: "data.set",
      total: 1,
      datas: [{ model: "error", code: "auth.password.wrong" }]
    }
  })
  expect(await api.sendCode("test@example.invalid", "reset", signal)).toEqual({
    retryAfterSeconds: 60,
    mockCode: "246810"
  })
  expect(fixtureValue(exchanges.at(-1)).request.body).toEqual({
    target: "test@example.invalid",
    target_type: "email",
    scene: "reset_password"
  })
  await expect(api.sendCode("missing@example.invalid", "reset", signal)).rejects.toMatchObject({
    serviceCode: "verification.target.not.registered",
    status: 404
  })
})
/** 全账号登出吊销本实例所有会话，清理临时本地引用则只影响该引用。 */
test("distinguishes account-wide logout from unpublished mock cleanup", async () => {
  const api = createMockAuthApi({ delayMs: 0, configured: () => false })
  const signal = new AbortController().signal
  const first = await api.login("demo", "demo-password", signal)
  await expect(api.logout({ ...first, accountId: "test-user" }, signal)).rejects.toMatchObject({
    serviceCode: "auth.token.revoked"
  })
  const temporary = await api.login("demo", "demo-password", signal)
  fixtureValue(api.discardUnpublishedSession)(temporary)
  await expect(api.validateSession(first, signal)).resolves.toMatchObject({
    accountId: "demo-user"
  })
  const second = await api.login("demo", "demo-password", signal)
  await api.logout(first, signal)
  await expect(api.validateSession(second, signal)).rejects.toMatchObject({
    serviceCode: "auth.token.revoked"
  })
})
/** 匿名操作忽略传入 Token，非法 Bearer 与错误模型不得进入领域层。 */
test("rejects invalid tokens and model shapes without inferring permissions", () => {
  expect(authServiceHeaders("login", "en-US", "request-example", "old-token")).not.toHaveProperty(
    "Authorization"
  )
  expect(() => authServiceHeaders("profile", "en", "request-example", "bad\ntoken")).toThrow()
  expect(
    isServiceAuthResult({
      token: {
        access_token: "a",
        refresh_token: "b",
        token_type: "Basic",
        expires_in: 7200
      }
    })
  ).toBe(false)
  expect(isServiceProfile({ account: { id: "2108046372081659904" }, org: null })).toBe(false)
})
/** 非 Mock 不能携带验证码明文，且目标类型和用途必须受约束。 */
test("validates verification mode and scene rather than trusting duration fields", () => {
  const sent = {
    target: "d***@example.invalid",
    target_type: "email",
    scene: "register",
    expires_in: 300,
    resend_after: 60,
    mock: true,
    code: "246810"
  }
  expect(isServiceVerification(sent)).toBe(true)
  expect(isServiceVerification({ ...sent, mock: false })).toBe(false)
  const delivered: Omit<typeof sent, "code"> & {
    code?: string
  } = { ...sent }
  delete delivered.code
  expect(isServiceVerification({ ...delivered, mock: false })).toBe(true)
  expect(isServiceVerification({ ...sent, scene: "unknown" })).toBe(false)
  expect(isServiceVerification({ ...sent, target_type: "unknown" })).toBe(false)
  expect(isServiceVerification({ ...sent, code: "secret-text" })).toBe(false)
})
/** 注册不返回令牌，找回采用服务字段并使已签发的旧引用失效。 */
test("maps registration and reset wire bodies while preserving session effects", async () => {
  const exchanges: MockServiceExchange[] = []
  const api = createMockAuthApi({
    delayMs: 0,
    configured: () => false,
    observeExchange: (exchange) => exchanges.push(exchange)
  })
  const signal = new AbortController().signal
  await api.sendCode("new@example.invalid", "register", signal)
  const registered = await api.register(
    {
      username: "new_user",
      email: "new@example.invalid",
      password: "Example123",
      code: "246810"
    },
    signal
  )
  const registration = fixtureValue(exchanges.at(-1))
  expect(registration.request).toMatchObject({
    method: "POST",
    path: "/api/v1/auth/register",
    body: {
      username: "new_user",
      email: "new@example.invalid",
      password: "Example123",
      code: "246810",
      code_type: "email"
    }
  })
  expect(registration.response).toMatchObject({
    model: "account",
    data: { id: registered.id, last_login_at: null }
  })
  expect(registration.response).not.toHaveProperty("data.token")
  const credential = await api.login("new_user", "Example123", signal)
  const sent = await api.sendCode("new@example.invalid", "reset", signal)
  await api.resetPassword(
    { email: "new@example.invalid", password: "Changed123", code: fixtureValue(sent.mockCode) },
    signal
  )
  const reset = fixtureValue(exchanges.at(-1))
  expect(reset.request).toMatchObject({
    method: "POST",
    path: "/api/v1/auth/password/reset",
    body: {
      target: "new@example.invalid",
      target_type: "email",
      code: sent.mockCode,
      new_password: "Changed123"
    }
  })
  expect(reset.response).toMatchObject({ model: "empty" })
  expect(reset.request.headers.Authorization).toBeUndefined()
  await expect(api.validateSession(credential, signal)).rejects.toMatchObject({
    serviceCode: "auth.token.revoked"
  })
})
