import { BRAND } from "../../../brand"
import { APP_CONFIG } from "../../../config/environment"
import { requireAuthStorage, withAuthLock } from "../../storage/auth"
import { AuthFailure, type AuthApi, type AuthContext, type Membership } from "./contracts"

/** 固定演示账户；密码只存在内存，不写 Storage、日志或 URL。 */
interface DemoAccount {
  id: string
  username: string
  email: string
  password: string
  memberships: Membership[]
}
/** Mock 服务仅持久保存非生产会话引用，账户和密码在刷新后回到演示 fixture。 */
interface DemoSession {
  accountId: string
  organizationId: string | null
}
/** 明确独立于客户端权限状态的演示服务会话登记。 */
interface DemoServiceState {
  version: 1
  sessions: Record<string, DemoSession>
}
/** 建立无 Fetch 的类型化认证 Mock，服务上下文不能从客户端凭据记录推导。 */
export function createMockAuthApi(): AuthApi {
  const organizations: Membership[] = [
    { id: "demo-team", name: "Demo Team", permissions: ["workspace:view", "account:view"] },
    { id: "restricted-team", name: "Restricted Team", permissions: [] }
  ]
  const accounts: DemoAccount[] = [
    {
      id: "demo",
      username: "demo",
      email: "demo@example.test",
      password: "Demo-only-24",
      memberships: organizations
    },
    {
      id: "viewer",
      username: "viewer",
      email: "viewer@example.test",
      password: "Demo-only-24",
      memberships: organizations
        .slice(0, 1)
        .map((organization) => ({ ...organization, permissions: ["account:view"] }))
    }
  ]
  const codes = new Map<string, { expiresAt: number; lastSentAt: number }>()
  const key = `${BRAND.id}:mock-auth-service:${APP_CONFIG.environment}:v1`
  /** 每次读取演示服务登记，确保跨标签页退出和切换后不复活旧会话。 */
  function readState(): DemoServiceState {
    let raw: string | null
    try {
      raw = requireAuthStorage().getItem(key)
    } catch {
      throw new AuthFailure("authStorageError")
    }
    if (raw === null) return { version: 1, sessions: {} }
    try {
      const value: unknown = JSON.parse(raw)
      if (
        typeof value !== "object" ||
        value === null ||
        !("version" in value) ||
        value.version !== 1 ||
        !("sessions" in value) ||
        typeof value.sessions !== "object" ||
        value.sessions === null ||
        Array.isArray(value.sessions)
      )
        throw new Error("invalid")
      const sessions: Record<string, DemoSession> = {}
      for (const [token, record] of Object.entries(value.sessions)) {
        if (
          !/^mock-[a-f0-9-]{36}$/.test(token) ||
          typeof record !== "object" ||
          record === null ||
          !("accountId" in record) ||
          typeof record.accountId !== "string" ||
          !("organizationId" in record) ||
          (record.organizationId !== null && typeof record.organizationId !== "string")
        )
          throw new Error("invalid")
        sessions[token] = { accountId: record.accountId, organizationId: record.organizationId }
      }
      return { version: 1, sessions }
    } catch {
      throw new AuthFailure("authDemoStateError")
    }
  }
  /** 核实演示服务登记写入，不能把失败的写操作当成成功。 */
  function writeState(state: DemoServiceState): void {
    try {
      const storage = requireAuthStorage()
      const raw = JSON.stringify(state)
      storage.setItem(key, raw)
      if (storage.getItem(key) !== raw) throw new AuthFailure("authStorageError")
    } catch {
      throw new AuthFailure("authStorageError")
    }
  }
  /** 解析当前会话并从演示服务账户/成员关系取得权限。 */
  function resolve(token: string): { account: DemoAccount; session: DemoSession } {
    const session = readState().sessions[token]
    const account = accounts.find((value) => value.id === session?.accountId)
    if (session === undefined || account === undefined) throw new AuthFailure("authSessionExpired")
    return { account, session }
  }
  /** 邮箱规则仅为演示 fixture，不作为未确认真实服务的密码策略。 */
  function assertEmail(email: string): void {
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) throw new AuthFailure("authInvalidEmail")
  }
  /** 验证用途、邮箱、期限与一次性消费；数值仅属于演示合同。 */
  function consumeCode(email: string, purpose: "register" | "reset", code: string): void {
    const id = `${purpose}:${email}`
    const sent = codes.get(id)
    if (sent === undefined || code !== "246810" || sent.expiresAt < Date.now())
      throw new AuthFailure("authInvalidCode")
    codes.delete(id)
  }
  return {
    /* 校验账号凭据并签发演示会话。 */ async login(username, password) {
      return withAuthLock(async () => {
        const account = accounts.find(
          (value) => value.username === username && value.password === password
        )
        if (account === undefined) throw new AuthFailure("authInvalidCredentials")
        const state = readState()
        const token = `mock-${crypto.randomUUID()}`
        state.sessions[token] = {
          accountId: account.id,
          organizationId: account.memberships[0]?.id ?? null
        }
        writeState(state)
        return token
      })
    },
    /* 向演示服务校验当前会话并取得权限上下文。 */ async validate(token) {
      const { account, session } = resolve(token)
      const organization =
        account.memberships.find((value) => value.id === session.organizationId) ?? null
      if (session.organizationId !== null && organization === null)
        throw new AuthFailure("authSessionExpired")
      return structuredClone({
        user: { id: account.id, username: account.username, email: account.email },
        organization,
        organizations: account.memberships,
        permissions: organization?.permissions ?? []
      }) satisfies AuthContext
    },
    /* 吊销指定演示会话，不扩大到其他签发。 */ async logout(token) {
      await withAuthLock(async () => {
        const state = readState()
        delete state.sessions[token]
        writeState(state)
      })
    },
    /* 校验成员资格后切换会话作用域。 */ async switchOrganization(token, organizationId) {
      return withAuthLock(async () => {
        const { account } = resolve(token)
        if (!account.memberships.some((value) => value.id === organizationId))
          throw new AuthFailure("authNoAccess")
        const state = readState()
        const nextToken = `mock-${crypto.randomUUID()}`
        delete state.sessions[token]
        state.sessions[nextToken] = { accountId: account.id, organizationId }
        writeState(state)
        return nextToken
      })
    },
    /* 按邮箱和用途建立有界演示验证码状态。 */ async sendCode(email, purpose) {
      assertEmail(email)
      const id = `${purpose}:${email}`
      const previous = codes.get(id)
      if (previous !== undefined && Date.now() - previous.lastSentAt < 30_000)
        throw new AuthFailure("authCodeRateLimited")
      codes.set(id, { expiresAt: Date.now() + 300_000, lastSentAt: Date.now() })
    },
    /* 消费注册验证码并建立当前演示账号。 */ async register(username, email, password, code) {
      assertEmail(email)
      if (!username.trim() || !password) throw new AuthFailure("authRequired")
      if (accounts.some((value) => value.username === username || value.email === email))
        throw new AuthFailure("authAccountExists")
      consumeCode(email, "register", code)
      accounts.push({ id: crypto.randomUUID(), username, email, password, memberships: [] })
    },
    /* 消费找回验证码并吊销目标账号的演示会话。 */ async resetPassword(email, password, code) {
      if (!password) throw new AuthFailure("authRequired")
      consumeCode(email, "reset", code)
      const account = accounts.find((value) => value.email === email)
      // 找回反馈保持中性，不暴露目标账号是否存在。
      if (account === undefined) return
      await withAuthLock(async () => {
        const state = readState()
        state.sessions = Object.fromEntries(
          Object.entries(state.sessions).filter(([, value]) => value.accountId !== account.id)
        )
        writeState(state)
        account.password = password
      })
    }
  }
}
/** 非空 BaseURL 不能静默启用 Mock；真实 API 适配须先落实可信契约。 */
export function createAuthApi(baseUrl: string = APP_CONFIG.apiBaseUrl): AuthApi {
  if (!baseUrl) return createMockAuthApi()
  const blocked = async (): Promise<never> => {
    throw new AuthFailure("authServiceNotConnected")
  }
  return {
    login: blocked,
    validate: blocked,
    logout: blocked,
    switchOrganization: blocked,
    sendCode: blocked,
    register: blocked,
    resetPassword: blocked
  }
}
