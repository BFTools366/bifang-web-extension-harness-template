/** 在单元环境提供确定的锁替身；不冒称真实跨标签页浏览器验证。 */
beforeEach(() => {
  localStorage.clear()
  vi.restoreAllMocks()
  Object.defineProperty(navigator, "locks", {
    configurable: true,
    value: { request: async (_name: string, operation: () => Promise<unknown>) => operation() }
  })
})
/** 为各场景独立创建 API、会话和 Query 所有者。 */
function fixture() {
  const api = createMockAuthApi()
  const storage = createCredentialStorage()
  const query = new QueryClient({ defaultOptions: { queries: { retry: false } } })
  const session = createAccountSession(api, storage, query)
  return { api, storage, query, session }
}
/** 登录必须真实校验演示身份；持久记录不包含用户或权限。 */
test("login validates credentials and restores through the API after refresh", async () => {
  const { session, storage } = fixture()
  await expect(session.login("demo", "wrong")).rejects.toMatchObject({
    key: "authInvalidCredentials"
  })
  expect(storage.read()).toBeNull()
  const context = await session.login("demo", "Demo-only-24")
  expect(context.user.username).toBe("demo")
  expect(context.permissions).toContain("workspace:view")
  const record = JSON.parse(localStorage.getItem(storage.key) ?? "null")
  expect(Object.keys(record).sort()).toEqual(["token", "version"])
  const refreshed = fixture()
  expect((await refreshed.session.check()).user.id).toBe("demo")
  await session.logout()
  await expect(refreshed.session.check()).rejects.toMatchObject({ key: "authAnonymous" })
  expect(session.context()).toBeUndefined()
})
/** 存储失败不能得到持久登录成功，也不公布私有上下文。 */
test("blocked persistence never grants a private UI session", async () => {
  const { session, storage } = fixture()

    throw new AuthFailure("authStorageError")
  })
  await expect(session.login("demo", "Demo-only-24")).rejects.toMatchObject({
    key: "authStorageError"
  })
  expect(session.getSnapshot().phase).toBe("anonymous")
  expect(session.context()).toBeUndefined()
})
/** 空权限组织与普通身份不继承工作台授权，未知组织明确拒绝。 */
test("organization changes use new permissions and reject unknown memberships", async () => {
  const { session, api, storage } = fixture()
  await session.login("demo", "Demo-only-24")
  const previous = storage.read()
  const context = await session.switchOrganization("restricted-team")
  expect(context.permissions).toEqual([])
  expect(context.organization?.id).toBe("restricted-team")
  expect(session.context()?.permissions).toEqual([])
  await expect(api.validate(previous ?? "")).rejects.toMatchObject({ key: "authSessionExpired" })
  await expect(api.switchOrganization(storage.read() ?? "", "unknown")).rejects.toMatchObject({
    key: "authNoAccess"
  })
  await session.logout()
  const viewer = await session.login("viewer", "Demo-only-24")
  expect(viewer.permissions).toEqual(["account:view"])
})
/** 注册验证码绑定邮箱与用途，一次性消费；新账号不自动拥有组织。 */
test("registration consumes a scoped code and does not sign in", async () => {
  const { api, session, storage } = fixture()
  await api.sendCode("new@example.test", "register")
  await expect(
    api.register("new", "other@example.test", "new-password", "246810")
  ).rejects.toMatchObject({ key: "authInvalidCode" })
  await api.register("new", "new@example.test", "new-password", "246810")
  expect(storage.read()).toBeNull()
  await expect(
    api.register("another", "new@example.test", "new-password", "246810")
  ).rejects.toThrow()
  const context = await session.login("new", "new-password")
  expect(context.organization).toBeNull()
  expect(context.permissions).toEqual([])
})
/** 找回用途不能消费注册验证码，改密后当前账号会话失效。 */
test("password reset revokes the target account without automatically signing in", async () => {
  const { api, session, storage } = fixture()
  await session.login("demo", "Demo-only-24")
  await api.sendCode("demo@example.test", "register")
  await expect(
    session.resetPassword("demo@example.test", "changed", "246810")
  ).rejects.toMatchObject({ key: "authInvalidCode" })
  await api.sendCode("demo@example.test", "reset")
  expect(await session.resetPassword("demo@example.test", "changed", "246810")).toBe(true)
  expect(storage.read()).toBeNull()
  await expect(session.login("demo", "Demo-only-24")).rejects.toMatchObject({
    key: "authInvalidCredentials"
  })
  expect((await session.login("demo", "changed")).user.id).toBe("demo")
})

/** 导航取消不删除有效会话；迟到签发不覆盖后来登录。 */
test("stale login and cancelled navigation cannot replace a new session", async () => {
  const { api, storage, query } = fixture()
  let release: (() => void) | undefined
  const original = api.login.bind(api)
  api.login = async (username, password) => {
    const issued = await original(username, password)
    if (username === "demo")
      await new Promise<void>((resolve) => {
        release = resolve
      })
    return issued
  }
  const session = createAccountSession(api, storage, query)
  const late = session.login("demo", "Demo-only-24").catch((error: unknown) => error)
  await vi.waitFor(() => expect(release).toBeTypeOf("function"))
  await session.login("viewer", "Demo-only-24")
  release?.()
  await expect(late).resolves.toMatchObject({ key: "authSuperseded" })
  expect(session.context()?.user.id).toBe("viewer")
  const abort = new AbortController()
  abort.abort()
  await expect(session.check(abort.signal)).rejects.toMatchObject({ key: "authSuperseded" })
  expect(storage.read()).not.toBeNull()
})
/** Mock 通过公开 API 零联网；返回目标拒绝外站、未知路径和协议相对地址。 */

})

/** 新凭据尚未收到 storage 事件时，断网不得复用旧用户权限。 */
test("network fallback is limited to the same confirmed credential", async () => {
  const { session, api, storage } = fixture()
  await session.login("demo", "Demo-only-24")
  const next = await api.login("viewer", "Demo-only-24")
  storage.write(next)
  vi.spyOn(api, "validate").mockRejectedValueOnce(new AuthFailure("authNetwork"))
  await expect(session.check()).rejects.toMatchObject({ key: "authNetwork" })
  expect(session.context()).toBeUndefined()
  expect((await session.check()).user.id).toBe("viewer")
  vi.spyOn(api, "validate").mockRejectedValueOnce(new AuthFailure("authNetwork"))
  expect((await session.check()).user.id).toBe("viewer")
})

/** 旧校验失效的迟到结果不能删除后来已确认的凭据。 */

/** 两个会话实例模拟独立标签页，锁内复核防止迟到登录覆盖新签发。 */

/** 旧退出等待锁时，新登录的 storage 信号不能继承旧删除阻断。 */
test("queued logout does not block a new external login", async () => {
  const { session, api, storage } = fixture()
  await session.login("demo", "Demo-only-24")
  const next = await api.login("viewer", "Demo-only-24")
  let release: (() => void) | undefined
  let queued = true
  Object.defineProperty(navigator, "locks", {
    configurable: true,
    value: {
      request: async (_name: string, operation: () => Promise<unknown>) => {
        if (queued) {
          queued = false
          await new Promise<void>((resolve) => {
            release = resolve
          })
        }
        return operation()
      }
    }
  })
  const stop = session.listenStorage()
  try {
    const exiting = session.logout()
    storage.write(next)
    window.dispatchEvent(new StorageEvent("storage", { key: storage.key }))
    await vi.waitFor(() => expect(session.context()?.user.id).toBe("viewer"))
    release?.()
    expect(await exiting).toBe(false)
    expect((await session.check()).user.id).toBe("viewer")
    expect(storage.read()).toBe(next)
  } finally {
    stop()
  }
})

/** 浏览器存储访问拒绝和损坏记录都经过统一分类与真实清理。 */
test("storage denial is explicit and malformed credentials are removed", () => {
  const { storage } = fixture()
  localStorage.setItem(storage.key, '{"version":1,"token":"demo","permissions":["all"]}')
  expect(storage.read()).toBeNull()
  expect(localStorage.getItem(storage.key)).toBeNull()
  const original = Object.getOwnPropertyDescriptor(window, "localStorage")
  if (original === undefined) throw new Error("Missing test Storage descriptor")
  try {
    Object.defineProperty(window, "localStorage", {
      configurable: true,
      get: () => {
        throw new Error("denied")
      }
    })
    expect(() => storage.read()).toThrow(AuthFailure)
  } finally {
    Object.defineProperty(window, "localStorage", original)
  }
})

/** 已发起身份转换时，新导航不能重新采用仍存储的旧身份或组织。 */
test("blocks navigation checks during login and organization transitions", async () => {
  const { api, session, storage } = fixture()
  await session.login("demo", "Demo-only-24")
  const originalLogin = api.login.bind(api)
  let finishLogin: (() => void) | undefined
  api.login = async (username, password) => {
    await new Promise<void>(resolve => { finishLogin = resolve })
    return originalLogin(username, password)
  }
  const login = session.login("viewer", "Demo-only-24")
  await expect(session.check()).rejects.toMatchObject({ key: "authSuperseded" })
  expect(session.context()).toBeUndefined()
  expect(session.getSnapshot().phase).toBe("restoring")
  finishLogin?.()
  await login
  expect(session.context()?.user.id).toBe("viewer")
  api.login = originalLogin
  await session.login("demo", "Demo-only-24")
  const originalSwitch = api.switchOrganization.bind(api)
  let finishSwitch: (() => void) | undefined
  api.switchOrganization = async (token, organizationId) => {
    await new Promise<void>(resolve => { finishSwitch = resolve })
    return originalSwitch(token, organizationId)
  }
  const switching = session.switchOrganization("restricted-team")
  await expect(session.check()).rejects.toMatchObject({ key: "authSuperseded" })
  expect(session.context()).toBeUndefined()
  finishSwitch?.()
  await switching
  expect(session.context()?.organization?.id).toBe("restricted-team")
  expect(storage.read()).not.toBeNull()
})

/** 取消只保留同凭据已确认上下文；已发现外部凭据变化必须撤下旧权限。 */
test("settles cancelled checks and removes stale context after external credential changes", async () => {
  const { api, session, storage } = fixture()
  await session.login("demo", "Demo-only-24")
  const validate = api.validate.bind(api)
  let finish: (() => void) | undefined
  api.validate = async token => {
    await new Promise<void>(resolve => { finish = resolve })
    return validate(token)
  }
  const abort = new AbortController()
  const checking = session.check(abort.signal)
  abort.abort()
  finish?.()
  await expect(checking).rejects.toMatchObject({ key: "authSuperseded" })
  expect(session.getSnapshot().phase).toBe("ready")
  expect(session.context()?.user.id).toBe("demo")
  const changed = session.check()
  storage.write("external-credential")
  finish?.()
  await expect(changed).rejects.toMatchObject({ key: "authSuperseded" })
  expect(session.context()).toBeUndefined()
  expect(session.getSnapshot().phase).toBe("anonymous")
  expect(storage.read()).toBe("external-credential")
})

/** 转换失败的异步清理完成前，不恢复旧组织或清掉后来转换的保护。 */
test("keeps failed scope transition blocked throughout asynchronous cleanup", async () => {
  const { api, session, storage } = fixture()
  await session.login("demo", "Demo-only-24")
  const oldToken = storage.read()
  api.switchOrganization = async () => "issued-scope"
  const validate = api.validate.bind(api)
  api.validate = async value => {
    if (value === "issued-scope") throw new AuthFailure("authNetwork")
    return validate(value)
  }
  let releaseCleanup: (() => void) | undefined
  api.logout = async () => { await new Promise<void>(resolve => { releaseCleanup = resolve }) }
  const failure = session.switchOrganization("restricted-team").catch((error: unknown) => error)
  await vi.waitFor(() => expect(releaseCleanup).toBeTypeOf("function"))
  await expect(session.check()).rejects.toMatchObject({ key: "authSuperseded" })
  expect(session.context()).toBeUndefined()
  expect(storage.read()).toBe(oldToken)
  releaseCleanup?.()
  await expect(failure).resolves.toMatchObject({ key: "authNetwork" })
  await expect(session.check()).rejects.toMatchObject({ key: "authStorageError" })
  expect(session.context()).toBeUndefined()
})
