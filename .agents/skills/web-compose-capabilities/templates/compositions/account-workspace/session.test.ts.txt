import { expect, test, vi } from "vitest"

/** 登录、刷新恢复、按实际导航重新验证与退出只影响私有 scope。 */
test("restores persisted demo sessions and validates every navigation", async () => {
  const fixture = authFixture()
  const validate = vi.spyOn(fixture.api, "validateSession")
  await fixture.session.login("demo", "demo-password")
  expect(fixture.session.getSnapshot().phase).toBe("ready")
  expect(canAccess(fixture.session.getContext(), "account")).toBe(true)
  expect(fixture.storage.read().raw).not.toContain("demo-password")
  expect(fixture.storage.read().raw).not.toContain("capabilities")
  fixture.queryClient.setQueryData(["public-data"], "retained")
  await fixture.session.validateNavigation(new AbortController().signal)
  await fixture.session.validateNavigation(new AbortController().signal)
  expect(validate).toHaveBeenCalledTimes(3)
  const restored = createAuthSession({
    api: fixture.api,
    storage: fixture.storage,
    queryClient: fixture.queryClient,
    clearHttpSession: () => undefined
  })
  await restored.restore()
  expect(restored.getSnapshot().phase).toBe("ready")
  await restored.logout()
  expect(fixture.storage.read().raw).toBeNull()
  expect(fixture.queryClient.getQueriesData({ queryKey: ["auth-private"] })).toEqual([])
  expect(fixture.queryClient.getQueryData(["public-data"])).toBe("retained")
  fixture.queryClient.clear()
})

/** 冷启动网络失败保留引用但没有权限；明确校验失效统一清理。 */
test("does not grant cold offline permissions and clears invalid sessions", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const offlineApi = {
    ...fixture.api,
    validateSession: vi.fn().mockRejectedValue(new AuthError("connectionUnavailable", "network"))
  }
  const restored = createAuthSession({
    api: offlineApi,
    storage: fixture.storage,
    queryClient: fixture.queryClient,
    clearHttpSession: () => undefined
  })
  await restored.restore()
  expect(restored.getSnapshot().phase).toBe("offline")
  expect(restored.getContext()).toBeUndefined()
  expect(fixture.storage.read().credential).not.toBeNull()
  offlineApi.validateSession.mockRejectedValue(new AuthError("sessionExpired", "invalid-session"))
  await restored.restore()
  expect(restored.getSnapshot().phase).toBe("anonymous")
  expect(fixture.storage.read().raw).toBeNull()
  fixture.queryClient.clear()
})

/** 不可写持久存储阻断登录，禁止只在 UI 假装已登录。 */
test("refuses login when the credential cannot be persisted", async () => {
  const fixture = authFixture()
  const revoke = vi.spyOn(fixture.api, "logout")
  const discard = vi.spyOn(fixture.api, "discardUnpublishedSession")
  fixture.backend.setItem = () => {
    throw new DOMException("Denied", "QuotaExceededError")
  }
  await expect(fixture.session.login("demo", "demo-password")).rejects.toMatchObject({
    code: "storageUnavailable"
  })
  expect(revoke).not.toHaveBeenCalled()
  expect(discard).toHaveBeenCalledTimes(1)
  expect(fixture.session.getSnapshot().phase).not.toBe("ready")
  expect(fixture.session.getContext()).toBeUndefined()
  fixture.queryClient.clear()
})

/** 退出清理失败立即隐藏数据，后续自动恢复被禁止直到成功重试。 */
test("blocks restoration after deletion failure and recovers through explicit cleanup", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const remove = fixture.backend.removeItem
  fixture.backend.removeItem = () => {
    throw new Error("Denied")
  }
  await fixture.session.logout()
  expect(fixture.session.getSnapshot()).toMatchObject({
    phase: "blocked",
    issue: "storageUnavailable"
  })
  expect(fixture.session.getContext()).toBeUndefined()
  await fixture.session.restore()
  expect(fixture.session.getSnapshot().phase).toBe("blocked")
  fixture.backend.removeItem = remove
  await fixture.session.retryCleanup()
  expect(fixture.session.getSnapshot().phase).toBe("anonymous")
  expect(fixture.storage.read().raw).toBeNull()
  fixture.queryClient.clear()
})

/** 组织请求可能已提交时不回滚旧权限，恢复后获取权威新组织。 */

/** 异步上下文即使忽略取消，也不能在退出后写回旧身份。 */
test("ignores late private reads after logout", async () => {
  const fixture = authFixture()
  const original = fixture.api.getContext
  let finish: (() => void) | undefined
  let started: (() => void) | undefined
  const ready = new Promise<void>((resolve) => {
    started = resolve
  })
  const pause = new Promise<void>((resolve) => {
    finish = resolve
  })
  fixture.api.getContext = async (credential, signal) => {
    const result = await original(credential, signal)
    started!()
    await pause
    return result
  }
  const pending = fixture.session.login("demo", "demo-password").catch(() => undefined)
  await ready
  await fixture.session.logout()
  finish!()
  await pending
  expect(fixture.session.getSnapshot().phase).toBe("anonymous")
  expect(fixture.session.getContext()).toBeUndefined()
  expect(fixture.storage.read().raw).toBeNull()
  fixture.queryClient.clear()
})

/** 跨标签页变化重读真实记录，旧退出不能删除新登录。 */
test("synchronizes external sessions and rejects stale logout deletion", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const otherFixture = authFixture({}, fixture.backend, fixture.service)
  const other = otherFixture.session
  await other.login("demo", "demo-password")
  const current = fixture.storage.read()
  await fixture.session.logout()
  expect(fixture.storage.read()).toEqual(current)
  expect(fixture.session.getSnapshot().phase).toBe("blocked")
  await other.switchOrganization("studio")
  const observerFixture = authFixture({}, fixture.backend, fixture.service)
  const observer = observerFixture.session
  await observer.restore()
  expect(observer.getContext()?.currentOrganization?.id).toBe("studio")
  await other.logout()
  await observer.synchronize()
  expect(observer.getSnapshot().phase).toBe("anonymous")
  fixture.queryClient.clear()
  otherFixture.queryClient.clear()
  observerFixture.queryClient.clear()
})

/** 深链接只允许已声明内部页面，未知/空权限从不默认放行。 */

})

/** 真实标签页各有 API 内存与 Query，外部新记录采用后不能回写旧组织。 */
test("adopts organization changes across independent tab runtimes", async () => {
  const first = authFixture()
  const second = authFixture({}, first.backend, first.service)
  await first.session.login("demo", "demo-password")
  await second.session.restore()
  await first.session.switchOrganization("studio")
  const committed = first.storage.read().raw
  await second.session.synchronize()
  expect(second.session.getContext()?.currentOrganization?.id).toBe("studio")
  expect(first.storage.read().raw).toBe(committed)
  await first.session.logout()
  await second.session.logout()
  expect(second.session.getSnapshot().phase).toBe("anonymous")
  first.queryClient.clear()
  second.queryClient.clear()
})

/** 快速连续导航的 Query 具有独立代次，旧请求不能取消新导航或永久挂起。 */
test("settles the latest navigation when private reads overlap", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const original = fixture.api.getContext
  let release: (() => void) | undefined
  let started: (() => void) | undefined
  const entered = new Promise<void>((resolve) => {
    started = resolve
  })
  const paused = new Promise<void>((resolve) => {
    release = resolve
  })
  let count = 0
  fixture.api.getContext = async (credential, signal) => {
    const result = await original(credential, signal)
    if (++count === 1) {
      started!()
      await paused
    }
    return result
  }
  const first = fixture.session.validateNavigation(new AbortController().signal)
  await entered
  const second = fixture.session.validateNavigation(new AbortController().signal)
  await second
  release!()
  await first
  expect(fixture.session.getSnapshot().phase).toBe("ready")
  expect(fixture.session.getContext()?.user.username).toBe("demo")
  fixture.queryClient.clear()
})

/** 切组织中的导航保持私有内容隐藏，不能提前恢复旧权限。 */
test("does not restore an old scope while organization mutation is pending", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const original = fixture.api.switchOrganization
  let release: (() => void) | undefined
  const paused = new Promise<void>((resolve) => {
    release = resolve
  })
  fixture.api.switchOrganization = async (...args) => {
    await paused
    return original(...args)
  }
  const pending = fixture.session.switchOrganization("studio")
  await fixture.session.validateNavigation(new AbortController().signal)
  expect(fixture.session.getSnapshot().phase).toBe("switching")
  expect(fixture.session.getContext()).toBeUndefined()
  release!()
  await pending
  expect(fixture.session.getContext()?.currentOrganization?.id).toBe("studio")
  fixture.queryClient.clear()
})

/** 校验端明确响应错误清理会话；只有无响应网络/超时保留引用。 */
test.each(["http", "invalid-response", "network", "timeout"] as const)(
  "classifies validation %s failures",
  async (kind) => {
    const fixture = authFixture()
    await fixture.session.login("demo", "demo-password")
    fixture.api.validateSession = async () => {
      throw new ApiError({
        kind,
        code: "test",
        message: "private detail",
        retryable: false,
        status: kind === "http" ? 500 : undefined
      })
    }
    await fixture.session.validateNavigation(new AbortController().signal)
    const retained = kind === "network" || kind === "timeout"
    expect(fixture.session.getSnapshot().phase).toBe(retained ? "offline" : "anonymous")
    expect(fixture.storage.read().credential !== null).toBe(retained)
    expect(fixture.session.getContext()).toBeUndefined()
    fixture.queryClient.clear()
  }
)

/** 外部新引用采用中断后保留待确认标志，重试不能回写旧组织。 */
test("retries adoption after offline cross-tab synchronization", async () => {
  const first = authFixture()
  const second = authFixture({}, first.backend, first.service)
  await first.session.login("demo", "demo-password")
  await second.session.restore()
  await first.session.switchOrganization("studio")
  const validate = second.api.validateSession
  second.api.validateSession = async () => {
    throw new AuthError("connectionUnavailable", "network")
  }
  await second.session.synchronize()
  expect(second.session.getSnapshot().phase).toBe("offline")
  second.api.validateSession = validate
  await second.session.validateNavigation(new AbortController().signal)
  expect(second.session.getContext()?.currentOrganization?.id).toBe("studio")
  expect(first.storage.read().credential?.organizationId).toBe("studio")
  first.queryClient.clear()
  second.queryClient.clear()
})
