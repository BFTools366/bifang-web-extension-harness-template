import { Button, Group, Paper, SimpleGrid, Stack, Text, Title } from "@mantine/core"
import { Link } from "@tanstack/react-router"
import { useTranslation } from "react-i18next"
import { useAuth } from "../auth/AuthProvider"
import { canAccess } from "../auth/access"
import { AuthStatePanel } from "../auth/AuthStatePanel"
import { ClientIcon, type ClientIconName } from "./ClientIcon"
import classes from "./client.module.css"

/** 顶部信息卡只展示已取得的账号上下文，不虚构余额、配额和业务统计。 */
function ContextCard({
  icon,
  label,
  value
}: {
  icon: ClientIconName
  label: string
  value: string
}) {
  return (
    <Paper radius="lg" p="lg" className={classes.contextCard}>
      <Text c="brand" className={classes.contextIcon}>
        <ClientIcon name={icon} size={28} />
      </Text>
      <Stack gap="xs" miw={0}>
        <Text c="dimmed" size="sm">
          {label}
        </Text>
        <Text size="xl" fw={650} className={classes.contextValue}>
          {value}
        </Text>
      </Stack>
    </Paper>
  )
}

/** 顶部上下文卡与工作面板共享当前身份上下文。 */
export function DashboardPage() {
  const { t } = useTranslation()
  const { context } = useAuth()
  if (!canAccess(context, "workspace") || !context) return <AuthStatePanel denied />
  return (
    <Stack gap="lg">
      <div className={classes.pageHeading}>
        <Title order={1} size="h2">
          {t("client.welcome", { name: context.user.displayName })}
        </Title>
        <Text c="dimmed" size="sm" mt="xs">
          {t("client.workspaceDescription")}
        </Text>
      </div>
      <SimpleGrid cols={{ base: 1, md: 3 }} spacing="lg">
        <ContextCard icon="account" label={t("client.account")} value={context.user.displayName} />
        <ContextCard
          icon="organization"
          label={t("client.currentOrganization")}
          value={context.currentOrganization?.name ?? t("auth.states.noOrganization")}
        />
        <ContextCard
          icon="workspace"
          label={t("client.availableOrganizations")}
          value={t("client.organizationCount", { count: context.organizations.length })}
        />
      </SimpleGrid>
      <Paper radius="lg" className={classes.workspacePanel}>
        <Group p="lg" className={classes.panelHeader}>
          <Title order={2} size="h4">
            {t("client.myWorkspace")}
          </Title>
        </Group>
        <div className={classes.empty}>
          <div className={classes.emptyIcon}>
            <ClientIcon name="workspace" size={36} />
          </div>
          <Title order={3} size="h3">
            {t("client.emptyTitle")}
          </Title>
          <Text c="dimmed" maw="34em" size="sm">
            {t("client.emptyDescription")}
          </Text>
          {canAccess(context, "account") && (
            <Button component={Link} to="/app/account" variant="light">
              {t("client.viewAccount")}
            </Button>
          )}
        </div>
        <Text role="note" className={classes.demoFootnote} size="xs" c="dimmed">
          {t("auth.workspaceDemoDescription")}
        </Text>
      </Paper>
    </Stack>
  )
}
