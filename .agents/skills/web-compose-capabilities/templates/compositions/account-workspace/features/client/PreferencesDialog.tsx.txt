import { Modal } from "@mantine/core"
import { useTranslation } from "react-i18next"
import { Preferences } from "./Preferences"

/** 认证页和账号菜单共用偏好窗口，关闭后返回原操作入口。 */
export function PreferencesDialog({
  opened,
  onClose,
  onExited,
  returnFocus = true
}: {
  opened: boolean
  onClose: () => void
  onExited?: () => void
  returnFocus?: boolean
}) {
  const { t } = useTranslation()
  return (
    <Modal
      opened={opened}
      onClose={onClose}
      title={t("client.preferences")}
      centered
      returnFocus={returnFocus}
      onExitTransitionEnd={onExited}
      closeButtonProps={{ "aria-label": t("client.close") }}
    >
      <Preferences plain />
    </Modal>
  )
}
