import { act, screen, waitFor, within } from "@testing-library/react"
import { expect, test, vi } from "vitest"
import { authFixture } from "../auth/authTestFixtures"
import { AuthError } from "../../data/api/auth/contracts"
import { mountClient } from "./clientTestFixtures"
/** 慢子路由只更新正文，账号菜单与侧栏身份节点不卸载、不清空。 */
test("preserves the account and sidebar nodes while a child route is checked", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const { user, session, api, router } = mountClient("/app/dashboard", fixture)
  await screen.findByRole("heading", { name: "Welcome back, demo" })
  await waitFor(() => expect(router.state.status).toBe("idle"))
  const sidebar = screen.getByRole("navigation", { name: "Client sidebar" })
  const account = screen.getByRole("button", { name: "Account menu" })
  const organization = within(sidebar).getByText("Example workspace")
  const identity = within(sidebar).getByText("demo")
  const generation = session.getSnapshot().generation
  const previousContext = session.getContext()
  const original = api.validateSession
  let release = () => {}
  let started = () => {}
  const ready = new Promise<void>((resolve) => {
    started = resolve
  })
  const paused = new Promise<void>((resolve) => {
    release = resolve
  })
  api.validateSession = async (...args) => {
    started()
    await paused
    return original(...args)
  }
  await user.click(within(sidebar).getByRole("link", { name: "Current account" }))
  await ready
  expect(session.getSnapshot()).toMatchObject({ phase: "checking", generation })
  expect(session.getContext()).toBe(previousContext)
  expect(screen.getByRole("button", { name: "Account menu" })).toBe(account)
  expect(screen.getByRole("navigation", { name: "Client sidebar" })).toBe(sidebar)
  expect(organization.isConnected).toBe(true)
  expect(identity.isConnected).toBe(true)
  expect(
    screen.queryByRole("heading", { level: 1, name: "Current account" })
  ).not.toBeInTheDocument()
  await act(async () => {
    release()
  })
  await screen.findByRole("heading", { level: 1, name: "Current account" })
  expect(screen.getByRole("button", { name: "Account menu" })).toBe(account)
  expect(within(sidebar).getByText("Example workspace")).toBe(organization)
  expect(within(sidebar).getByText("demo")).toBe(identity)
  expect(session.getSnapshot()).toMatchObject({ phase: "ready", generation })
})

/** 展示旧身份不授权新页；校验拒绝后立即隐藏布局并返回登录。 */
test("clears the preserved shell when child navigation is rejected", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const { user, api, router } = mountClient("/app/dashboard", fixture)
  await screen.findByRole("heading", { name: "Welcome back, demo" })
  await waitFor(() => expect(router.state.status).toBe("idle"))
  api.validateSession = async () => {
    throw new AuthError("sessionExpired", "invalid-session")
  }
  await user.click(
    within(screen.getByRole("navigation", { name: "Client sidebar" })).getByRole("link", {
      name: "Current account"
    })
  )
  await screen.findByRole("heading", { name: "Welcome back" })
  expect(router.state.location.pathname).toBe("/login")
  expect(screen.queryByRole("button", { name: "Account menu" })).not.toBeInTheDocument()
  expect(screen.queryByRole("navigation", { name: "Client sidebar" })).not.toBeInTheDocument()
  expect(screen.queryByText("demo@example.invalid")).not.toBeInTheDocument()
})

/** 认证子页共享稳定页脚，切换表单清理旧输入并保留安全返回目标。 */
test("shares the auth layout across sign-in, recovery and registration", async () => {
  const { user, router } = mountClient("/login?returnTo=/app/account")
  await screen.findByRole("heading", { name: "Welcome back" })
  const footer = screen.getByRole("contentinfo")
  await user.type(screen.getByLabelText("Username or email", { exact: false }), "draft-user")
  await user.click(screen.getByRole("link", { name: "Recover password" }))
  await screen.findByRole("heading", { name: "Recover your password" })
  expect(router.state.location.pathname).toBe("/forgot-password")
  expect(router.state.location.search).toEqual({ returnTo: "/app/account" })
  expect(screen.getByRole("contentinfo")).toBe(footer)
  expect(screen.getByLabelText("Confirm new password", { exact: false })).toHaveAttribute(
    "autocomplete",
    "new-password"
  )
  await user.click(screen.getByRole("link", { name: "Back to sign in" }))
  await screen.findByRole("heading", { name: "Welcome back" })
  expect(screen.getByLabelText("Username or email", { exact: false })).toHaveValue("")
  await user.click(screen.getByRole("link", { name: "Need an account? Register" }))
  await screen.findByRole("heading", { name: "Create your account" })
  expect(screen.getByRole("contentinfo")).toBe(footer)
  expect(router.state.location.search).toEqual({ returnTo: "/app/account" })
})

/** 找回的新密码规则在提交前可见，校验失败聚焦密码且不发送重置。 */
test("validates recovery password strength before calling the API", async () => {
  const { user, api } = mountClient("/forgot-password")
  const reset = vi.spyOn(api, "resetPassword")
  await screen.findByRole("heading", { name: "Recover your password" })
  await user.type(screen.getByLabelText("Email address", { exact: false }), "demo@example.invalid")
  const password = screen.getByLabelText("New password", {
    exact: false,
    selector: 'input[name="password"]'
  })
  await user.type(password, "weakpassword")
  await user.type(screen.getByLabelText("Confirm new password", { exact: false }), "weakpassword")
  await user.click(screen.getByRole("button", { name: "Send code" }))
  await screen.findByText(/Demo code: 246810/)
  await user.type(screen.getByLabelText("Email verification code", { exact: false }), "246810")
  await user.click(screen.getByRole("button", { name: "Reset password" }))
  expect(
    screen.getByText("Use an 8–64 character password containing letters and digits.")
  ).toBeInTheDocument()
  expect(password).toHaveFocus()
  expect(reset).not.toHaveBeenCalled()
})

/** 退出Storage锁尚未释放也立即隐藏旧身份，不能因保留Query observer泄露失效正文。 */
test("hides private identity before invalid-session storage cleanup completes", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const { user, api, storage, router } = mountClient("/app/dashboard", fixture)
  await screen.findByRole("heading", { name: "Welcome back, demo" })
  await waitFor(() => expect(router.state.status).toBe("idle"))
  const original = storage.commit
  let release = () => {}
  const paused = new Promise<void>((resolve) => {
    release = resolve
  })
  storage.commit = async (...args) => {
    if (args[1] === null) await paused
    return original(...args)
  }
  api.validateSession = async () => {
    throw new AuthError("sessionExpired", "invalid-session")
  }
  await user.click(
    within(screen.getByRole("navigation", { name: "Client sidebar" })).getByRole("link", {
      name: "Current account"
    })
  )
  await screen.findByRole("heading", { name: "Signing out" })
  expect(screen.queryByRole("button", { name: "Account menu" })).not.toBeInTheDocument()
  expect(screen.queryByRole("heading", { name: "Welcome back, demo" })).not.toBeInTheDocument()
  expect(screen.queryByText("Example workspace")).not.toBeInTheDocument()
  await act(async () => {
    release()
  })
  await screen.findByRole("heading", { name: "Welcome back" })
  expect(storage.read().raw).toBeNull()
})

/** 校验中转向公开入口仍按已确认会话分流，不暂时放行登录表单。 */
test("redirects public entry navigation while a private check is pending", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const { api, router } = mountClient("/app/dashboard", fixture)
  await screen.findByRole("heading", { name: "Welcome back, demo" })
  await waitFor(() => expect(router.state.status).toBe("idle"))
  const original = api.validateSession
  let release = () => {}
  let started = () => {}
  let count = 0
  const ready = new Promise<void>((resolve) => {
    started = resolve
  })
  const paused = new Promise<void>((resolve) => {
    release = resolve
  })
  api.validateSession = async (...args) => {
    if (++count === 1) {
      started()
      await paused
    }
    return original(...args)
  }
  let pending: Promise<void> | undefined
  act(() => {
    pending = router.navigate({ to: "/app/account" })
  })
  await ready
  await act(async () => {
    await router.navigate({ to: "/login" })
  })
  expect(router.state.location.pathname).toBe("/app/dashboard")
  expect(screen.queryByRole("heading", { name: "Welcome back" })).not.toBeInTheDocument()
  await act(async () => {
    release()
    await pending
  })
  expect(await screen.findByRole("heading", { name: "Welcome back, demo" })).toBeInTheDocument()
})

/** 卸载只撤销页面路由后效，已开始的会话同步由协调器完成。 */
test("does not invalidate a disposed router after storage synchronization", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const { router, unmount } = mountClient("/app/dashboard", fixture)
  await screen.findByRole("heading", { name: "Welcome back, demo" })
  let release!: () => void
  const completion = new Promise<void>((resolve) => {
    release = resolve
  })
  const synchronize = vi.spyOn(fixture.session, "synchronize").mockReturnValue(completion)
  const invalidate = vi.spyOn(router, "invalidate")
  window.dispatchEvent(new StorageEvent("storage", { key: fixture.session.storageKey }))
  expect(synchronize).toHaveBeenCalledOnce()
  unmount()
  await act(async () => {
    release()
    await completion
  })
  expect(invalidate).not.toHaveBeenCalled()
})
