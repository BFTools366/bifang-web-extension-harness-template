import { Button, Container, Group, Stack, Text, Title } from "@mantine/core"
import { defaultParseSearch, Link, useLocation } from "@tanstack/react-router"
import { useTranslation } from "react-i18next"
import { Preferences } from "./Preferences"
import { settingsReturnTo } from "./settingsNavigation"
import classes from "./client.module.css"

/** 公开设置只操作已有主题和语言，标题与导航焦点继续由根级唯一拥有者处理。 */
export function SettingsPage() {
  const { t } = useTranslation()
  const { state } = useLocation()
  const backTo = new URL(
    settingsReturnTo("settingsOrigin" in state ? state.settingsOrigin : undefined),
    "https://settings.invalid"
  )
  return (
    <main id="main" tabIndex={-1} className={classes.settingsPage}>
      <Container size="sm" w="100%">
        <div>
          <Stack gap="xl">
            <div>
              <Group justify="space-between">
                <Title order={1}>{t("client.settings")}</Title>
                <Button
                  variant="subtle"
                  color="gray"
                  size="sm"
                  renderRoot={(props) => (
                    <Link
                      {...props}
                      to={backTo.pathname}
                      search={defaultParseSearch(backTo.search)}
                      hash={backTo.hash.slice(1)}
                    />
                  )}
                >
                  {t("client.settingsBack")}
                </Button>
              </Group>
              <Text c="dimmed" mt="xs">
                {t("client.settingsDescription")}
              </Text>
            </div>
            <Preferences plain />
            <Text size="sm" c="dimmed">
              {t("client.settingsImmediate")}
            </Text>
          </Stack>
        </div>
      </Container>
    </main>
  )
}
