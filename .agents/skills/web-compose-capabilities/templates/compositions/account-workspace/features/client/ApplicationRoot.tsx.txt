import { Outlet, useRouter, useRouterState } from "@tanstack/react-router"
import { useEffect, useRef } from "react"
import { useTranslation } from "react-i18next"
import { useAuth } from "../auth/AuthProvider"
import { clientPage } from "../auth/access"

const titles: Record<string, string> = {
  "/": "auth.login",
  "/login": "auth.login",
  "/register": "auth.register",
  "/forgot-password": "auth.forgotPassword",
  "/settings": "client.settings",
  "/settings/": "client.settings",
  "/app/forbidden": "auth.states.denied",
  "/app/forbidden/": "auth.states.denied"
}

/** 根级唯一的标题、完成导航焦点和跨页会话事件装配。 */
export function ApplicationRoot() {
  const { t, i18n } = useTranslation()
  const { session } = useAuth()
  const router = useRouter()
  const pathname = useRouterState({ select: (state) => state.location.pathname })
  const status = useRouterState({ select: (state) => state.status })
  const previousPath = useRef<string | undefined>(undefined)
  useEffect(() => {
    document.title = t("client.documentTitle", {
      page: t(clientPage(pathname)?.title ?? titles[pathname] ?? "notFoundTitle"),
      name: t("projectName")
    })
  }, [pathname, i18n.resolvedLanguage, t])
  useEffect(() => {
    if (status !== "idle" || previousPath.current === pathname) return
    const heading =
      document.querySelector<HTMLElement>("main h1") ?? document.querySelector<HTMLElement>("main")
    if (heading) {
      heading.tabIndex = -1
      heading.focus({ preventScroll: true })
      previousPath.current = pathname
    }
  }, [pathname, status])
  useEffect(() => {
    if (pathname.replace(/\/$/u, "").toLowerCase() === "/settings") return
    let active = true
    let eventRevision = 0
    const synchronize = (event: StorageEvent) => {
      if (event.key !== session.storageKey && event.key !== null) return
      const revision = ++eventRevision
      void session
        .synchronize()
        .then(async () => {
          if (active && revision === eventRevision) await router.invalidate({ sync: true })
        })
        .catch(() => {
          /* 会话错误由协调器显示，取消或销毁的路由不继续产生后效。 */
        })
    }
    window.addEventListener("storage", synchronize)
    return () => {
      active = false
      eventRevision++
      window.removeEventListener("storage", synchronize)
    }
  }, [session, router, pathname])
  return <Outlet />
}
