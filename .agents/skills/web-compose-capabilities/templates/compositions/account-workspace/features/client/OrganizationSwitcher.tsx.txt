import { useOperationGuard } from "../auth/useOperationGuard"
import { otherOrganizations } from "./organizationChoices"
import { Alert, Button, Group, Modal, NativeSelect, Stack, Text } from "@mantine/core"
import { useRouter } from "@tanstack/react-router"
import { useEffect, useState } from "react"
import { useTranslation } from "react-i18next"
import { useAuth } from "../auth/AuthProvider"
import { canAccess } from "../auth/access"
import { authIssue } from "../auth/session"

/** 仅在多个已确认成员组织存在时挂载，提交后等待完整新作用域。 */
export function OrganizationSwitcher({
  opened,
  onClose,
  onExited,
  returnFocus = true
}: {
  opened: boolean
  onClose: () => void
  onExited?: () => void
  returnFocus?: boolean
}) {
  const capture = useOperationGuard(opened)
  const { t } = useTranslation()
  const { context, state, session } = useAuth()
  const router = useRouter()
  const [target, setTarget] = useState("")
  const [pending, setPending] = useState(false)
  const [issue, setIssue] = useState<string>()
  const candidates = otherOrganizations(
    context?.organizations ?? [],
    context?.currentOrganization?.id
  )
  const canSubmit =
    state.phase === "ready" &&
    canAccess(context, "switch-organization") &&
    candidates.some((entry) => entry.id === target)
  useEffect(() => {
    if (opened) {
      setPending(false)
      setTarget("")
      setIssue(undefined)
    }
  }, [opened])
  useEffect(() => {
    if (
      ["anonymous", "restoring", "blocked", "offline", "recovery", "logging-out"].includes(
        state.phase
      ) ||
      (state.phase === "ready" && !canAccess(context, "switch-organization"))
    )
      onClose()
  }, [state.phase, context, onClose])
  return (
    <Modal
      opened={opened}
      onClose={onClose}
      title={t("auth.switchOrganization")}
      centered
      closeButtonProps={{ "aria-label": t("client.close") }}
      closeOnClickOutside={!pending}
      closeOnEscape={!pending}
      withCloseButton={!pending}
      returnFocus={returnFocus}
      onExitTransitionEnd={onExited}
    >
      <Stack gap="lg">
        <Text c="dimmed">{t("auth.switchDescription")}</Text>
        {issue && (
          <Alert color="red" role="alert">
            {t(`auth.errors.${issue}`, { defaultValue: t("auth.errors.operationFailed") })}
          </Alert>
        )}
        <NativeSelect
          label={t("auth.targetOrganization")}
          value={target}
          disabled={pending || state.phase !== "ready"}
          data={[
            { value: "", label: t("auth.selectOrganization") },
            ...candidates.map((entry) => ({ value: entry.id, label: entry.name }))
          ]}
          onChange={(event) => setTarget(event.currentTarget.value)}
        />
        {!pending && candidates.length === 0 && (
          <Text role="status">{t("auth.noOtherOrganizations")}</Text>
        )}
        <Group justify="flex-end" wrap="wrap">
          <Button variant="default" disabled={pending} onClick={onClose}>
            {t("client.cancel")}
          </Button>
          <Button
            disabled={!canSubmit}
            loading={pending}
            onClick={async () => {
              if (pending || !canSubmit || session.getSnapshot().phase !== "ready") return
              const current = capture()
              setPending(true)
              setIssue(undefined)
              try {
                await session.switchOrganization(target)
                if (current() && session.getSnapshot().phase === "ready") {
                  onClose()
                  await router.invalidate({ sync: true })
                }
              } catch (error) {
                if (current()) setIssue(authIssue(error))
              } finally {
                if (current()) setPending(false)
              }
            }}
          >
            {t("auth.confirmSwitch")}
          </Button>
        </Group>
      </Stack>
    </Modal>
  )
}
