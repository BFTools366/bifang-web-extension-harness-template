import { clientPages, safeReturnTo } from "../auth/access"

declare module "@tanstack/react-router" {
  /** 来源只在当前历史项保存，认证输入不进入设置地址或偏好存储。 */
  interface HistoryState { settingsOrigin?: string }
}

/** 设置页只返回已声明的认证或私有页面，认证查询按原合同逐项清洗。 */
export function settingsReturnTo(input: unknown): string {
  if (typeof input !== "string" || !input.startsWith("/") || input.startsWith("//") ||
    input.includes("\\") || /[\u0000-\u0020\u007f]/u.test(input)) return "/login"
  const url = new URL(input, "https://settings.invalid")
  if (url.origin !== "https://settings.invalid" || input.split(/[?#]/u)[0] !== url.pathname)
    return "/login"
  const pathname = (url.pathname.replace(/\/$/u, "") || "/").toLowerCase()
  if (["/login", "/register", "/forgot-password"].includes(pathname)) {
    const query = new URLSearchParams()
    if (url.searchParams.has("returnTo")) query.set("returnTo", safeReturnTo(url.searchParams.get("returnTo")))
    const notice = url.searchParams.get("notice")
    if (pathname === "/login" && notice && ["registered", "reset"].includes(notice)) query.set("notice", notice)
    return pathname + (query.size ? "?" + query.toString() : "")
  }
  if (clientPages.some((page) => page.path === pathname) || pathname === "/app/forbidden")
    return pathname + url.search + url.hash
  return "/login"
}
