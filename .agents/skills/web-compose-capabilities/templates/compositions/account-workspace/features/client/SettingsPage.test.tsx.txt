// @vitest-environment jsdom
import { act, screen } from "@testing-library/react"
import { expect, test, vi } from "vitest"
import { authFixture } from "../auth/authTestFixtures"
import { mountClient } from "./clientTestFixtures"
import { settingsReturnTo } from "./settingsNavigation"

/** 匿名直达设置和认证存储事件均不触发会话恢复或服务请求。 */
test("public settings never initializes or synchronizes authentication", async () => {
  const fixture = authFixture()
  const initialize = vi.spyOn(fixture.session, "initialize")
  const restore = vi.spyOn(fixture.session, "restore")
  const synchronize = vi.spyOn(fixture.session, "synchronize")
  const validate = vi.spyOn(fixture.api, "validateSession")
  const { session, router } = mountClient("/settings", fixture)
  expect(await screen.findByRole("heading", { name: "Settings", level: 1 })).toBeInTheDocument()
  expect(screen.getByRole("link", { name: "Back" })).toHaveAttribute("href", "/login")
  await act(async () => {
    window.dispatchEvent(new StorageEvent("storage", { key: session.storageKey }))
  })
  for (const spy of [initialize, restore, synchronize, validate]) expect(spy).not.toHaveBeenCalled()
  expect(router.state.location.pathname).toBe("/settings")
})

/** 三个认证入口均有底部设置链接，往返保留安全目标，字段具有占位提示。 */
test.each([
  ["/login", "Welcome back"],
  ["/register", "Create your account"],
  ["/forgot-password", "Recover your password"]
])("returns from settings to %s with the safe destination", async (path, title) => {
  const { user, router } = mountClient(path + "?returnTo=%2Fapp%2Faccount")
  await screen.findByRole("heading", { name: title })
  expect(screen.getAllByRole("link", { name: "Settings" })).toHaveLength(1)
  expect(screen.queryByRole("combobox")).not.toBeInTheDocument()
  for (const field of screen.getAllByRole("textbox")) expect(field).toHaveAttribute("placeholder")
  expect(screen.getByLabelText(/password/i, { selector: 'input[name="password"]' })).toHaveAttribute("placeholder")
  await user.click(screen.getByRole("link", { name: "Settings" }))
  await screen.findByRole("heading", { name: "Settings", level: 1 })
  expect(router.state.location.href).toBe("/settings")
  await user.click(screen.getByRole("link", { name: "Back" }))
  await screen.findByRole("heading", { name: title })
  expect(router.state.location.pathname).toBe(path)
  expect(router.state.location.search).toMatchObject({ returnTo: "/app/account" })
})

/** 设置来源拒绝外站、未知地址与路径逃逸，认证查询只保留可信字段。 */
test("settings return destinations fail closed", () => {
  for (const target of [undefined, "https://evil.invalid", "//evil.invalid", "/\\evil.invalid", "/unknown", "/settings", "/app/../login", "/%6cogin"])
    expect(settingsReturnTo(target)).toBe("/login")
  expect(settingsReturnTo("/register?returnTo=https%3A%2F%2Fevil.invalid&token=secret"))
    .toBe("/register?returnTo=%2Fapp%2Fdashboard")
})
