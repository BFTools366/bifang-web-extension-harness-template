import { hasCapability } from "./capability"
import type { AuthContext, ClientCapability } from "../../data/api/auth/contracts"

/** 每个菜单对应独立页面，标题、权限与登录返回目标复用这一合同。 */
export const clientPages = [
  { path: "/app/dashboard", capability: "workspace", title: "client.dashboard" },
  { path: "/app/account", capability: "account", title: "client.account" },
  { path: "/app/test", capability: "test-menu", title: "client.testMenu" }
] as const satisfies readonly {
  path: string
  capability: ClientCapability
  title: string
}[]

/** 已声明的私有页面路径；无权限页是有会话即可访问的恢复入口。 */
export type ClientPagePath = (typeof clientPages)[number]["path"] | "/app/forbidden"

/** 统一处理尾斜线，拒绝未知路径而不按父路径授予能力。 */
export function clientPage(pathname: string) {
  return clientPages.find((page) => page.path === pathname.replace(/\/$/, ""))
}

/** 菜单、深链接及操作共用当前组织的精确能力判定，未知能力不授予权限。 */
export function canAccess(context: AuthContext | undefined, capability: ClientCapability): boolean {
  return !!context?.currentOrganization && hasCapability(context.capabilities, capability)
}

/** 返回目标仅允许已有私有页面；外部地址、额外查询和未知路径全部回落。 */
export function safeReturnTo(value: unknown): ClientPagePath {
  const target = typeof value === "string" ? value.replace(/\/$/, "") : undefined
  return (
    clientPages.find((page) => page.path === target)?.path ??
    (target === "/app/forbidden" ? target : "/app/dashboard")
  )
}

/** 已声明返回目标失权时进入无权限页，不能静默把深链接改成其它业务页。 */
export function accessibleReturnTo(
  value: unknown,
  context: AuthContext | undefined
): ClientPagePath {
  const target = safeReturnTo(value)
  const page = clientPage(target)
  if (page && canAccess(context, page.capability)) return page.path
  return "/app/forbidden"
}
