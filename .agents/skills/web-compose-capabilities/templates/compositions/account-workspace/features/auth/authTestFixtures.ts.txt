import { QueryClient } from "@tanstack/react-query"
import { createMockService } from "../../data/api/auth/mockAuth"
import type { AuthApi } from "../../data/api/auth/contracts"
import { createAuthSessionStorage } from "../../data/storage/authSessionStorage"
import { createAuthSession } from "./session"

/** 测试只注入原生存储替身，不把产品存储逻辑一并 Mock。 */
export function memoryAuthStorage() {
  const entries = new Map<string, string>()
  const storage: Storage = {
    /** 模拟标准 Storage 的条目数量。 */
    get length() {
      return entries.size
    },
    clear: () => entries.clear(),
    getItem: (key) => entries.get(key) ?? null,
    key: (index) => [...entries.keys()][index] ?? null,
    setItem: (key, value) => {
      entries.set(key, value)
    },
    removeItem: (key) => {
      entries.delete(key)
    }
  }
  return storage
}

/** 只替换跨页锁调度，仍使用真实版本校验、冲突检查、Query 和认证协调器。 */
export function authFixture(
  overrides: Partial<AuthApi> = {},
  backend = memoryAuthStorage(),
  service = createMockService({ delayMs: 0, configured: () => false })
) {
  const api = { ...service.createClient().api, ...overrides }
  const queryClient = new QueryClient({
    defaultOptions: { queries: { retry: false }, mutations: { retry: false } }
  })
  const storage = createAuthSessionStorage(
    () => backend,
    async (_name, work) => work()
  )
  const session = createAuthSession({
    api,
    storage,
    queryClient,
    clearHttpSession: () => undefined
  })
  return { api, backend, queryClient, storage, session, service }
}
