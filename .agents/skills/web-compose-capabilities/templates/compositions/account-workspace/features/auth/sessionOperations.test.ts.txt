/** 显式刷新轮换仅在原子写入后公布，原组织保持。 */
test("refreshes through the coordinator without automatic navigation refresh", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  await fixture.session.switchOrganization("studio")
  const previous = fixture.storage.read()
  const refresh = vi.spyOn(fixture.api, "refresh")
  await fixture.session.refresh()
  expect(fixture.session.getSnapshot().phase).toBe("ready")
  expect(fixture.storage.read().credential).toMatchObject({
    accountId: "demo-user",
    organizationId: "studio"
  })
  expect(fixture.storage.read().credential?.sessionId).not.toBe(previous.credential?.sessionId)
  fixture.api.validateSession = async () => {
    throw new AuthError("sessionExpired", "invalid-session")
  }
  await fixture.session.validateNavigation(new AbortController().signal)
  expect(refresh).toHaveBeenCalledTimes(1)
  expect(fixture.session.getSnapshot().phase).toBe("anonymous")
  fixture.queryClient.clear()
})
/** 刷新存储失败丢弃新临时引用，不能撤销账号的旧会话。 */
test("discards an unpublished refresh when storage fails", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const previous = fixture.storage.read()
  const discard = vi.spyOn(fixture.api, "discardUnpublishedSession")
  fixture.backend.setItem = () => {
    throw new Error("Denied")
  }
  await expect(fixture.session.refresh()).rejects.toMatchObject({
    code: "storageUnavailable"
  })
  expect(discard).toHaveBeenCalledTimes(1)
  expect(fixture.storage.read()).toEqual(previous)
  expect(fixture.session.getContext()).toBeUndefined()
  await expect(
    fixture.api.validateSession(fixtureValue(previous.credential), new AbortController().signal)
  ).resolves.toMatchObject(fixtureValue(previous.credential))
  fixture.queryClient.clear()
})
/** 改密输入错误可恢复旧会话，成功只清本地，不额外发送已吊销的logout。 */
test("preserves the session on wrong old password and clears after success", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const logout = vi.spyOn(fixture.api, "logout")
  await expect(fixture.session.changePassword("wrong", "Changed123")).rejects.toMatchObject({
    code: "oldPasswordWrong"
  })
  expect(fixture.session.getSnapshot().phase).toBe("ready")
  await fixture.session.changePassword("demo-password", "Changed123")
  expect(fixture.session.getSnapshot().phase).toBe("anonymous")
  expect(fixture.storage.read().raw).toBeNull()
  expect(logout).not.toHaveBeenCalled()
  await expect(fixture.session.login("demo", "Changed123")).resolves.toBeUndefined()
  fixture.queryClient.clear()
})

/** 迟到重置不得删除后来其它账号的登录，也不能导航回旧表单结果。 */
test("does not clear a later login when a reset response arrives late", async () => {
  const fixture = authFixture()
  let release = () => {}
  let started = () => {}
  const ready = new Promise<void>((resolve) => {
    started = resolve
  })
  const wait = new Promise<void>((resolve) => {
    release = resolve
  })
  fixture.api.resetPassword = async () => {
    started()
    await wait
  }
  const reset = fixture.session.resetPassword(
    { email: "demo@example.invalid", password: "Changed123", code: "246810" },
    new AbortController().signal
  )
  await ready
  await fixture.session.login("test", "test-password")
  const current = fixture.storage.read()
  release()
  expect(await reset).toBe(false)
  expect(fixture.storage.read()).toEqual(current)
  expect(fixture.session.getContext()?.user.username).toBe("test")
  fixture.queryClient.clear()
})
/** 真实删除失败仍尝试一次服务吊销，恢复保持阻断。 */
test("attempts server revocation even when local deletion fails", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const credential = fixtureValue(fixture.storage.read().credential)
  const logout = vi.spyOn(fixture.api, "logout")

  expect(fixture.session.getSnapshot()).toMatchObject({

    fixture.api.validateSession(credential, new AbortController().signal)
  ).rejects.toMatchObject({ serviceCode: "auth.token.revoked" })
  fixture.queryClient.clear()
})
/** 两个客户端共享服务与存储，旧退出不能吊销替代登录。 */
test("does not revoke a replacement session after a storage conflict", async () => {
  const backend = createMockAuthBackend()
  const native = memoryAuthStorage()
  const storage = createAuthSessionStorage(
    () => native,
    async (_name, work) => work()
  )
  const firstApi = createMockService({ backend, delayMs: 0, configured: () => false }).api
  const secondApi = createMockService({ backend, delayMs: 0, configured: () => false }).api
  const firstQueries = new QueryClient()
  const secondQueries = new QueryClient()
  const first = createAuthSession({
    api: firstApi,
    storage,
    queryClient: firstQueries,
    clearHttpSession: () => {}
  })
  const second = createAuthSession({
    api: secondApi,
    storage,
    queryClient: secondQueries,
    clearHttpSession: () => {}
  })
  await first.login("demo", "demo-password")
  await second.login("demo", "demo-password")
  const current = storage.read()
  const logout = vi.spyOn(firstApi, "logout")
  await first.logout()
  expect(logout).not.toHaveBeenCalled()
  expect(storage.read()).toEqual(current)
  await second.validateNavigation(new AbortController().signal)
  expect(second.getSnapshot().phase).toBe("ready")
  firstQueries.clear()
  secondQueries.clear()
})
/** 冷恢复共享服务时保留已提交但还未写回Storage的权威组织。 */
test("does not roll back an authoritative organization during cold restoration", async () => {
  const backend = createMockAuthBackend()
  const firstService = createMockService({ backend, delayMs: 0, configured: () => false })
  const secondService = createMockService({ backend, delayMs: 0, configured: () => false })
  const fixture = authFixture(firstService.api)
  await fixture.session.login("demo", "demo-password")
  await firstService.api.switchOrganization(
    fixtureValue(fixture.storage.read().credential),
    "studio",
    new AbortController().signal
  )
  const queries = new QueryClient()
  const restored = createAuthSession({
    api: secondService.api,
    storage: fixture.storage,
    queryClient: queries,
    clearHttpSession: () => {}
  })
  await restored.restore()
  expect(restored.getSnapshot().phase).toBe("ready")
  expect(restored.getContext()?.currentOrganization?.id).toBe("studio")
  expect(fixture.storage.read().credential?.organizationId).toBe("studio")
  queries.clear()
  fixture.queryClient.clear()
})
/** 新登录取消尚未提交的旧注销，旧请求释放后不吊销新凭据。 */
test("cancels a pending logout when a new login starts", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const original = fixture.api.logout
  let release = () => {}
  let started = () => {}
  let oldSignal: AbortSignal | undefined
  const ready = new Promise<void>((resolve) => {
    started = resolve
  })
  const wait = new Promise<void>((resolve) => {
    release = resolve
  })
  fixture.api.logout = async (credential, signal) => {
    oldSignal = signal
    started()
    await wait
    return original(credential, signal)
  }
  const logout = fixture.session.logout()
  await ready
  await fixture.session.login("demo", "demo-password")
  const replacement = fixture.storage.read()
  expect(oldSignal?.aborted).toBe(true)
  release()
  await logout
  expect(fixture.storage.read()).toEqual(replacement)
  expect(fixture.session.getSnapshot().phase).toBe("ready")
  await expect(
    fixture.api.validateSession(fixtureValue(replacement.credential), new AbortController().signal)
  ).resolves.toMatchObject(fixtureValue(replacement.credential))
  fixture.queryClient.clear()
})
/** 刷新迟到只丢弃自己的临时引用，不覆盖或吊销后来新登录。 */
test("discards a late refresh without overwriting a later login", async () => {
  const fixture = authFixture()
  await fixture.session.login("demo", "demo-password")
  const original = fixture.api.refresh
  let release = () => {}
  let started = () => {}
  let issued: Awaited<ReturnType<typeof original>> | undefined
  const ready = new Promise<void>((resolve) => {
    started = resolve
  })
  const wait = new Promise<void>((resolve) => {
    release = resolve
  })
  fixture.api.refresh = async (...args) => {
    issued = await original(...args)
    started()
    await wait
    return issued
  }
  const discard = vi.spyOn(fixture.api, "discardUnpublishedSession")
  const pending = fixture.session.refresh().catch((error: unknown) => error)
  await ready
  await fixture.session.login("demo", "demo-password")
  const replacement = fixture.storage.read()
  release()
  expect(await pending).toMatchObject({ name: "AbortError" })
  expect(discard).toHaveBeenCalledWith(issued)
  expect(fixture.storage.read()).toEqual(replacement)
  expect(fixture.session.getSnapshot().phase).toBe("ready")
  await expect(
    fixture.api.validateSession(fixtureValue(replacement.credential), new AbortController().signal)
  ).resolves.toMatchObject(fixtureValue(replacement.credential))
  fixture.queryClient.clear()
})
/** 订阅者失败不能把已提交认证事务改判成失败或阻止后续监听者。 */
test("isolates observer exceptions from session commits", async () => {
  const fixture = authFixture()
  const observed = vi.fn()
  fixture.session.subscribe(() => {
    throw new Error("observer failed")
  })
  fixture.session.subscribe(observed)
  await fixture.session.login("demo", "demo-password")

})
