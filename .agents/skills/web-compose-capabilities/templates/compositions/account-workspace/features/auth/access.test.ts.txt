import { expect, test } from "vitest"
import { isAuthContext, type AuthContext } from "../../data/api/auth/contracts"
import { accessibleReturnTo, canAccess, clientPage, clientPages, safeReturnTo } from "./access"

const context: AuthContext = {
  user: {
    id: "test-user",
    username: "test",
    displayName: "test",
    email: "test@example.invalid"
  },
  organizations: [{ id: "workspace", name: "Example workspace" }],
  currentOrganization: { id: "workspace", name: "Example workspace" },
  capabilities: ["workspace", "account", "test-menu"]
}

/** 新权限只接受明确枚举，空权限、未知码和缺失组织均不能获得页面授权。 */
test("validates test permissions and rejects unknown or incomplete contexts", () => {
  expect(isAuthContext(context)).toBe(true)
  expect(isAuthContext({ ...context, capabilities: ["admin"] })).toBe(false)
  expect(isAuthContext({ ...context, capabilities: null })).toBe(false)
  expect(canAccess({ ...context, capabilities: [] }, "test-menu")).toBe(false)
  expect(canAccess({ ...context, currentOrganization: null }, "test-menu")).toBe(false)
  expect(canAccess(undefined, "test-menu")).toBe(false)
})

/** 返回目标只接收已知页面，失权深链接进入独立无权限页。 */
test("preserves permitted targets and directs denied targets to the forbidden page", () => {
  expect(safeReturnTo("/app/test")).toBe("/app/test")
  expect(safeReturnTo("/app/test/")).toBe("/app/test")
  expect(safeReturnTo("/app/account/")).toBe("/app/account")
  expect(accessibleReturnTo("/app/test", context)).toBe("/app/test")
  expect(accessibleReturnTo("/app/test", { ...context, capabilities: ["workspace"] })).toBe(
    "/app/forbidden"
  )
  expect(accessibleReturnTo("/app/forbidden", context)).toBe("/app/forbidden")
  expect(accessibleReturnTo("/app/dashboard", { ...context, capabilities: [] })).toBe(
    "/app/forbidden"
  )
  for (const target of [
    "https://outside.invalid",
    "//outside.invalid",
    "/app/test?token=private",
    "/app/test/unknown",
    "javascript:alert(1)"
  ])
    expect(safeReturnTo(target)).toBe("/app/dashboard")
})

/** 每个菜单有唯一的路径与能力，尾斜线不把子页面误判成父菜单。 */
test("keeps menu routes unique and matches only exact page paths", () => {
  expect(new Set(clientPages.map((page) => page.path)).size).toBe(clientPages.length)
  expect(new Set(clientPages.map((page) => page.capability)).size).toBe(clientPages.length)
  expect(clientPage("/app/dashboard/")?.capability).toBe("workspace")
  expect(clientPage("/app")).toBeUndefined()
  expect(clientPage("/")).toBeUndefined()
  expect(safeReturnTo("/app")).toBe("/app/dashboard")
  expect(clientPage("/app/account/")?.capability).toBe("account")
  expect(clientPage("/app/test/")?.capability).toBe("test-menu")
  expect(clientPage("/app/test/unknown")).toBeUndefined()
})
