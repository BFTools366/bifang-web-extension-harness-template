import { redirect } from "@tanstack/react-router"
import type { AuthSession } from "./session"
import { canAccess, clientPage, safeReturnTo, type ClientPagePath } from "./access"

/** beforeLoad 在私有页面挂载前校验会话，取消和预取不能授权实际导航。 */
export async function protectRoute(
  session: AuthSession,
  target: string,
  signal: AbortSignal,
  preload = false
) {
  if (preload) return
  await session.validateNavigation(signal)
  signal.throwIfAborted()
  if (session.getSnapshot().phase === "anonymous")
    throw redirect({
      to: "/login",
      search: { returnTo: safeReturnTo(target.split(/[?#]/, 1)[0]) },
      replace: true
    })
}

/** 每个受限页面在挂载前按同一菜单合同判权，未知目标也默认拒绝。 */
export async function protectPage(
  session: AuthSession,
  target: ClientPagePath,
  signal: AbortSignal,
  preload = false
) {
  await protectRoute(session, target, signal, preload)
  const page = clientPage(target)
  if (
    !preload &&
    session.getSnapshot().phase === "ready" &&
    (!page || !canAccess(session.getContext(), page.capability))
  )
    throw redirect({ to: "/app/forbidden", replace: true })
}

/** 已恢复登录的用户不重复填写认证表单，避免公共页与私有页循环。 */
export async function publicAuthRoute(session: AuthSession, returnTo: unknown, preload = false) {
  if (preload) return
  await session.initialize()
  if (session.getSnapshot().phase === "ready" || session.getSnapshot().phase === "checking")
    throw redirect({ to: safeReturnTo(returnTo), replace: true })
}
