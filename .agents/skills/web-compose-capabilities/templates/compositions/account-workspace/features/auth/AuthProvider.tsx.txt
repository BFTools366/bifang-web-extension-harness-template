import { createContext, useContext, useSyncExternalStore, type ReactNode } from "react"
import { skipToken, useQuery } from "@tanstack/react-query"
import type { AuthContext } from "../../data/api/auth/contracts"
import type { AuthSession } from "./session"

const SessionContext = createContext<AuthSession | null>(null)

/** 应用根只注入一个会话协调器；Provider 不复制任何账号状态。 */
export function AuthProvider({ session, children }: { session: AuthSession; children: ReactNode }) {
  return <SessionContext.Provider value={session}>{children}</SessionContext.Provider>
}

/** 界面读取协调状态并订阅唯一的 Query 上下文缓存。 */
export function useAuth() {
  const session = useContext(SessionContext)
  if (!session) throw new Error("AuthProvider is required")
  const state = useSyncExternalStore(session.subscribe, session.getSnapshot, session.getSnapshot)
  const { data } = useQuery<AuthContext>({
    queryKey: session.contextKey(),
    enabled: false,
    queryFn: skipToken,
    retry: false
  })
  return {
    session,
    state,
    context: state.phase === "ready" || state.phase === "checking" ? data : undefined
  }
}
