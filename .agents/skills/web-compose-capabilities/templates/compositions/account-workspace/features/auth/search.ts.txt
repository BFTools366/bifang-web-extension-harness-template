import { safeReturnTo, type ClientPagePath } from "./access"

/** 认证子页共同保留安全返回目标，成功提示仅由登录页消费。 */
export interface AuthSearch {
  returnTo?: ClientPagePath
  notice?: "registered" | "reset"
}

/** 公共路由不保存原始参数、凭据或任意外部返回地址。 */
export function authSearch(search: Record<string, unknown>): AuthSearch {
  return {
    returnTo: search.returnTo === undefined ? undefined : safeReturnTo(search.returnTo),
    notice: search.notice === "registered" || search.notice === "reset" ? search.notice : undefined
  }
}

/** 认证父布局只拥有可省略的返回目标，不把URL查询字段误建成必填参数。 */
export function authReturnSearch(search: Record<string, unknown>): Pick<AuthSearch, "returnTo"> {
  return { returnTo: authSearch(search).returnTo }
}
