import { useOperationGuard } from "./useOperationGuard"
import { Alert, Button, Container, Group, Loader, Paper, Stack, Text, Title } from "@mantine/core"
import { Link, useNavigate, useRouter } from "@tanstack/react-router"
import { useState } from "react"
import { useTranslation } from "react-i18next"
import { useAuth } from "./AuthProvider"
import { canAccess, clientPages } from "./access"

/** 未取得完整作用域时只显示恢复面板，不挂载任何旧私有正文。 */
export function AuthStatePanel({
  denied = false,
  noOrganization = false
}: {
  denied?: boolean
  noOrganization?: boolean
}) {
  const capture = useOperationGuard()
  const { t } = useTranslation()
  const { session, state, context } = useAuth()
  const router = useRouter()
  const navigate = useNavigate()
  const [busy, setBusy] = useState(false)
  const pending = ["restoring", "checking", "switching", "logging-out"].includes(state.phase)
  const title = noOrganization ? "noOrganization" : denied ? "denied" : state.phase
  const availableTarget = clientPages.find((page) => canAccess(context, page.capability))?.path
  return (
    <Container size="sm" py="xl">
      <Paper withBorder radius="lg" p="xl">
        <Stack gap="lg">
          <Title order={1}>{t(`auth.states.${title}`)}</Title>
          {pending ? (
            <Group role="status">
              <Loader size="sm" />
              <Text>{t("auth.waiting")}</Text>
            </Group>
          ) : (
            <>
              <Text c="dimmed">
                {t(
                  noOrganization
                    ? "auth.noOrganizationDescription"
                    : denied
                      ? "auth.deniedDescription"
                      : "auth.recoveryDescription"
                )}
              </Text>
              {state.issue && (
                <Alert color="yellow" role="alert">
                  {t(`auth.errors.${state.issue}`, {
                    defaultValue: t("auth.errors.operationFailed")
                  })}
                </Alert>
              )}
              <Group wrap="wrap">
                {denied && availableTarget && (
                  <Button component={Link} to={availableTarget}>
                    {t("client.returnAccessiblePage")}
                  </Button>
                )}
                {!denied && !noOrganization && (
                  <Button
                    loading={busy}
                    onClick={async () => {
                      if (busy) return
                      const current = capture()
                      setBusy(true)
                      try {
                        await session.restore()
                        if (current()) await router.invalidate({ sync: true })
                      } finally {
                        if (current()) setBusy(false)
                      }
                    }}
                  >
                    {t("auth.retryRecovery")}
                  </Button>
                )}
                {state.phase === "blocked" && (
                  <Button
                    variant="default"
                    loading={busy}
                    onClick={async () => {
                      if (busy) return
                      const current = capture()
                      setBusy(true)
                      try {
                        await session.retryCleanup()
                        if (current() && session.getSnapshot().phase === "anonymous")
                          await navigate({ to: "/login", replace: true })
                      } finally {
                        if (current()) setBusy(false)
                      }
                    }}
                  >
                    {t("auth.clearSession")}
                  </Button>
                )}
                <Button
                  variant="default"
                  onClick={async () => {
                    const current = capture()
                    await session.logout()
                    if (current() && session.getSnapshot().phase === "anonymous")
                      await navigate({ to: "/login", replace: true })
                  }}
                >
                  {t("auth.logout")}
                </Button>
              </Group>
            </>
          )}
        </Stack>
      </Paper>
    </Container>
  )
}
