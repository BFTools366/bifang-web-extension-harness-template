import {
  Alert,
  Anchor,
  Button,
  Container,
  Group,
  Image,
  Paper,
  PasswordInput,
  Stack,
  Text,
  TextInput,
  Title
} from "@mantine/core"
import { Link, useNavigate, useSearch } from "@tanstack/react-router"
import { useEffect, useRef, useState, type FormEvent } from "react"
import { useTranslation } from "react-i18next"
import { BRAND } from "../../brand"
import type { RegisterInput } from "../../data/api/auth/contracts"
import classes from "../client/client.module.css"
import { useAuth } from "./AuthProvider"
import { authIssue, isCancelled } from "./session"
import { accessibleReturnTo, safeReturnTo } from "./access"
import { AuthStatePanel } from "./AuthStatePanel"

/** 三种认证表单的实际字段，密码和验证码只留在组件生命周期。 */
type AuthFields = RegisterInput & { confirmation: string }
/** 页面只接收已固定的客户端认证流程。 */
export type AuthPageMode = "login" | "register" | "reset"
const emptyFields: AuthFields = {
  username: "",
  email: "",
  password: "",
  code: "",
  confirmation: ""
}

/** 单一 form 提交入口同时覆盖 Enter、校验焦点及提交防重。 */
export function AuthPage({ mode }: { mode: AuthPageMode }) {
  const { t } = useTranslation()
  const { session, state } = useAuth()
  const navigate = useNavigate()
  const search = useSearch({ strict: false }) as { returnTo?: string; notice?: string }
  const [fields, setFields] = useState<AuthFields>(emptyFields)
  const [errors, setErrors] = useState<Partial<Record<keyof AuthFields, string>>>({})
  const [issue, setIssue] = useState<string>()
  const [pending, setPending] = useState(false)
  const [sending, setSending] = useState(false)
  const [resendAt, setResendAt] = useState(0)
  const [remaining, setRemaining] = useState(0)
  const [sentEmail, setSentEmail] = useState("")
  const [mockCode, setMockCode] = useState<string>()
  const busy = useRef(false)
  const sendBusy = useRef(false)
  const alive = useRef(true)
  const request = useRef(new AbortController())
  const form = useRef<HTMLFormElement>(null)
  const summary = useRef<HTMLDivElement>(null)
  const emailValid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(fields.email)
  const passwordLabel = mode === "reset" ? "newPassword" : "password"
  const returnTo = search.returnTo === undefined ? undefined : safeReturnTo(search.returnTo)

  useEffect(() => {
    alive.current = true
    busy.current = false
    sendBusy.current = false
    setPending(false)
    setSending(false)
    request.current = new AbortController()
    return () => {
      alive.current = false
      request.current.abort()
    }
  }, [mode])
  useEffect(() => {
    if (!resendAt) {
      setRemaining(0)
      return
    }
    const update = () => setRemaining(Math.max(0, Math.ceil((resendAt - Date.now()) / 1000)))
    update()
    const timer = setInterval(update, 1000)
    return () => clearInterval(timer)
  }, [resendAt])

  /** 编辑邮箱使旧验证码状态失效，不让旧请求结果绑定到新邮箱。 */
  function changeField(field: keyof AuthFields, value: string) {
    setFields((current) => ({
      ...current,
      [field]: value,
      ...(field === "email" ? { code: "" } : {})
    }))
    setErrors((current) => ({ ...current, [field]: undefined }))
    if (field === "email") {
      request.current.abort()
      request.current = new AbortController()
      setSentEmail("")
      setMockCode(undefined)
      setResendAt(0)
      setSending(false)
      sendBusy.current = false
    }
  }
  /** 注册与找回采用已确认的新密码规则，登录保留现有账号密码语义。 */
  function validate() {
    const result: Partial<Record<keyof AuthFields, string>> = {}
    if (mode !== "reset" && !fields.username.trim())
      result.username = t("auth.errors.fieldsRequired")
    else if (mode === "register" && !/^[A-Za-z0-9_-]{3,32}$/.test(fields.username))
      result.username = t("auth.errors.usernameInvalid")
    if (mode !== "login" && !emailValid) result.email = t("auth.errors.emailInvalid")
    if (!fields.password) result.password = t("auth.errors.fieldsRequired")
    else if (mode !== "login" && !/^(?=.*[A-Za-z])(?=.*[0-9]).{8,64}$/.test(fields.password))
      result.password = t("auth.errors.passwordInvalid")
    if (mode !== "login" && (!fields.code || sentEmail !== fields.email.trim().toLowerCase()))
      result.code = t("auth.errors.codeRequired")
    else if (mode !== "login" && !/^[0-9]{1,16}$/.test(fields.code))
      result.code = t("auth.errors.codeInvalid")
    if (mode === "reset" && fields.password !== fields.confirmation)
      result.confirmation = t("auth.errors.passwordMismatch")
    setErrors(result)
    const first = Object.keys(result)[0]
    if (first) form.current?.querySelector<HTMLInputElement>(`[name="${first}"]`)?.focus()
    return first === undefined
  }
  /** 发送成功才启动服务返回的重发倒计时，失败保留输入。 */
  async function sendCode() {
    if (sendBusy.current || !emailValid || remaining > 0 || busy.current) return
    sendBusy.current = true
    setSending(true)
    setIssue(undefined)
    const email = fields.email.trim().toLowerCase()
    const signal = request.current.signal
    try {
      const result = await session.api.sendCode(
        email,
        mode === "register" ? "register" : "reset",
        signal
      )
      if (!alive.current || signal.aborted) return
      setSentEmail(email)
      setMockCode(result.mockCode)
      setFields((current) => ({ ...current, code: "" }))
      setRemaining(result.retryAfterSeconds)
      setResendAt(Date.now() + result.retryAfterSeconds * 1000)
    } catch (error) {
      if (alive.current && !signal.aborted && !isCancelled(error)) setIssue(authIssue(error))
    } finally {
      if (alive.current && !signal.aborted) {
        sendBusy.current = false
        setSending(false)
      }
    }
  }
  /** 注册和找回成功都返回登录；任何写入失败都不自动重放。 */
  async function submit(event: FormEvent) {
    event.preventDefault()
    if (busy.current || sending || !validate()) return
    busy.current = true
    setPending(true)
    setIssue(undefined)
    const signal = request.current.signal
    try {
      if (mode === "login") {
        await session.login(fields.username, fields.password)
        if (!alive.current || signal.aborted || signal !== request.current.signal) return
        if (session.getSnapshot().phase === "ready")
          await navigate({
            to: accessibleReturnTo(search.returnTo, session.getContext()),
            replace: true
          })
      } else if (mode === "register") {
        await session.api.register(fields, signal)
        if (!alive.current || signal.aborted) return
        setFields(emptyFields)
        await navigate({
          to: "/login",
          search: { notice: "registered", returnTo },
          replace: true
        })
      } else {
        const current = await session.resetPassword(
          { email: fields.email, password: fields.password, code: fields.code },
          signal
        )
        if (!alive.current || signal.aborted || !current) return
        setFields(emptyFields)
        await navigate({
          to: "/login",
          search: { notice: "reset", returnTo },
          replace: true
        })
      }
    } catch (error) {
      if (!alive.current || signal.aborted || isCancelled(error)) return
      setIssue(authIssue(error))
      queueMicrotask(() => {
        if (alive.current && !signal.aborted && signal === request.current.signal)
          summary.current?.focus()
      })
    } finally {
      if (alive.current && !signal.aborted && signal === request.current.signal) {
        busy.current = false
        setPending(false)
      }
    }
  }

  const showRecovery = ["offline", "recovery", "blocked"].includes(state.phase)
  return (
    <main id="main" tabIndex={-1} className={classes.authMain}>
      {showRecovery ? (
        <Stack w="100%">
          <AuthStatePanel />
        </Stack>
      ) : (
        <Container size="xl">
          <Paper className={classes.authLayout} radius="lg">
            <aside className={classes.authAside}>
              <Stack gap="xl" align="flex-start">
                <Group gap="md">
                  <Image src={BRAND.logoPath} alt="" w={72} h={72} />
                  <Text fw={700} className={classes.authBrandName} c="brand">
                    {t("projectName")}
                  </Text>
                </Group>
                <Text component="p" fw={700} className={classes.authTitle}>
                  {t("auth.asideTitle")}
                </Text>
                <Text c="dimmed" size="lg">
                  {t("auth.asideDescription")}
                </Text>
                <Stack gap="xs">
                  <Text fw={600}>{t("auth.identityTitle")}</Text>
                  <Text c="dimmed">{t("auth.identityDescription")}</Text>
                </Stack>
                <Stack gap="xs">
                  <Text fw={600}>{t("auth.continuityTitle")}</Text>
                  <Text c="dimmed">{t("auth.continuityDescription")}</Text>
                </Stack>
              </Stack>
            </aside>
            <section className={classes.authCard}>
              <Stack gap="lg">
                <Group gap="sm" className={classes.authMobileBrand}>
                  <Image src={BRAND.logoPath} alt="" w={32} h={32} />
                  <Text fw={700}>{t("projectName")}</Text>
                </Group>
                <div className={classes.authFormHeading}>
                  <Title order={1} size="h3">
                    {t(`auth.titles.${mode}`)}
                  </Title>
                  <Text c="dimmed" size="sm" mt="xs">
                    {t(`auth.descriptions.${mode}`)}
                  </Text>
                </div>
                {search.notice && ["registered", "reset"].includes(search.notice) && (
                  <Alert color="green" role="status">
                    {t(`auth.notices.${search.notice}`)}
                  </Alert>
                )}
                {issue && (
                  <div ref={summary} tabIndex={-1}>
                    <Alert color="red" role="alert">
                      {t(`auth.errors.${issue}`, {
                        defaultValue: t("auth.errors.operationFailed")
                      })}
                    </Alert>
                  </div>
                )}
                {state.phase === "anonymous" && state.issue && (
                  <Alert color="yellow" role="status">
                    {t(`auth.errors.${state.issue}`, {
                      defaultValue: t("auth.errors.operationFailed")
                    })}
                  </Alert>
                )}
                <form
                  ref={form}
                  className={classes.authForm}
                  noValidate
                  onSubmit={submit}
                  aria-busy={pending}
                  aria-label={t(`auth.titles.${mode}`)}
                >
                  {mode !== "reset" && (
                    <TextInput
                      label={t(mode === "login" ? "auth.loginIdentifier" : "auth.username")}
                      placeholder={t(mode === "login" ? "auth.placeholders.account" : "auth.placeholders.username")}
                      classNames={{ input: classes.authPlaceholder }}
                      name="username"
                      autoComplete="username"
                      maxLength={mode === "login" ? 128 : 32}
                      required
                      size="md"
                      radius="xl"
                      value={fields.username}
                      error={errors.username}
                      disabled={pending}
                      onChange={(event) => changeField("username", event.currentTarget.value)}
                    />
                  )}
                  {mode !== "login" && (
                    <TextInput
                      label={t("auth.email")}
                      placeholder={t("auth.placeholders.email")}
                      classNames={{ input: classes.authPlaceholder }}
                      name="email"
                      type="email"
                      autoComplete="email"
                      maxLength={128}
                      required
                      size="md"
                      radius="xl"
                      value={fields.email}
                      error={errors.email}
                      disabled={pending}
                      onChange={(event) => changeField("email", event.currentTarget.value)}
                    />
                  )}
                  <PasswordInput
                    label={t(`auth.${passwordLabel}`)}
                    placeholder={t(mode === "login" ? "auth.placeholders.password" : "auth.placeholders.newPassword")}
                    classNames={{ innerInput: classes.authPlaceholder }}
                    description={mode === "login" ? undefined : t("auth.passwordRequirement")}
                    name="password"
                    autoComplete={mode === "login" ? "current-password" : "new-password"}
                    maxLength={64}
                    required
                    size="md"
                    radius="xl"
                    value={fields.password}
                    error={errors.password}
                    disabled={pending}
                    onChange={(event) => changeField("password", event.currentTarget.value)}
                    visibilityToggleButtonProps={{ "aria-label": t("auth.togglePassword") }}
                  />
                  {mode === "reset" && (
                    <PasswordInput
                      label={t("auth.confirmPassword")}
                      placeholder={t("auth.placeholders.confirmation")}
                      classNames={{ innerInput: classes.authPlaceholder }}
                      name="confirmation"
                      autoComplete="new-password"
                      maxLength={64}
                      required
                      size="md"
                      radius="xl"
                      value={fields.confirmation}
                      error={errors.confirmation}
                      disabled={pending}
                      onChange={(event) => changeField("confirmation", event.currentTarget.value)}
                      visibilityToggleButtonProps={{
                        "aria-label": t("auth.togglePassword")
                      }}
                    />
                  )}
                  {mode !== "login" && (
                    <>
                      <div className={classes.codeRow}>
                        <TextInput
                          label={t("auth.code")}
                          placeholder={t("auth.placeholders.code")}
                          classNames={{ input: classes.authPlaceholder }}
                          name="code"
                          inputMode="numeric"
                          autoComplete="one-time-code"
                          maxLength={16}
                          required
                          size="md"
                          radius="xl"
                          value={fields.code}
                          error={errors.code}
                          disabled={pending}
                          onChange={(event) => changeField("code", event.currentTarget.value)}
                        />
                        <Button
                          type="button"
                          variant="default"
                          className={classes.wrapButton}
                          loading={sending}
                          disabled={!emailValid || remaining > 0 || pending}
                          onClick={() => {
                            void sendCode()
                          }}
                        >
                          {remaining > 0
                            ? t("auth.resendCountdown", { count: remaining })
                            : t("auth.sendCode")}
                        </Button>
                      </div>
                      <Text size="sm" c="dimmed" role={sentEmail ? "status" : undefined}>
                        {t(
                          sentEmail
                            ? mockCode
                              ? "auth.codeSent"
                              : "auth.codeDelivered"
                            : "auth.codeHint",
                          { code: mockCode }
                        )}
                      </Text>
                    </>
                  )}
                  <Button
                    type="submit"
                    size="md"
                    radius="xl"
                    className={classes.wrapButton}
                    loading={pending}
                    disabled={sending}
                  >
                    {t(`auth.actions.${mode}`)}
                  </Button>
                  <Group justify="space-between" wrap="wrap">
                    <Anchor
                      renderRoot={(props) => (
                        <Link
                          {...props}
                          to={mode === "login" ? "/register" : "/login"}
                          search={{ returnTo }}
                        />
                      )}
                    >
                      {t(mode === "login" ? "auth.needAccount" : "auth.backToLogin")}
                    </Anchor>
                    {mode === "login" && (
                      <Anchor
                        renderRoot={(props) => (
                          <Link {...props} to="/forgot-password" search={{ returnTo }} />
                        )}
                      >
                        {t("auth.forgotPassword")}
                      </Anchor>
                    )}
                  </Group>
                </form>
                <details className={classes.demoDetails}>
                  <summary>{t("auth.demoTitle")}</summary>
                  <Text size="xs" c="dimmed" mt="sm">
                    {t("auth.demoDescription")}
                  </Text>
                </details>
              </Stack>
            </section>
          </Paper>
        </Container>
      )}
    </main>
  )
}
