import { applicationHttp, requestHttpQuery } from "../client/httpRuntime"
import { ApiError } from "../client/httpClient"
import { parseServiceEmpty } from "../foundationProtocol"
import type {
  TeamApi,
  TeamContext,
  TeamInvitation,
  TeamMember,
  TeamPage,
  TeamRole
} from "./teamContracts"

/** 缺失、损坏或 GraphQL 拒绝不转为空集合，错误不包含服务原文或私有变量。 */
function invalid(): never {
  throw new ApiError({
    kind: "invalid-response",
    code: "invalidTeamResponse",
    message: "invalidTeamResponse",
    retryable: false
  })
}
/** 只收窄非数组对象；所有边界字段在当前领域校验。 */
function object(value: unknown): Record<string, unknown> {
  if (!value || typeof value !== "object" || Array.isArray(value)) return invalid()
  return value as Record<string, unknown>
}
/** 雪花 ID 必须保留精确字符串。 */
function id(value: unknown): string {
  if (typeof value !== "string" || !/^[1-9][0-9]*$/u.test(value)) return invalid()
  return value
}
/** 字符串集合不能混入空权限码。 */
function codes(value: unknown): string[] {
  if (!Array.isArray(value) || value.some((item) => typeof item !== "string" || !item.trim()))
    return invalid()
  return value as string[]
}
/** 成员身份、状态和团队关联全部由服务返回。 */
function member(value: unknown): TeamMember {
  const row = object(value)
  if (
    typeof row.displayName !== "string" ||
    !["owner", "member"].includes(String(row.role)) ||
    row.status !== "active" ||
    !Number.isSafeInteger(row.version) ||
    Number(row.version) < 1 ||
    !Array.isArray(row.roleIds) ||
    !Array.isArray(row.roleNames) ||
    row.roleNames.length !== row.roleIds.length ||
    typeof row.canRemove !== "boolean" ||
    typeof row.canAssignRoles !== "boolean"
  )
    return invalid()
  return {
    id: id(row.id),
    accountId: id(row.accountId),
    organizationId: id(row.organizationId),
    displayName: row.displayName,
    role: row.role === "owner" ? "owner" : "member",
    status: "active",
    version: Number(row.version),
    permissions: codes(row.permissions),
    roleIds: row.roleIds.map(id),
    roleNames: codes(row.roleNames),
    canRemove: row.canRemove,
    canAssignRoles: row.canAssignRoles
  }
}
/** 角色定义保留团队和版本，前端只消费有效权限码，不按角色名授权。 */
function role(value: unknown): TeamRole {
  const row = object(value)
  if (
    typeof row.code !== "string" ||
    typeof row.name !== "string" ||
    typeof row.protected !== "boolean" ||
    !Number.isSafeInteger(row.version) ||
    Number(row.version) < 1
  )
    return invalid()
  return {
    id: id(row.id),
    organizationId: id(row.organizationId),
    code: row.code,
    name: row.name,
    protected: row.protected,
    canManage: typeof row.canManage === "boolean" ? row.canManage : invalid(),
    version: Number(row.version),
    permissions: codes(row.permissions)
  }
}
/** 邀请版本和期限拒绝损坏数值，不能从客户端时间产生授权。 */
function invitation(value: unknown): TeamInvitation {
  const row = object(value)
  const status = row.status
  if (
    status !== "pending" &&
    status !== "accepted" &&
    status !== "rejected" &&
    status !== "revoked" &&
    status !== "expired"
  )
    return invalid()
  if (
    typeof row.organizationName !== "string" ||
    !Number.isSafeInteger(row.version) ||
    Number(row.version) < 1 ||
    !Number.isSafeInteger(row.expiresAt) ||
    Number(row.expiresAt) < 1
  )
    return invalid()
  return {
    id: id(row.id),
    accountId: id(row.accountId),
    organizationId: id(row.organizationId),
    organizationName: row.organizationName,
    status,
    version: Number(row.version),
    expiresAt: Number(row.expiresAt)
  }
}
/** 校验分页结果总数与本批次，不接受无边界的服务器数据。 */
function page<T extends { id: string }>(
  value: unknown,
  parse: (row: unknown) => T,
  current = 1,
  size = 100
): TeamPage<T> {
  const result = object(value)
  if (
    !Array.isArray(result.items) ||
    result.items.length > size ||
    !Number.isSafeInteger(result.total) ||
    Number(result.total) < result.items.length
  )
    return invalid()
  const items = result.items.map(parse)
  if (
    new Set(items.map((row) => row.id)).size !== items.length ||
    (items.length > 0 && (current - 1) * size + items.length > Number(result.total))
  )
    return invalid()
  return { items, total: Number(result.total) }
}
/** GraphQL 使用现有 Ky/session owner；Query 的缓存和错误反馈仍归当前 feature。 */
async function query(
  field: string,
  document: string,
  variables: Record<string, unknown>,
  signal?: AbortSignal
): Promise<unknown> {
  const response = await requestHttpQuery("graphql", {
    method: "POST",
    signal,
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ query: document, variables })
  })
  const result = object(await response.json())
  if (result.errors !== undefined) {
    if (!Array.isArray(result.errors)) return invalid()
    if (result.errors.length) {
      const error = object(result.errors[0]),
        extensions = object(error.extensions)
      const unauthorized =
        typeof extensions.code === "string" &&
        /^(auth\.|common.unauthorized)/u.test(extensions.code)
      const denied = extensions.code === "common.forbidden"
      const status = unauthorized
        ? 401
        : denied
          ? 403
          : extensions.code === "common.bad.request"
            ? 400
            : extensions.code === "common.conflict"
              ? 409
              : 500
      const code = unauthorized ? "teamIdentityRequired" : denied ? "teamDenied" : "teamQueryFailed"
      throw new ApiError({ kind: "http", code, message: code, status, retryable: status === 500 })
    }
  }
  return object(result.data)[field]
}
/** REST 快照复用受管查询入口并校验 model/data 信封，子数组不改变接口语义。 */
async function snapshot(endpoint: string, model: string, signal?: AbortSignal): Promise<unknown> {
  const response = await requestHttpQuery("api/v1/auth/teams/" + endpoint, {
    method: "GET",
    cache: "no-store",
    signal
  })
  const result = object(await response.json())
  if (result.model !== model || typeof result.request_id !== "string") return invalid()
  return result.data
}
/** REST 写入不自动重放；未知结果由页面只读核对。 */
async function command(
  endpoint: string,
  body: unknown,
  method = "POST",
  signal?: AbortSignal
): Promise<void> {
  const response = await applicationHttp.request("api/v1/auth/teams/" + endpoint, {
    method,
    signal,
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body)
  })
  if (endpoint !== "invitations") parseServiceEmpty(await response.json())
  else {
    const result = object(await response.json())
    if (result.model !== "team.invitation" || !object(result.data).id) invalid()
  }
}
/** 团队查询、切换和成员生命周期共用一个实际 adapter。 */
export const api: TeamApi = {
  /** 当前选择由服务端账号偏好恢复，本地不维护竞争选择。 */
  async read(signal): Promise<TeamContext> {
    const result = object(await snapshot("context", "team.context", signal))
    if (
      !Array.isArray(result.available) ||
      result.available.length < 1 ||
      result.available.length > 100
    )
      return invalid()
    return {
      current: id(result.current),
      available: result.available.map((value) => {
        const row = object(value)
        if (typeof row.name !== "string" || !row.name.trim()) return invalid()
        const user = member(row.user)
        const organizationId = id(row.id)
        const ownerAccountId = id(row.ownerAccountId)
        if (
          user.organizationId !== organizationId ||
          (user.role === "owner") !== (user.accountId === ownerAccountId)
        )
          return invalid()
        return { id: organizationId, name: row.name, ownerAccountId, user }
      })
    }
  },
  switch: (organizationId) => command("switch", { organizationId }),
  rename: (organizationId, name, signal) =>
    command("name", { organizationId, name }, "PUT", signal),
  /** 团队管理集合始终按当前团队查询。 */
  async members(organizationId, current, pageSize, signal) {
    return page(
      await query(
        "teamMembers",
        "query Members($org:ID!,$page:Int!,$size:Int!) { teamMembers(organizationId:$org,page:$page,pageSize:$size) { total items { id accountId organizationId displayName role status version permissions roleIds roleNames canRemove canAssignRoles } } }",
        { org: id(organizationId), page: current, size: pageSize },
        signal
      ),
      member,
      current,
      pageSize
    )
  },
  /** organizationId 为空只读取当前账号收到的邀请。 */
  async invitations(organizationId, current, pageSize, signal) {
    return page(
      await query(
        "teamInvitations",
        "query Invitations($org:ID,$page:Int!,$size:Int!) { teamInvitations(organizationId:$org,page:$page,pageSize:$size) { total items { id organizationId organizationName accountId status version expiresAt } } }",
        { org: organizationId === null ? null : id(organizationId), page: current, size: pageSize },
        signal
      ),
      invitation,
      current,
      pageSize
    )
  },
  /** 可授予范围以当前服务目录为准，不从前端页面自动制造权限码。 */
  async permissionCatalog(organizationId, signal) {
    return codes(
      await query(
        "teamPermissionCatalog",
        "query PermissionCatalog($org:ID!) { teamPermissionCatalog(organizationId:$org) }",
        { org: id(organizationId) },
        signal
      )
    )
  },
  invite: (organizationId, email, signal) =>
    command("invitations", { organizationId, email }, "POST", signal),
  resolve: (invitation, action, signal) =>
    command(
      "invitations/" + action,
      { invitationId: invitation.id, version: invitation.version },
      "POST",
      signal
    ),
  remove: (member, signal) =>
    command(
      "members/remove",
      {
        organizationId: member.organizationId,
        memberId: member.id,
        version: member.version
      },
      "POST",
      signal
    ),
  /** 角色集合由目标团队的权限门禁保护。 */
  async roles(organizationId, current, pageSize, signal) {
    return page(
      await query(
        "teamRoles",
        "query TeamRoles($org:ID!,$page:Int!,$size:Int!) { teamRoles(organizationId:$org,page:$page,pageSize:$size) { total items { id organizationId code name protected canManage version permissions } } }",
        { org: id(organizationId), page: current, size: pageSize },
        signal
      ),
      role,
      current,
      pageSize
    )
  },
  /** 成员角色候选独立授权，不能借用完整角色页权限。 */
  async assignableRoles(organizationId, signal) {
    return page(
      await query(
        "teamAssignableRoles",
        "query AssignableRoles($org:ID!) { teamAssignableRoles(organizationId:$org) { total items { id organizationId code name protected canManage version permissions } } }",
        { org: id(organizationId) },
        signal
      ),
      role
    )
  },
  nickname: (member, nickname, signal) =>
    command(
      "members/nickname",
      {
        organizationId: member.organizationId,
        memberId: member.id,
        version: member.version,
        nickname
      },
      "PUT",
      signal
    ),
  assignRoles: (member, roleIds, signal) =>
    command(
      "members/roles",
      {
        organizationId: member.organizationId,
        memberId: member.id,
        version: member.version,
        roleIds
      },
      "PUT",
      signal
    ),
  createRole: (organizationId, code, name, codes, signal) =>
    command("roles", { organizationId, code, name, codes }, "POST", signal),
  updateRole: (role, name, codes, signal) =>
    command(
      "roles",
      { organizationId: role.organizationId, roleId: role.id, version: role.version, name, codes },
      "PUT",
      signal
    ),
  deleteRole: (role, signal) =>
    command(
      "roles/remove",
      {
        organizationId: role.organizationId,
        roleId: role.id,
        version: role.version
      },
      "POST",
      signal
    )
}
/** 权限快照按路径 ID 走 REST，服务逐次绑定账号、团队与成员主体。 */
export async function readTeamPermissions(
  organizationId: string,
  subjectId: string,
  signal?: AbortSignal
): Promise<readonly string[]> {
  const result = object(
    await snapshot(
      `${id(organizationId)}/members/${id(subjectId)}/permissions`,
      "team.permissions",
      signal
    )
  )
  return codes(result.permissions)
}
