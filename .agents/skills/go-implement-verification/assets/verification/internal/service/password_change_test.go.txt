// passwordReadBarrier 冻结真实查询后的快照，使并发回归不依赖调度先后。
func passwordReadBarrier(t *testing.T, f *verificationFixture, id model.ID, count int) (<-chan struct{}, func()) {
	t.Helper()
	pool, err := f.db.DB()
	if err != nil {
		t.Fatal(err)
	}
	pool.SetMaxOpenConns(1)
	ready := make(chan struct{}, count)
	release := make(chan struct{})
	var once sync.Once
	var seen atomic.Int32
	resume := func() { once.Do(func() { close(release) }) }
	t.Cleanup(resume)
	if err := f.db.Callback().Query().After("gorm:after_query").Register("test:password_snapshot", func(tx *gorm.DB) {
		account, ok := tx.Statement.Dest.(*model.Account)
		if tx.Error != nil || !ok || account.ID != id || seen.Add(1) > int32(count) {
			return
		}
		ready <- struct{}{}
		select {
		case <-release:
		case <-tx.Statement.Context.Done():
		}
	}); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { f.db.Callback().Query().Remove("test:password_snapshot") })
	return ready, resume
}

// TestConcurrentPasswordChange 证明两个旧密码快照只有一次可以提交并吊销令牌。
func TestConcurrentPasswordChange(t *testing.T) {
	f := newVerificationFixture(t)
	account := registerFixtureAccount(t, f, "password_race")
	ready, resume := passwordReadBarrier(t, f, account.ID, 2)
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	type result struct {
		password string
		err      error
	}
	results := make(chan result, 2)
	service := NewAccountService(f.auth.accounts, f.auth.orgs)
	for _, next := range []string{"NextPassword123", "OtherPassword456"} {
		go func(password string) {
			results <- result{password, service.ChangePassword(ctx, account.ID, "TestPassword123!", password)}
		}(next)
	}
	for range 2 {
		select {
		case <-ready:
		case <-ctx.Done():
			t.Fatal("password reads did not reach barrier")
		}
	}
	resume()
	winner := ""
	conflicts := 0
	for range 2 {
		select {
		case value := <-results:
			if value.err == nil {
				winner = value.password
			} else if errors.Is(value.err, apperr.ErrAccountPasswordMismatch) {
				conflicts++
			} else {
				t.Fatal(value.err)
			}
		case <-ctx.Done():
			t.Fatal("password change did not finish")
		}
	}
	saved, err := f.auth.accounts.GetByID(ctx, account.ID)
	if err != nil || saved == nil || winner == "" || conflicts != 1 || saved.TokenVersion != account.TokenVersion+1 || !hash.VerifyPassword(saved.Password, winner) {
		t.Fatalf("unexpected concurrent result: winner=%q conflicts=%d account=%v err=%v", winner, conflicts, saved, err)
	}
}

// TestPasswordChangeCannotOverrideLogout 证明旧快照不能跨过令牌版本变更提交密码。
func TestPasswordChangeCannotOverrideLogout(t *testing.T) {
	f := newVerificationFixture(t)
	account := registerFixtureAccount(t, f, "password_logout")
	ready, resume := passwordReadBarrier(t, f, account.ID, 1)
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	results := make(chan error, 1)
	service := NewAccountService(f.auth.accounts, f.auth.orgs)
	go func() { results <- service.ChangePassword(ctx, account.ID, "TestPassword123!", "Changed123") }()
	select {
	case <-ready:
	case <-ctx.Done():
		t.Fatal("password read did not reach barrier")
	}
	if err := f.auth.accounts.BumpTokenVersion(ctx, account.ID); err != nil {
		t.Fatal(err)
	}
	resume()
	select {
	case err := <-results:
		if !errors.Is(err, apperr.ErrAccountPasswordMismatch) {
			t.Fatalf("expected conflict, got %v", err)
		}
	case <-ctx.Done():
		t.Fatal("password change did not finish")
	}
	saved, err := f.auth.accounts.GetByID(ctx, account.ID)
	if err != nil || saved == nil || saved.TokenVersion != account.TokenVersion+1 || !hash.VerifyPassword(saved.Password, "TestPassword123!") {
		t.Fatal("old password snapshot overwrote logout")
	}
}
