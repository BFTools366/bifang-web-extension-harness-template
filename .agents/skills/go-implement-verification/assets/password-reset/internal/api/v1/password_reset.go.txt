package v1

import (
	"errors"

	"github.com/gin-gonic/gin"
	"project_id/internal/apperr"
	"project_id/internal/dto"
	"project_id/internal/model"
	"project_id/internal/pkg/contextx"
	"project_id/internal/pkg/response"
)

// requestResetCode 只采用邮箱找回场景，合法目标统一响应以减少账号枚举。
func (m *accountModule) requestResetCode(c *gin.Context, req *dto.SendVerificationCodeRequest) {
	if !m.verification.Enabled() || m.verification.Mock() {
		panic(apperr.ErrVerificationChannelUnavailable)
	}
	if req.TargetType != model.VerificationTargetEmail || req.Scene != model.VerificationSceneResetPassword {
		panic(apperr.ErrVerificationSceneUnsupported)
	}
	if _, err := m.verification.NormalizeTarget(req.TargetType, req.Target); err != nil {
		panic(err)
	}
	_, err := m.verification.Send(c.Request.Context(), req, c.ClientIP())
	if err != nil && !errors.Is(err, apperr.ErrVerificationTargetNotRegistered) {
		// 不记录地址、验证码或 provider 原始错误；发送失败也不泄露账户存在性。
		if logger := contextx.Logger(c.Request.Context()); logger != nil {
			logger.WithField("event", "password_reset.delivery_not_completed").Warn("找回密码请求未完成投递")
		}
	}
	response.NoContent(c)
}
