/** 仅对应自建 Go 用户合同，已有服务须按实际合同另外映射。 */
export interface RegisterInput {
  email: string
  password: string
  code?: string
}
/** 完整资料中的成员必须来自服务端，不从组织 owner 字段推导。 */
export interface UserProfile {
  account: ServiceAccount
  organization: { id: string; name: string; ownerAccountId: string }
  organizationUser: { id: string; organizationId: string; accountId: string; role: "owner" }
}
/** 不把服务原文或凭据存进错误信息。 */
function object(value: unknown): Record<string, unknown> {
  if (!value || typeof value !== "object" || Array.isArray(value))
    throw new IdentityError("invalidIdentity")
  return value as Record<string, unknown>
}
/** Snowflake ID 必须保持十进制字符串，不能转为 JavaScript number。 */
function id(value: unknown): string {
  if (typeof value !== "string" || !/^[1-9][0-9]*$/u.test(value))
    throw new IdentityError("invalidIdentity")
  return value
}
/** 要求真实字符串字段；空昵称由页面选择 username 显示。 */
function text(value: unknown): string {
  if (typeof value !== "string") throw new IdentityError("invalidIdentity")
  return value
}
/** 上游资料与本地成员扩展共用完整账号 DTO，原始 org 允许 null。 */
function serviceProfile(value: unknown): value is ServiceProfile {
  return (
    isServiceProfile(value) &&
    id(value.account.id) === value.account.id &&
    (value.org === null ||
      (id(value.org.id) === value.org.id && id(value.org.owner_id) === value.org.owner_id))
  )
}
/** 单对象解包复用 Go wire 校验，不把任意 2xx 当成认证成功。 */
async function data<T>(
  response: Response,
  model: string,
  valid: (value: unknown) => value is T
): Promise<T> {
  return parseServiceObject(await response.json(), model, valid).data
}

/** 解析并验证完整资料，公开资料读取和会话校验共用同一事实。 */
function profileData(result: ServiceProfile): UserProfile {
  const account = result.account,
    org = object(result.org),
    member = object(object(result).organization_user)
  const accountId = id(account.id),
    orgId = id(org.id)
  if (
    id(org.owner_id) !== accountId ||
    id(member.account_id) !== accountId ||
    id(member.organization_id) !== orgId ||
    member.role !== "owner"
  )
    throw new IdentityError("invalidIdentity")
  return {
    account,
    organization: { id: orgId, name: text(org.name), ownerAccountId: accountId },
    organizationUser: { id: id(member.id), organizationId: orgId, accountId, role: "owner" }
  }
}

/** 公开登录与指定凭据验证使用短生命周期 client，不修改已接受的应用会话。 */
export function createGoIdentityAdapter(
  config?: HttpClientOptions["config"],
  profileMode: "workspace" | "upstream" = "workspace"
) {
  /** 复用受管 Ky/超时/取消/错误边界；401 只返回该次验证，失效归 session owner。 */
  async function request(
    endpoint: string,
    method: string,
    body?: unknown,
    credential?: string,
    signal?: AbortSignal
  ) {
    let failure: ServiceFailure | undefined
    const client = createHttpClient({
      config,
      fetch: async (input, init) => {
        const response = await requestGoTransport(input, init)
        if (!response.ok) {
          try {
            failure = parseServiceFailure(await response.clone().json(), response.status)
          } catch {
            failure = new ServiceFailure("invalid_service_response", response.status)
          }
        }
        return response
      },
      unauthorizedHandler: {
        /** 执行 handleUnauthorized 类成员的处理逻辑。 */ handleUnauthorized() {}
      },
      extractAuthToken: () => undefined
    })
    if (credential) client.setAuthToken(credential)
    try {
      return await client.request("api/v1/auth/" + endpoint, {
        method,
        signal,
        feedback: "owner",
        ...(body === undefined
          ? {}
          : { headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) })
      })
    } catch (error) {
      if (error instanceof ApiError && error.status === 401 && credential)
        throw new IdentityError("expired")
      if (error instanceof ApiError && error.kind === "http" && failure)
        throw new ApiError({
          kind: "http",
          code: failure.code,
          message: failure.code,
          status: failure.status,
          retryable: error.retryable,
          ignoreGlobalFeedback: error.ignoreGlobalFeedback
        })
      throw error
    }
  }
  /** 注册不自动登录；Workspace 装配器另在同一事务中创建 owner 成员。 */
  async function register(input: RegisterInput, signal?: AbortSignal): Promise<string> {
    const account = await data(
      await request(
        "register",
        "POST",
        {
          email: normalizeEmail(input.email),
          password: input.password,
          code: input.code,
          code_type: "email"
        },
        undefined,
        signal
      ),
      "account",
      isServiceAccount
    )
    return id(account.id)
  }
  /** 注册发码与登录共用匿名传输，不携带已接受的应用凭据。 */
  async function sendRegistrationCode(
    email: string,
    signal?: AbortSignal
  ): Promise<RegistrationCodeReceipt> {
    const response = await request(
      "verification-code",
      "POST",
      { target: normalizeEmail(email), target_type: "email", scene: "register" },
      undefined,
      signal
    )
    try {
      const result = parseServiceObject(
        await response.json(),
        "verification.code",
        (value): value is Record<string, unknown> =>
          !!value && typeof value === "object" && !Array.isArray(value)
      )
      return registrationCodeReceipt(result.data, APP_CONFIG.environment === "development")
    } catch {
      throw new ApiError({
        kind: "invalid-response",
        code: "invalidVerificationResponse",
        message: "invalidVerificationResponse",
        retryable: false
      })
    }
  }
  /** 原样读取 Go Harness 的本人资料；空默认组织不是成员授权。 */
  async function readServiceProfile(
    credential: string,
    signal?: AbortSignal
  ): Promise<ServiceProfile> {
    return data(
      await request("profile", "GET", undefined, credential, signal),
      "account",
      serviceProfile
    )
  }
  /** Workspace 完整资料必须另有服务端成员扩展，不能从 owner_id 生成权限。 */
  async function readProfile(credential: string, signal?: AbortSignal): Promise<UserProfile> {
    return profileData(await readServiceProfile(credential, signal))
  }
  /** 登录和刷新均返回完整令牌对；采用与持久化由唯一 session owner 决定。 */
  async function tokenResult(response: Response): Promise<ServiceAuthResult> {
    const result = await data(response, "auth.token", isServiceAuthResult)
    id(result.account.id)
    if (result.account.status !== 1) throw new IdentityError("invalidIdentity")
    return result
  }
  /** 显式刷新是公开请求，不附旧 access token，也不声称撤销旧 refresh。 */
  async function refresh(refreshToken: string, signal?: AbortSignal): Promise<ServiceAuthResult> {
    if (!refreshToken || refreshToken.length > 8192 || /[^\x21-\x7e]/u.test(refreshToken))
      throw new IdentityError("invalidIdentity")
    return tokenResult(
      await request("refresh", "POST", { refresh_token: refreshToken }, undefined, signal)
    )
  }
  /** 改密会撤销全账号令牌；成功后由调用方使当前会话失效，不自动重放。 */
  async function changePassword(
    credential: string,
    oldPassword: string,
    newPassword: string,
    signal?: AbortSignal
  ): Promise<void> {
    parseServiceEmpty(
      await (
        await request(
          "password",
          "PUT",
          {
            old_password: oldPassword,
            new_password: newPassword
          },
          credential,
          signal
        )
      ).json()
    )
  }

  const api: LoginApi = {
    /** 执行 login 类成员的处理逻辑。 */
    async login(input, signal) {
      const result = await tokenResult(await request("login", "POST", input, undefined, signal))
      const pair = result.token
      // 当前状态机只接受 access token；refresh 持久化与轮换由明确需求另行实现。
      return pair.access_token
    },
    /** 执行 validate 类成员的处理逻辑。 */
    async validate(credential, signal) {
      try {
        const profile = await readServiceProfile(credential, signal)
        if (profile.account.status !== 1) throw new IdentityError("invalidIdentity")
        if (profileMode === "workspace") profileData(profile)
        return {
          id: profile.account.id,
          displayName: profile.account.nickname || profile.account.username
        }
      } catch (error) {
        if (error instanceof ApiError && ["network", "timeout", "aborted"].includes(error.kind))
          throw error
        if (error instanceof IdentityError) throw error
        throw new IdentityError("invalidIdentity")
      }
    },
    /** 执行 logout 类成员的处理逻辑。 */
    async logout(credential) {
      parseServiceEmpty(await (await request("logout", "POST", undefined, credential)).json())
    }
  }
  return {
    api,
    register,
    sendRegistrationCode,
    readProfile,
    readServiceProfile,
    refresh,
    changePassword
  }
}
const adapter = createGoIdentityAdapter()
export const api = adapter.api
export const registerAccount = adapter.register
export const sendRegistrationCode = adapter.sendRegistrationCode
/** 与 Go password 校验标签一致，只用于新账号密码，不限制旧账号登录。 */
export const registrationPasswordPolicy = {
  messageKey: "registrationPasswordGoRule",
  /** Go len(string) 按 UTF-8 字节计数，不能用 JS 的 UTF-16 长度替代。 */
  validate(password: string): boolean {
    const length = new TextEncoder().encode(password).length
    return length >= 8 && length <= 64 && /[A-Za-z]/u.test(password) && /[0-9]/u.test(password)
  }
}
/** 资料页面复用已接受会话的 HTTP owner，不读取或暴露凭据存储。 */
export async function readUserProfile(signal?: AbortSignal): Promise<UserProfile> {
  return profileData(
    await data(await requestHttpQuery("api/v1/auth/profile", { signal }), "account", serviceProfile)
  )
}
